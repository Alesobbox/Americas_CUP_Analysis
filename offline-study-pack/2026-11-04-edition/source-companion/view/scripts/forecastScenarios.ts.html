24async function acquireWorkspaceLock(root: string): Promise<WorkspaceLock> {
25 const directory = path.join(root, '.forecast')
26 await mkdir(directory, { recursive: true })
27 const filename = path.join(directory, 'scenario-generation.lock')
28 const record = { version: 1, pid: process.pid, token: randomUUID(), started_at_ms: Date.now(), child_pid: undefined as number | undefined }
29 const create = async () => {
30 const handle = await open(filename, 'wx')
31 try { await handle.writeFile(JSON.stringify(record), 'utf8') } finally { await handle.close() }
32 }
33 try { await create() } catch (error) {
34 if ((error as NodeJS.ErrnoException).code !== 'EEXIST') throw error
35 let old: string
36 try { old = await readFile(filename, 'utf8') } catch { throw new ScenarioBusyError() }
37 let owner: { version?: number; pid?: number; child_pid?: number } = {}
38 try { owner = JSON.parse(old) } catch { /* A crashed initial write cannot have launched a publisher. */ }
39 const validPid = (value: unknown): value is number => typeof value === 'number' && Number.isInteger(value) && value > 0
40 // A child can outlive its server. Both must have exited before recovery.
41 // Version 1 children cannot publish until their PID is written and acknowledged.
42 const abandoned = validPid(owner.pid)
43 ? !alive(owner.pid) && (validPid(owner.child_pid) ? !alive(owner.child_pid) : owner.version === 1)
44 : Date.now() - (await stat(filename)).mtimeMs > 10 * 60 * 1000
45 if (!abandoned || await readFile(filename, 'utf8') !== old) throw new ScenarioBusyError()
46 try { await unlink(filename); await create() } catch { throw new ScenarioBusyError() }
47 }
48 return {
49 childReady: async (pid) => {
50 const handle = await open(filename, 'r+')
51 try {
52 const current = JSON.parse(await handle.readFile('utf8'))
53 if (current.token !== record.token) throw new Error('Forecast preparation ownership changed. Please retry.')
54 record.child_pid = pid
55 const bytes = Buffer.from(JSON.stringify(record))
56 await handle.truncate(0)
57 await handle.write(bytes, 0, bytes.length, 0)
58 await handle.sync()
59 } finally { await handle.close() }
60 },
61 release: async () => {
62 try {
63 const current = JSON.parse(await readFile(filename, 'utf8'))
64 if (current.token === record.token) await unlink(filename)
65 } catch { /* Never remove another job's lock. */ }
66 },
67 }

70export function validateScenarioInput(value: unknown): ScenarioInput {
71 if (!value || typeof value !== 'object' || Array.isArray(value)) throw new Error('Enter a race and exact wind conditions.')
72 const input = value as Record<string, unknown>
73 if (Object.keys(input).some((key) => !['race_id', 'tws_kt', 'direction_offset_deg'].includes(key))) throw new Error('Unexpected forecast input.')
74 if (typeof input.race_id !== 'string' || !/^[a-zA-Z0-9_-]{1,128}$/.test(input.race_id)) throw new Error('Choose a valid race.')
75 if (typeof input.tws_kt !== 'number' || !Number.isFinite(input.tws_kt) || input.tws_kt < 0 || input.tws_kt > 80) throw new Error('Wind speed must be between 0 and 80 knots.')
76 if (typeof input.direction_offset_deg !== 'number' || !Number.isFinite(input.direction_offset_deg) || Math.abs(input.direction_offset_deg) > 180) throw new Error('Direction offset must be between −180° and +180°.')
77 if (input.direction_offset_deg !== 0) throw new Error('The current forecast model supports a fixed course direction (0° offset). A direction response has not passed validation.')
78 return input as unknown as ScenarioInput
79}

91function pythonRunner(root: string): Runner {
92 return (input, progress, childReady) => new Promise((resolve, reject) => {
93 const local = path.join(root, '.venv', process.platform === 'win32' ? 'Scripts/python.exe' : 'bin/python')
94 const python = process.env.REGATTA_PYTHON || (existsSync(local) ? local : 'python')
95 // Constant bootstrap, separate validated argv, no shell. The publisher waits
96 // until the shared lock records its PID; EOF before that means no publication.
97 const child = spawn(python, ['-u', '-c', scenarioBootstrap, 'build', '--race', input.race_id,
98 '--tws-knots', String(input.tws_kt), '--direction-offset-deg', String(input.direction_offset_deg)],
99 { cwd: root, shell: false, windowsHide: true, stdio: ['pipe', 'pipe', 'pipe'] })
100 let output = '', error = '', timedOut = false
101 const timeout = setTimeout(() => { timedOut = true; child.kill() }, 10 * 60 * 1000)
102 child.stdin.on('error', () => { /* An exited child can close the handshake pipe. */ })
103 child.once('spawn', () => {
104 if (!child.pid || !childReady) { child.kill(); reject(new Error('Forecast preparation ownership is unavailable. Please retry.')); return }
105 void childReady(child.pid).then(() => child.stdin.end('start\n'), () => {
106 child.kill(); reject(new Error('The forecast workspace could not be reserved. Please retry.'))
107 })
108 })
109 child.stdout.on('data', (chunk) => { output = (output + String(chunk)).slice(-16384) })
110 child.stderr.on('data', (chunk) => {
111 const text = String(chunk); error = (error + text).slice(-4000)
112 if (text.includes('Verifying dataset')) progress('Checking the recorded race data and preparing the exact scenario…')
113 })
114 child.once('error', () => { clearTimeout(timeout); reject(new Error('The local forecasting service could not start. Retry, or use a published scenario.')) })
115 child.once('close', (code) => {
116 clearTimeout(timeout)
117 if (code !== 0) {
118 const reason = timedOut || code === 124 ? 'Forecast preparation took too long. Retry, or use a published scenario.'
119 : /Raw source changed|cache content hash|Unsealed forecasting/.test(error) ? 'The local forecasting data needs a refresh before this scenario can be prepared.'
120 : 'This forecast could not be prepared. Retry, or use the recorded start conditions.'
121 reject(new Error(reason)); return
122 }
123 try {
124 const result = JSON.parse(output.trim().split(/\r?\n/).at(-1) ?? '{}') as ScenarioResult
125 if (!artifactPath.test(result.path) || !['available', 'estimated', 'unavailable'].includes(result.status)) throw new Error()
126 resolve(result)
127 } catch { reject(new Error('The forecasting service returned an incomplete result. Please retry.')) }
128 })
129 })
