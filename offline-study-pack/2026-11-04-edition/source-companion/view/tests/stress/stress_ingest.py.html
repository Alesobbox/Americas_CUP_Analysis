48 def test_pair_359_to_1(self):
49 """Angle pair (359°, 1°): Shortest arc is 2° clockwise across 0° North, midpoint must be 0.0°."""
50 # Check midpoint
51 mid = interpolate_circular_deg(150, 100, 200, 359.0, 1.0)
52 self.assertAlmostEqual(mid, 0.0, places=6)
53
54 # Check fine trajectory samples: u in [0, 1]
55 for step in range(11):
56 u = step / 10.0
57 t = 100 + u * 100
58 val = interpolate_circular_deg(t, 100, 200, 359.0, 1.0)
59 # Delta is +2°. Angle should be (359 + u * 2) % 360
60 expected = (359.0 + u * 2.0) % 360.0
61 self.assertAlmostEqual(val, expected, places=5)
62 # Ensure it never goes through the long arc (e.g., ~180°)
63 self.assertTrue(val >= 359.0 or val <= 1.0)
