93async function verifyAnalysisPalette(label) {
94 const metrics = await page.evaluate(() => {
95 const root = document.querySelector('.tactical-dashboard-root .analysis-modal')
96 const canvas = document.createElement('canvas'), context = canvas.getContext('2d', { willReadFrequently: true })
97 canvas.width = canvas.height = 1
98 const rgba = (css) => { context.clearRect(0, 0, 1, 1); context.fillStyle = '#000'; context.fillStyle = css; context.fillRect(0, 0, 1, 1); return [...context.getImageData(0, 0, 1, 1).data] }
99 const background = (element) => {
100 for (let node = element; node; node = node.parentElement) { const color = rgba(getComputedStyle(node).backgroundColor); if (color[3] === 255) return color }
101 return [255, 255, 255, 255]
102 }
103 const luminance = (color) => color.slice(0, 3).map((part) => part / 255).map((part) => part <= .04045 ? part / 12.92 : ((part + .055) / 1.055) ** 2.4).reduce((sum, part, index) => sum + part * [.2126, .7152, .0722][index], 0)
104 const contrast = (a, b) => { const left = luminance(a), right = luminance(b); return (Math.max(left, right) + .05) / (Math.min(left, right) + .05) }
105 const surfaces = ['.modal-head', '.analysis-main-tabs', '.analysis-section', '.analysis-metric', '.analysis-definitions', '.debrief-item', '.analysis-controls select'].flatMap((selector) => [...root.querySelectorAll(selector)].slice(0, 3).map((element) => ({ selector, background: background(element) })))
106 const text = ['.modal-head h2', '.analysis-section-heading h4', '.analysis-main-tabs button', '.analysis-table th', '.analysis-metric strong', '.analysis-metric-meta', '.analysis-definitions summary', '.debrief-item h5', '.radar-label', '.analysis-state.error'].flatMap((selector) => [...root.querySelectorAll(selector)].slice(0, 4).map((element) => {
107 const style = getComputedStyle(element), foreground = rgba(element instanceof SVGElement ? style.fill : style.color), backdrop = background(element)
108 return { selector, color: foreground, background: backdrop, contrast: contrast(foreground, backdrop) }
109 }))
110 return { rootBackground: background(root), colorScheme: getComputedStyle(root).colorScheme, surfaces, text }
111 })
112 assert.deepEqual(metrics.rootBackground.slice(0, 3), [255, 255, 255], `${label}: Analysis must be white`)
113 assert.equal(metrics.colorScheme, 'light')
114 for (const surface of metrics.surfaces) assert.ok(surface.background.slice(0, 3).every((part) => part >= 235), `${label}: dark ${surface.selector}`)
115 for (const sample of metrics.text) assert.ok(sample.contrast >= 4.5, `${label}: insufficient ${sample.selector} contrast ${sample.contrast.toFixed(2)}`)
116 visualMetrics.push({ label, ...metrics })
117}
