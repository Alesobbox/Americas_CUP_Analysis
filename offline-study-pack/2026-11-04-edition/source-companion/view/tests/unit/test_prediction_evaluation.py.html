116 def test_primary_improvement_must_hold_under_both_race_and_day_weighting(self):
117 reference_rows = [{"date": "A", "race_id": f"A-{k}", "log_loss": 1., "brier": .4}
118 for k in range(3)] + [{"date": "B", "race_id": "B", "log_loss": 1., "brier": .4}]
119 candidate_rows = [dict(row, log_loss=.8 if row["date"] == "A" else 1.5) for row in reference_rows]
120 reference, candidate = aggregate(reference_rows), aggregate(candidate_rows)
121 self.assertLess(candidate["race_equal"]["log_loss"], reference["race_equal"]["log_loss"])
122 self.assertGreater(candidate["day_equal"]["log_loss"], reference["day_equal"]["log_loss"])
123 self.assertFalse(promotion(candidate, reference, applicable=("brier",))["promoted"])
