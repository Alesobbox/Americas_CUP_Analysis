34 def test_mps_to_knots_conversion(self):
35 """Conversion factor must strictly adhere to 1 m/s = 1.94384449 kt."""
36 self.assertAlmostEqual(0.0 * MPS_TO_KNOTS, 0.0, places=6)
37 self.assertAlmostEqual(1.0 * MPS_TO_KNOTS, 1.94384449, places=8)
38 self.assertAlmostEqual(10.0 * MPS_TO_KNOTS, 19.4384449, places=6)
39 # Typical AC40 foiling speed: 15 m/s (~29.16 kt)
40 self.assertAlmostEqual(15.0 * MPS_TO_KNOTS, 29.15766735, places=6)
41 # AC40 top speed: 22 m/s (~42.76 kt)
42 self.assertAlmostEqual(22.0 * MPS_TO_KNOTS, 42.76457878, places=6)

53 def test_circular_shortest_arc_interpolation(self):
54 """Circular interpolation must follow the shortest arc and wrap correctly across 0/360."""
55 # Crossing 0 deg North: 350 deg to 10 deg -> midpoint must be 0.0 deg
56 mid1 = interpolate_circular_deg(150, 100, 200, 350.0, 10.0)
57 self.assertAlmostEqual(mid1, 0.0, places=6)

59 # Crossing 0 deg North in reverse: 10 deg to 350 deg -> midpoint must be 0.0 deg
60 mid2 = interpolate_circular_deg(150, 100, 200, 10.0, 350.0)
61 self.assertAlmostEqual(mid2, 0.0, places=6)
62
63 # 359 deg to 1 deg at 50% -> 0.0 deg
64 mid3 = interpolate_circular_deg(150, 100, 200, 359.0, 1.0)
65 self.assertAlmostEqual(mid3, 0.0, places=6)
66
67 # Standard non-wrapping arc: 40 deg to 60 deg -> midpoint 50 deg
68 mid4 = interpolate_circular_deg(150, 100, 200, 40.0, 60.0)
69 self.assertAlmostEqual(mid4, 50.0, places=6)
70
71 # Wrapping across 180 deg: 170 deg to 190 deg -> midpoint 180 deg

75 def test_compute_twa(self):
76 """TWA must correctly compute wind angle relative to boat heading in [-180, +180]."""
77 # Wind from 68 deg, boat heading 115 deg -> TWA = -47 deg (Port Tack)
78 twa_port = compute_twa(68.0, 115.0)
79 self.assertAlmostEqual(twa_port, -47.0, places=6)
80
81 # Wind from 68 deg, boat heading 21 deg -> TWA = +47 deg (Starboard Tack)
82 twa_stbd = compute_twa(68.0, 21.0)
83 self.assertAlmostEqual(twa_stbd, +47.0, places=6)
84
85 # Head to wind: heading == twd -> TWA = 0.0
86 self.assertAlmostEqual(compute_twa(50.0, 50.0), 0.0, places=6)
87
88 # Dead run: heading == twd + 180 -> TWA = 180.0 or -180.0
89 dead_run = compute_twa(40.0, 220.0)
90 self.assertAlmostEqual(abs(dead_run), 180.0, places=6)

92 def test_compute_vmg(self):
93 """VMG along target axis must project SOG using cosine of angle difference."""
94 target_axis = 40.0
95 sog = 30.0
96
97 # COG aligned with axis -> full VMG
98 self.assertAlmostEqual(compute_vmg(sog, 40.0, target_axis), 30.0, places=6)
99
100 # COG perpendicular to axis -> zero VMG
101 self.assertAlmostEqual(compute_vmg(sog, 130.0, target_axis), 0.0, places=6)
102
103 # COG directly opposite to axis -> negative full VMG
104 self.assertAlmostEqual(compute_vmg(sog, 220.0, target_axis), -30.0, places=6)
105
106 # COG at 45 deg to axis -> SOG * cos(45 deg)
107 self.assertAlmostEqual(compute_vmg(sog, 85.0, target_axis), 30.0 * math.sqrt(2) / 2.0, places=5)
