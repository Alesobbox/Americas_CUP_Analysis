31def tiny_source(day, *, session_end=True):
32 gun = int(datetime.fromisoformat(day + "T12:00:00").replace(tzinfo=timezone.utc).timestamp() * 1000)
33 gate0 = {"port_asset_id": "START_P", "starboard_asset_id": "START_S"}
34 gate1 = {"port_asset_id": "TOP_P", "starboard_asset_id": "TOP_S"}
35 marks = {"START_P": (-25, 0), "START_S": (25, 0), "TOP_P": (-25, 500), "TOP_S": (25, 500)}
36 assets = [{"id": "BOAT_A", "abbreviation": "TEST_A", "official_name": "Synthetic A", "type": "ASSET_TYPE_YACHT"},
37 {"id": "BOAT_B", "abbreviation": "TEST_B", "official_name": "Synthetic B", "type": "ASSET_TYPE_YACHT"}]
38 assets += [{"id": name, "type": "ASSET_TYPE_RACE_MARK"} for name in marks]
39 session = {"timestamp": gun - 1000, "race_start": gun, "window_start": gun - 1000,
40 "window_end": gun + 50000, "assets": assets,
41 "competitors": [{"asset_id": "BOAT_A"}, {"asset_id": "BOAT_B"}],
42 "official_wind_asset_id": "WIND", "race_course": {"course_length": 500,
43 "sequences": [{"gate": gate0}, {"gate": gate1}, {"gate": gate0}]}}
44 records = [{"session": session}]

46 def position(asset, second, x, y):
47 return {"telemetry": {"asset_id": asset, "timestamp": gun + second * 1000,
48 "gnss_position": {"latitude": 50 + y / 111320, "longitude": 1 + x / (111320 * np.cos(np.deg2rad(50)))}}}
49
50 for second in range(51):
51 for name, (x, y) in marks.items():
52 records.append(position(name, second, x, y))
53 records.append({"telemetry": {"asset_id": "WIND", "timestamp": gun + second * 1000,
54 "true_wind_speed": {"value": 12 / 1.943844}}})
55 for index, boat in enumerate(("BOAT_A", "BOAT_B")):
56 turn, finish = 20 + index * 2, 40 + index * 4
57 y = 500 * second / turn if second <= turn else max(0, 500 * (finish - second) / (finish - turn))
58 records.append(position(boat, second, index * 4, y))
59 for name, value in (("speed_over_ground", 25), ("course_over_ground", 0 if second < turn else 180)):
60 records.append({"telemetry": {"asset_id": boat, "timestamp": gun + second * 1000,
61 name: {"value": value}}})
62 if second in (0, turn, finish):
63 gates = [{"leg_number": 0, "gate_time": gun}]
64 if second >= turn:
65 gates.append({"leg_number": 1, "gate_time": gun + turn * 1000})
66 if second >= finish:
67 gates.append({"leg_number": 2, "gate_time": gun + finish * 1000})
68 records.append({"competitor": {"asset_id": boat, "timestamp": gun + second * 1000,
69 "classification": "FINISHED" if second == finish else "RACING",
70 "start_state": {"started": True}, "leg_timings": gates}})
71 if session_end:
72 records.append({"session": dict(session, timestamp=gun + 50000)})
73 return gun, records
