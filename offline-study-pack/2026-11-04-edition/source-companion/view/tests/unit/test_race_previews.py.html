28 def test_preview_is_deterministic_bounded_and_tied_to_the_correct_source(self):
29 with tempfile.TemporaryDirectory() as temporary:
30 root = Path(temporary)
31 data = self.fixture(root)
32 before = (data / "replays/race_one.json").read_bytes()
33 result = write_previews(root)
34 output = data / "presentation/race_previews.json"
35 first = output.read_bytes()
36 write_previews(root)
37 self.assertEqual(first, output.read_bytes())
38 self.assertEqual(before, (data / "replays/race_one.json").read_bytes())
39 preview = result["races"]["race_one"]
40 self.assertEqual(preview["boat_count"], 2)
41 self.assertEqual(preview["race_type"], "RACE_TYPE_MATCH")
42 self.assertEqual(preview["source_sha256"], hashlib.sha256(before).hexdigest())
43 self.assertTrue(all(0 <= x <= 420 and 0 <= y <= 250 for x, y in preview["boundary"]))
44 inventory = build_inventory(root)
45 entry = next(item for item in inventory["artifacts"] if item["id"] == "presentation/race_previews.json")
46 self.assertEqual(entry["url"], "/data/presentation/race_previews.json")
47 self.assertFalse(entry["primary_download"])
