330 def test_tied_n_rank_sum_identity_exhaustive(self):
331 """realized_rank_sum = ordinary_sum + M*(M-1)/2. Rank sum strictly exceeds ordinary_sum iff M >= 2."""
332 for n in range(2, 9):
333 ordinary_sum = n * (n + 1) // 2
334 for m in range(0, n + 1):
335 # k finishers get ranks 1..k, m nonfinishers get rank n
336 k = n - m
337 ranks = list(range(1, k + 1)) + [n] * m
338 realized_sum = sum(ranks)
339 expected_sum = ordinary_sum + m * (m - 1) // 2
340 self.assertEqual(realized_sum, expected_sum, f"Failed for N={n}, M={m}")
341 if m >= 2:
342 self.assertGreater(realized_sum, ordinary_sum)
343 else:
344 self.assertEqual(realized_sum, ordinary_sum)

398 cov = np.asarray(model.covariance)
399 # Contrast variance: Var(beta_0 - beta_1) = cov[0,0] + cov[1,1] - 2*cov[0,1]
400 contrast_var_01 = cov[0, 0] + cov[1, 1] - 2 * cov[0, 1]
401
402 rng = np.random.default_rng(42)
403 draws = model.sample_log_worth(["A", "B", "C"], rng, n=50_000)
404 # Mean across boats for each draw should be zero (identifiable contrast subspace)
405 np.testing.assert_allclose(draws.mean(axis=1), 0.0, atol=1e-10)
406 # Sample contrast variance
407 sample_diff = draws[:, 0] - draws[:, 1]
408 self.assertAlmostEqual(np.var(sample_diff), contrast_var_01, delta=0.03)

410 def test_cup_first_to_seven_series_dependence_check(self):
411 """Persistent win prob in {0.5, 0.9} yields ~75.00% series win vs ~93.76% for independent mean 0.7."""
412 def series_win(p):
413 # Best of 13: sum_{k=7}^{13} binom(13, k) p^k (1-p)^{13-k}
414 return sum(math.comb(13, k) * (p**k) * ((1 - p)**(13 - k)) for k in range(7, 14))
415
416 # Persistent probability check
417 w_50 = series_win(0.5)
418 w_90 = series_win(0.9)
419 persistent_series_win = 0.5 * w_50 + 0.5 * w_90
420 independent_series_win = series_win(0.7)
421
422 self.assertAlmostEqual(w_50, 0.5)
423 self.assertAlmostEqual(persistent_series_win, 0.75, delta=0.001)
424 self.assertAlmostEqual(independent_series_win, 0.9376, delta=0.001)
