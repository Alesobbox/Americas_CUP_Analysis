69 def setUp(self):
70 # Bottom gate line: RM8 at (-100, 50), RM6 at (100, -50)

74 def test_trajectory_intersecting_center(self):
75 """Trajectory cutting through midpoint of gate segment intersects with u=0.5, s=0.5."""
76 p_prev = (-20.0, -20.0)
77 p_curr = (20.0, 20.0)
78 crossed, u, s, cx, cy = intersect_segments_2d(p_prev, p_curr, self.m_port, self.m_stbd)

82 self.assertAlmostEqual(cx, 0.0, places=3)
83 self.assertAlmostEqual(cy, 0.0, places=3)

85 def test_port_mark_selection_classification(self):
86 """Crossing near RM8 (s < 0.5) is classified as Port mark."""
87 # Trajectory near RM8: crossing at s = 0.20
88 # Target point on gate: m_port + 0.2 * (m_stbd - m_port)
89 gx = self.m_port[0] + 0.20 * (self.m_stbd[0] - self.m_port[0])
90 gy = self.m_port[1] + 0.20 * (self.m_stbd[1] - self.m_port[1])
91 p_prev = (gx - 10.0, gy - 10.0)
92 p_curr = (gx + 10.0, gy + 10.0)
93
94 crossed, u, s, cx, cy = intersect_segments_2d(p_prev, p_curr, self.m_port, self.m_stbd)
95 self.assertTrue(crossed)
96 self.assertLess(s, 0.50)
97 mark_choice = "RM8" if s < 0.50 else "RM6"
98 side_choice = "PORT" if s < 0.50 else "STARBOARD"
99 self.assertEqual(mark_choice, "RM8")
100 self.assertEqual(side_choice, "PORT")
101
102 def test_starboard_mark_selection_classification(self):
103 """Crossing near RM6 (s >= 0.5) is classified as Starboard mark."""
104 # Target point on gate at s = 0.85
105 gx = self.m_port[0] + 0.85 * (self.m_stbd[0] - self.m_port[0])
106 gy = self.m_port[1] + 0.85 * (self.m_stbd[1] - self.m_port[1])
107 p_prev = (gx - 10.0, gy - 10.0)
108 p_curr = (gx + 10.0, gy + 10.0)
109
110 crossed, u, s, cx, cy = intersect_segments_2d(p_prev, p_curr, self.m_port, self.m_stbd)
111 self.assertTrue(crossed)
112 self.assertGreaterEqual(s, 0.50)
113 mark_choice = "RM8" if s < 0.50 else "RM6"
114 side_choice = "PORT" if s < 0.50 else "STARBOARD"
115 self.assertEqual(mark_choice, "RM6")
116 self.assertEqual(side_choice, "STARBOARD")
