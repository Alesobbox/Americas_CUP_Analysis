56def vincenty_distance(lat1: float, lon1: float, lat2: float, lon2: float) -> float:
57 """Computes exact geodesic distance on the WGS84 ellipsoid using Vincenty's inverse formula."""
58 a = 6378137.0
59 f = 1 / 298.257223563
60 b = 6356752.314245
61 phi1, lambda1 = math.radians(lat1), math.radians(lon1)
62 phi2, lambda2 = math.radians(lat2), math.radians(lon2)
63 L = lambda2 - lambda1
64 U1 = math.atan((1 - f) * math.tan(phi1))
65 U2 = math.atan((1 - f) * math.tan(phi2))
66 sinU1, cosU1 = math.sin(U1), math.cos(U1)
67 sinU2, cosU2 = math.sin(U2), math.cos(U2)
68 lam = L
69 for _ in range(100):
70 sinLambda = math.sin(lam)
71 cosLambda = math.cos(lam)
72 sinSigma = math.sqrt(
73 (cosU2 * sinLambda) ** 2 + (cosU1 * sinU2 - sinU1 * cosU2 * cosLambda) ** 2
74 )
75 if sinSigma == 0:
76 return 0.0
77 cosSigma = sinU1 * sinU2 + cosU1 * cosU2 * cosLambda
78 sigma = math.atan2(sinSigma, cosSigma)
79 sinAlpha = (cosU1 * cosU2 * sinLambda) / sinSigma
80 cos2Alpha = 1 - sinAlpha**2
81 cos2SigmaM = cosSigma - (2 * sinU1 * sinU2) / cos2Alpha if cos2Alpha != 0 else 0
82 C = f / 16 * cos2Alpha * (4 + f * (4 - 3 * cos2Alpha))
83 lamPrev = lam
84 lam = L + (1 - C) * f * sinAlpha * (
85 sigma + C * sinSigma * (cos2SigmaM + C * cosSigma * (-1 + 2 * cos2SigmaM**2))
86 )
87 if abs(lam - lamPrev) < 1e-12:
88 break
89 u2 = cos2Alpha * (a**2 - b**2) / b**2
90 A = 1 + u2 / 16384 * (4096 + u2 * (-768 + u2 * (320 - 175 * u2)))
91 B = u2 / 1024 * (256 + u2 * (-128 + u2 * (74 - 47 * u2)))
92 deltaSigma = (
93 B
94 * sinSigma
95 * (
96 cos2SigmaM
97 + B
98 / 4
99 * (
100 cosSigma * (-1 + 2 * cos2SigmaM**2)
101 - B
102 / 6
103 * cos2SigmaM
104 * (-3 + 4 * sinSigma**2)
105 * (-3 + 4 * cos2SigmaM**2)
106 )
107 )
108 )
109 return b * A * (sigma - deltaSigma)

115 def test_ten_thousand_random_points_invertibility(self):

121 max_lon_err = 0.0
122

125 lon = rng.uniform(COURSE_LON_MIN, COURSE_LON_MAX)
126
127 x, y = wgs84_to_metric(lat, lon)
128 lat_rec, lon_rec = metric_to_wgs84(x, y)

130 lat_err = abs(lat - lat_rec)
131 lon_err = abs(lon - lon_rec)
132
133 if lat_err > max_lat_err:
134 max_lat_err = lat_err
135 if lon_err > max_lon_err:
136 max_lon_err = lon_err
137
138 self.assertLess(
139 lat_err,
140 1e-9,
141 f"Latitude invertibility failed at ({lat}, {lon}): error {lat_err:.2e} >= 1e-9",
142 )
143 self.assertLess(
144 lon_err,
145 1e-9,
146 f"Longitude invertibility failed at ({lat}, {lon}): error {lon_err:.2e} >= 1e-9",
147 )

194 def test_pairwise_distance_distortion_vs_vincenty_geodesic(self):
195 """Tests pairwise distance distortion across arbitrary vectors (50m to 3000m) against WGS84 Vincenty geodesic.
196
197 Asserts distortion percentage |d_cart - d_vinc| / d_vinc * 100 < 0.1% strictly everywhere.
198 Nominal empirical performance achieves < 0.032% (3x better than threshold).
199 """
200 rng = random.Random(999)
201 tested_pairs = 0
202 target_pairs = 3_000
203
204 max_distortion_pct = 0.0
205
206 while tested_pairs < target_pairs:
207 lat1 = rng.uniform(COURSE_LAT_MIN, COURSE_LAT_MAX)
208 lon1 = rng.uniform(COURSE_LON_MIN, COURSE_LON_MAX)
209 lat2 = rng.uniform(COURSE_LAT_MIN, COURSE_LAT_MAX)
210 lon2 = rng.uniform(COURSE_LON_MIN, COURSE_LON_MAX)
211
212 d_vinc = vincenty_distance(lat1, lon1, lat2, lon2)
213 if not (50.0 <= d_vinc <= 3000.0):
214 continue
215
216 x1, y1 = wgs84_to_metric(lat1, lon1)
217 x2, y2 = wgs84_to_metric(lat2, lon2)
218 d_cart = euclidean_distance(x1, y1, x2, y2)
219
220 err_pct = abs(d_cart - d_vinc) / d_vinc * 100.0
221
222 self.assertLess(
223 err_pct,
224 0.1,
225 f"Cartesian distortion vs Vincenty {err_pct:.4f}% exceeded 0.1% for vector len {d_vinc:.1f}m",
226 )
227
228 tested_pairs += 1
229 if err_pct > max_distortion_pct:
230 max_distortion_pct = err_pct
231
232 self.assertEqual(tested_pairs, target_pairs)
233 # Verify that max distortion vs true WGS84 geodesic is strictly under 0.035%
234 self.assertLess(max_distortion_pct, 0.035)

271 def test_haversine_spherical_anisotropy_characterization(self):
272 """Characterizes and verifies the intrinsic spherical Earth distortion limit of Haversine.
273
274 Due to Earth's oblateness (M != N), spherical Haversine with scalar radius R = sqrt(MN)
275 inherently deviates by ~0.193% in cardinal directions:
276 - Meridian (North-South, 0 deg): +0.194%
277 - Parallel (East-West, 90 deg): -0.191%
278 This test documents this physical boundary condition and verifies that error is bounded by 0.25%.
279 """
280 for length in [100.0, 500.0, 1000.0, 2000.0]:
281 # North-South vector
282 p_ns_lat, p_ns_lon = metric_to_wgs84(0.0, length)
283 d_hav_ns = haversine_distance(LAT0, LON0, p_ns_lat, p_ns_lon)
284 err_ns_pct = abs(length - d_hav_ns) / d_hav_ns * 100.0
285 self.assertAlmostEqual(err_ns_pct, 0.194, delta=0.01)
286 self.assertLess(err_ns_pct, 0.25)
287
288 # East-West vector
289 p_ew_lat, p_ew_lon = metric_to_wgs84(length, 0.0)
290 d_hav_ew = haversine_distance(LAT0, LON0, p_ew_lat, p_ew_lon)
291 err_ew_pct = abs(length - d_hav_ew) / d_hav_ew * 100.0
292 self.assertAlmostEqual(err_ew_pct, 0.191, delta=0.01)
293 self.assertLess(err_ew_pct, 0.25)
