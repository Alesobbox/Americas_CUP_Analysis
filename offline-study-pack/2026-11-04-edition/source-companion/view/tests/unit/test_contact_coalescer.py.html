20 def test_sat_rotation_invariance(self):
21 """Rotating two colliding boats and their offset vector preserves collision and overlap."""
22 base_dist = 10.0 # 10m < 14.83m span
23 test_angles = [0.0, 45.0, 90.0, 180.0, 270.0]
24 results = []
25
26 for angle in test_angles:
27 rad = math.radians(angle)
28 boat_a_pos = (0.0, 0.0)
29 # Offset perpendicular to heading (starboard)
30 boat_b_pos = (base_dist * math.cos(rad), -base_dist * math.sin(rad))
31
32 poly_a = transform_local_to_world(DEFAULT_KEEP_CLEAR_BORDER, boat_a_pos[0], boat_a_pos[1], angle)
33 poly_b = transform_local_to_world(DEFAULT_KEEP_CLEAR_BORDER, boat_b_pos[0], boat_b_pos[1], angle)
34
35 has_contact, depth = test_sat_overlap(poly_a, poly_b)
36 area = compute_polygon_intersection_area(poly_a, poly_b)
37 results.append((has_contact, depth, area))
38
39 ref_contact, ref_depth, ref_area = results[0]
40 self.assertTrue(ref_contact)
41 self.assertGreater(ref_depth, 0.0)
42 self.assertGreater(ref_area, 0.0)
43
44 for has_c, d, a in results[1:]:
45 self.assertTrue(has_c)
46 self.assertAlmostEqual(d, ref_depth, places=2)
47 self.assertAlmostEqual(a, ref_area, places=1)

49 def test_lateral_separation_thresholds(self):
50 """15.0m lateral separation has no contact, 12.0m lateral separation has contact."""
51 # 15.0m: Keep Clear Border span is 14.83m -> clearance of 0.17m
52 poly_a_15 = transform_local_to_world(DEFAULT_KEEP_CLEAR_BORDER, 0.0, 0.0, 0.0)
53 poly_b_15 = transform_local_to_world(DEFAULT_KEEP_CLEAR_BORDER, 15.0, 0.0, 0.0)
54 has_contact_15, depth_15 = test_sat_overlap(poly_a_15, poly_b_15)
55 self.assertFalse(has_contact_15)
56 self.assertEqual(depth_15, 0.0)
57
58 # 12.0m: Keep Clear Border intersects
59 poly_b_12 = transform_local_to_world(DEFAULT_KEEP_CLEAR_BORDER, 12.0, 0.0, 0.0)
60 has_contact_12, depth_12 = test_sat_overlap(poly_a_15, poly_b_12)
61 self.assertTrue(has_contact_12)
62 self.assertAlmostEqual(depth_12, 2.73, places=1)
