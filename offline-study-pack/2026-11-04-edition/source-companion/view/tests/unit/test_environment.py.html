20def response(source="air", *, times=None):
21 times = times or [START, START + 3600]
22 units = {"time": "unixtime"}
23 hourly = {"time": times}
24 if source == "air":
25 units["temperature_2m"] = "°C"

27 else:
28 units.update(wave_height="m", sea_surface_temperature="°C")
