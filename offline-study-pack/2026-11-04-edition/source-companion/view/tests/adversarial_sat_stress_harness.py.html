28def run_rotation_sweep_metrics():
29 print("\n" + "=" * 80)
30 print(" [1] 360-DEGREE ROTATION SWEEP METRICS (1 DEGREE INCREMENTS)")
31 print("=" * 80)
32
33 base_dist = 10.0 # 10m < 14.83m span (overlap ~4.83m)
34 depths = []
35 areas = []
36 nan_count = 0
37 inf_count = 0
38
39 for deg in range(360):
40 rad = math.radians(deg)
41 boat_a_pos = (150.0, -80.0)
42 boat_b_pos = (
43 boat_a_pos[0] + base_dist * math.cos(rad),
44 boat_a_pos[1] - base_dist * math.sin(rad),
45 )
46
47 poly_a = transform_local_to_world(DEFAULT_KEEP_CLEAR_BORDER, boat_a_pos[0], boat_a_pos[1], deg)
48 poly_b = transform_local_to_world(DEFAULT_KEEP_CLEAR_BORDER, boat_b_pos[0], boat_b_pos[1], deg)
49
50 has_c, d = test_sat_overlap(poly_a, poly_b)
51 a = compute_polygon_intersection_area(poly_a, poly_b)
52
53 if math.isnan(d) or math.isnan(a):
54 nan_count += 1
55 if math.isinf(d) or math.isinf(a):
56 inf_count += 1
57
58 depths.append(d)
59 areas.append(a)
60
61 mean_d = sum(depths) / len(depths)
62 var_d = sum((x - mean_d) ** 2 for x in depths) / len(depths)
63 std_d = math.sqrt(var_d)
64
65 mean_a = sum(areas) / len(areas)
66 var_a = sum((x - mean_a) ** 2 for x in areas) / len(areas)
67 std_a = math.sqrt(var_a)
68
69 print(f" Orientations tested: 360 angles (0° to 359°)")
70 print(f" Overlap Depth (m): Min={min(depths):.4f}, Max={max(depths):.4f}, Mean={mean_d:.4f}, StdDev={std_d:.6f}")
71 print(f" Overlap Area (m²): Min={min(areas):.4f}, Max={max(areas):.4f}, Mean={mean_a:.4f}, StdDev={std_a:.6f}")
72 print(f" NaN Count: {nan_count}, Inf Count: {inf_count}")
73 print(f" Max Depth Discrepancy vs Theoretical (4.83m): {abs(mean_d - 4.83):.4f}m")
74
75 # Separated boats check
76 sep_dist = 20.0
77 false_positives = 0
78 for deg in range(360):
79 rad = math.radians(deg)
80 b_pos = (sep_dist * math.cos(rad), -sep_dist * math.sin(rad))
81 poly_a = transform_local_to_world(DEFAULT_KEEP_CLEAR_BORDER, 0.0, 0.0, deg)
