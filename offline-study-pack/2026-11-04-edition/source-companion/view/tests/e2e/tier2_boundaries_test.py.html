34def point_in_polygon_2d(x: float, y: float, polygon: List[Tuple[float, float]]) -> bool:
35 """Ray casting algorithm for testing if a 2D point is inside a polygon."""
36 n = len(polygon)
37 inside = False
38 p1x, p1y = polygon[0]
39 for i in range(1, n + 1):
40 p2x, p2y = polygon[i % n]
41 if y > min(p1y, p2y):
42 if y <= max(p1y, p2y):
43 if x <= max(p1x, p2x):
44 if p1y != p2y:
45 xinters = (y - p1y) * (p2x - p1x) / (p2y - p1y) + p1x
46 if p1x == p2x or x <= xinters:
47 inside = not inside
48 p1x, p1y = p2x, p2y
49 return inside

101 def test_zero_velocity_handling(self):
102 """Zero speed (0.0 m/s = 0.0 kt) yields 0 VMG, and time-to-mark handles gracefully."""
103 sog_kt = mps_to_knots(0.0)
104 vmg_kt = sog_kt * math.cos(0.0)
105 self.assertEqual(sog_kt, 0.0)
106 self.assertEqual(vmg_kt, 0.0)
107
108 # Time to mark with floor prevents ZeroDivisionError
109 dist_m = 500.0
110 vmg_floor_mps = 8.0 # 15.5 kt standard broadcasting floor
111 safe_time_s = dist_m / max(knots_to_mps(vmg_kt), vmg_floor_mps)
112 self.assertAlmostEqual(safe_time_s, dist_m / 8.0, places=2)
113
114 def test_creeping_speed_boundary(self):
115 """Creeping speed (0.05 kt = ~0.026 m/s) does not crash rate of closure."""
116 pos_a = (0.0, 0.0)
117 pos_b = (50.0, 0.0)
118 vel_a = (knots_to_mps(0.05), 0.0)
119 vel_b = (0.0, 0.0)
120 sep = 50.0
121 rate_of_closure = (vel_b[0] - vel_a[0]) * (pos_a[0] - pos_b[0]) / sep
122 self.assertAlmostEqual(rate_of_closure, knots_to_mps(0.05), places=5)
123
124 def test_maximum_foiling_speed_above_45_knots(self):
125 """Speed exceeding 45 knots (e.g. 48.6 kt = 25.0 m/s) updates position without overflow."""
126 speed_mps = 25.0
127 sog_kt = mps_to_knots(speed_mps)
128 self.assertGreater(sog_kt, 45.0)
129
130 # 100ms displacement at 25 m/s is 2.5m
131 disp_100ms = speed_mps * 0.100
132 self.assertAlmostEqual(disp_100ms, 2.5, places=3)
