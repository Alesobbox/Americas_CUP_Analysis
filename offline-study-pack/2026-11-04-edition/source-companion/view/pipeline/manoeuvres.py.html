1"""Tack and Gybe Manoeuvre Detector with Hysteresis and Performance Analytics.
2
3Implements:
41. True Wind Angle (TWA) zero/pi sign transition detection:
5 - Tack: Bow turns through the eye of the wind (TWA crosses through 0 deg).
6 - Gybe: Stern turns through the eye of the wind (TWA crosses through +/-180 deg).
72. Rate of turn (ROT) and angular deflection hysteresis:
8 - Heading deflection magnitude >= 50-60 deg.
9 - Peak ROT >= 5.0 deg/s.
10 - Speed dip >= 3.0 kt (1.5 m/s).
11 - Tack stability duration >= 2.0-2.5 seconds (prevents chatter/false triggers).
12 - ZERO false positives during steady-state straight runs with small heading corrections (<15 deg).
133. Performance metric derivation:
14 - Entry speed (SOG at turn onset).
15 - Minimum speed during manoeuvre.
16 - Speed loss (entry - min).
17 - Manoeuvre duration.
18 - Recovery time (time from min speed until 90% of entry speed recovered).
19"""

40def circular_shortest_diff_deg(from_deg: float, to_deg: float) -> float:
41 """Computes shortest signed angular difference from from_deg to to_deg in [-180, 180)."""
42 return ((to_deg - from_deg + 180.0) % 360.0) - 180.0
43
44
45def normalize_angle_180(deg: float) -> float:
46 """Normalizes angle to [-180.0, +180.0) degrees."""
47 return ((deg + 180.0) % 360.0) - 180.0

53 def __init__(
54 self,
55 min_heading_change_deg: float = 50.0,
56 min_rot_deg_s: float = 5.0,
57 min_speed_loss_kt: float = 2.5,
58 min_tack_stability_s: float = 2.0,
59 min_manoeuvre_separation_s: float = 8.0,
60 sample_interval_ms: int = 100,
61 ):
62 self.min_heading_change_deg = min_heading_change_deg
63 self.min_rot_deg_s = min_rot_deg_s
64 self.min_speed_loss_kt = min_speed_loss_kt
65 self.min_tack_stability_s = min_tack_stability_s
66 self.min_manoeuvre_separation_s = min_manoeuvre_separation_s
67 self.sample_interval_ms = sample_interval_ms
68 self.sample_rate_hz = 1000.0 / sample_interval_ms

70 def detect_manoeuvres_for_boat(
71 self,
72 timestamps: List[int],
73 heading_deg: List[float],
74 sog_kt: List[float],
75 twa_deg: List[float],
76 race_start_ms: Optional[int] = None,
77 race_finish_ms: Optional[int] = None,
78 ) -> List[Manoeuvre]:
79 """Scans time series telemetry to extract validated tacks and gybes."""
80 n = len(timestamps)
81 if n < 30:
82 return []
83
84 dt_s = self.sample_interval_ms / 1000.0
85 stability_samples = max(5, int(round(self.min_tack_stability_s * self.sample_rate_hz)))
86 min_separation_samples = int(round(self.min_manoeuvre_separation_s * self.sample_rate_hz))
87
88 # Compute smoothed rate of turn (ROT) in deg/s using 3-sample central difference
89 rot_deg_s = [0.0] * n
90 for i in range(1, n - 1):
91 dh = circular_shortest_diff_deg(heading_deg[i - 1], heading_deg[i + 1])
92 rot_deg_s[i] = abs(dh) / (2.0 * dt_s)
93
94 manoeuvres: List[Manoeuvre] = []
95 last_manoeuvre_idx = -min_separation_samples
96
97 # Candidate scan: identify TWA sign transitions with stability
98 i = stability_samples
99 while i < n - stability_samples:
100 t = timestamps[i]
101
102 # Only detect during racing window if provided
103 if race_start_ms is not None and t < race_start_ms:
104 i += 1
105 continue
106 if race_finish_ms is not None and t > race_finish_ms:
107 break
108
109 # Enforce minimum separation from prior detected manoeuvre
110 if i - last_manoeuvre_idx < min_separation_samples:
111 i += 1
112 continue
113
114 twa_curr = normalize_angle_180(twa_deg[i])
115 twa_prev = normalize_angle_180(twa_deg[i - stability_samples])
116 twa_next = normalize_angle_180(twa_deg[i + stability_samples])
117
118 # Check if there is a stable sign transition
119 is_tack_candidate = False
120 is_gybe_candidate = False
121
122 # TACK: TWA crosses through 0 deg (|TWA| < 90 deg before and after, opposite signs)
123 if (
124 abs(twa_prev) < 95.0
125 and abs(twa_next) < 95.0
126 and (twa_prev * twa_next < -10.0) # Distinct sign change
127 and abs(twa_curr) < 40.0 # Near the eye of the wind
128 ):
129 is_tack_candidate = True
130
131 # GYBE: TWA crosses through +/-180 deg (|TWA| > 85 deg before and after, opposite signs)
132 elif (
133 abs(twa_prev) > 85.0
134 and abs(twa_next) > 85.0
135 and (twa_prev * twa_next < -10.0) # Opposite signs across 180/-180
136 and abs(twa_curr) > 130.0 # Stern through wind
137 ):
138 is_gybe_candidate = True
139
140 if not (is_tack_candidate or is_gybe_candidate):
141 i += 1
142 continue
143
144 # Candidate found at index i. Now evaluate surrounding window [-6s, +6s] (60 samples)
145 win_half = min(60, min(i, n - 1 - i))
146 w_start = i - win_half
147 w_end = i + win_half
148
149 # Peak ROT in window
150 peak_rot = max(rot_deg_s[w_start : w_end + 1])
151 if peak_rot < self.min_rot_deg_s:
152 i += 1
153 continue
154
155 # Find turn onset (start of turn where ROT rises > 2.0 deg/s)
156 turn_start_idx = i
157 while turn_start_idx > w_start and rot_deg_s[turn_start_idx] > 2.0:
158 turn_start_idx -= 1
159
160 # Find turn completion (where ROT drops back < 2.0 deg/s)
161 turn_end_idx = i
162 while turn_end_idx < w_end and rot_deg_s[turn_end_idx] > 2.0:
163 turn_end_idx += 1
164
165 if turn_end_idx <= turn_start_idx:
166 i += 1
167 continue
168
169 # Total angular deflection across the turn
170 hdg_start = heading_deg[turn_start_idx]
171 hdg_end = heading_deg[turn_end_idx]
172 turn_delta = abs(circular_shortest_diff_deg(hdg_start, hdg_end))
173
174 if turn_delta < self.min_heading_change_deg:
175 i += 1
176 continue
177
178 # Speed performance profile
179 # Entry speed: speed before turn begins
180 entry_idx = max(0, turn_start_idx - 5)
181 entry_speed = max(sog_kt[entry_idx : turn_start_idx + 1]) if turn_start_idx >= entry_idx else sog_kt[turn_start_idx]
182
183 # Minimum speed in turn window
184 min_speed = min(sog_kt[turn_start_idx : turn_end_idx + 1])
185 min_speed_idx = turn_start_idx + sog_kt[turn_start_idx : turn_end_idx + 1].index(min_speed)
186
187 speed_loss = entry_speed - min_speed
188 if speed_loss < self.min_speed_loss_kt:
189 i += 1
190 continue
191
192 # Recovery time: time from min_speed until speed recovers to 90% of entry speed
193 recovery_target = 0.90 * entry_speed
194 rec_idx = min_speed_idx + 1
195 max_rec_window = min(n - 1, min_speed_idx + int(15.0 * self.sample_rate_hz))
196 recovered = False
197 while rec_idx <= max_rec_window:
198 if sog_kt[rec_idx] >= recovery_target:
199 recovered = True
200 break
201 rec_idx += 1
202
203 if recovered:
204 recovery_time_s = (timestamps[rec_idx] - timestamps[min_speed_idx]) / 1000.0
205 else:
206 recovery_time_s = (timestamps[max_rec_window] - timestamps[min_speed_idx]) / 1000.0
207
208 duration_s = (timestamps[turn_end_idx] - timestamps[turn_start_idx]) / 1000.0
209
210 m_type = "TACK" if is_tack_candidate else "GYBE"
211 manoeuvre = Manoeuvre(
212 type=m_type,
213 timestamp=timestamps[i],
214 entry_speed_kt=round(entry_speed, 2),
215 min_speed_kt=round(min_speed, 2),
216 speed_loss_kt=round(speed_loss, 2),
217 recovery_time_s=round(recovery_time_s, 2),
218 duration_s=round(duration_s, 2),
219 turn_delta_deg=round(turn_delta, 1),
220 )
221 manoeuvres.append(manoeuvre)
222 last_manoeuvre_idx = turn_end_idx
223 i = turn_end_idx + 5
224
225 return manoeuvres
