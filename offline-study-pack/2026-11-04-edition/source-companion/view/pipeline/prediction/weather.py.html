1"""Explicit scenario support and issue/availability/valid-time forecast contract.
2
3No reanalysis is accepted as a forecast that a crew could have known. The MVP
4uses raw observed wind at the cutoff and modeled future variation; archive
5forecast records may be supplied locally through this contract.
6"""
7from __future__ import annotations
8from dataclasses import dataclass
9import math
10
11
12@dataclass(frozen=True)
13class ForecastWind:
14 issued_at_ms: int
15 available_at_ms: int
16 valid_at_ms: int
17 tws_kt: float
18 twd_deg: float
19 source: str
20 product: str = "archived_forecast"
21
22 def __post_init__(self):
23 if (any(not isinstance(value, int) or isinstance(value, bool)
24 for value in (self.issued_at_ms, self.available_at_ms, self.valid_at_ms))
25 or self.product != "archived_forecast" or not self.source
26 or self.available_at_ms < self.issued_at_ms or self.valid_at_ms < self.issued_at_ms
27 or not math.isfinite(self.tws_kt) or self.tws_kt < 0
28 or not math.isfinite(self.twd_deg) or not 0<=self.twd_deg<360):
29 raise ValueError("Invalid available forecast; reanalysis is not a historical forecast")

32def available_run(records, cutoff_ms):
33 eligible=[r for r in records if r.available_at_ms <= cutoff_ms and r.issued_at_ms <= cutoff_ms]
34 if not eligible:
35 return []
36 issue=max(r.issued_at_ms for r in eligible)
37 return sorted([r for r in eligible if r.issued_at_ms==issue],key=lambda r:r.valid_at_ms)

40def condition_support(transitions, *, tws_kt, direction_offset_deg):
41 if not math.isfinite(tws_kt) or tws_kt<0 or not math.isfinite(direction_offset_deg):
42 return False,"Conditions must be finite, with nonnegative wind speed"
43 if abs(direction_offset_deg)>15:
44 return False,"Direction exceeds the declared ±15 degree fixed-course support"
45 matching=[r for r in transitions if not r.get("practice")
46 and r.get("competitive", True) is not False and r.get("valid", True) is not False
47 and isinstance(r.get("wind_knots"), (int, float)) and math.isfinite(r["wind_knots"])
48 and r["wind_knots"] >= 0 and abs(r["wind_knots"]-tws_kt)<=2]
49 if len({r["date"] for r in matching})<2 or len({r["race_id"] for r in matching})<3:
50 return False,"Fewer than two earlier competitive dates and three races within ±2 knots"
51 # Direction responses are not measured by a wind-speed-only reference.
52 if direction_offset_deg != 0:
53 return False,"A course-relative direction response has not passed validation"
54 return True,None
