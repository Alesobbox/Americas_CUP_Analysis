22def hazard_integral(u: float | np.ndarray, legs: int, shape: str = "constant") -> np.ndarray:
23 """Normalized H(u), so H(0)=0 and H(1)=1 for either declared shape."""
24 if legs < 1 or shape not in {"constant", "transition"}:
25 raise ValueError("Expected a positive leg count and constant/transition hazard")
26 values = np.clip(np.asarray(u, dtype=float), 0.0, 1.0)
27 if shape == "constant":
28 return values
29 edges, cumulative = _hazard_grid(legs)
30 return np.interp(values, edges, cumulative)

33def _hazard_grid(legs: int) -> tuple[np.ndarray, np.ndarray]:
34 edges = np.array(sorted({0.0, 1.0, 1.0 / legs,
35 *[(k + 0.95) / legs for k in range(legs)],
36 *[(k + 1) / legs for k in range(legs)]}))
37 midpoint = (edges[:-1] + edges[1:]) / 2
38 first_leg = midpoint < 1.0 / legs
39 pre_gate = (midpoint * legs) % 1.0 >= 0.95
40 # Union, not a double multiplier in overlapping first-leg/pre-gate zones.
41 weights = np.where(first_leg | pre_gate, 2.0, 1.0)
42 cumulative = np.r_[0.0, np.cumsum(np.diff(edges) * weights)]
43 return edges, cumulative / cumulative[-1]

54def _uniform_marginals(p: np.ndarray, fixed_order: Sequence[int]) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
55 """Exact uniform-order marginals conditional on each draw's completion risk.
56
57 The polynomial is Poisson-binomial, hence sums all finisher subsets without
58 adding numerical reference-order sampling noise. Rows of p may have shared
59 Beta latent risks; independence is only conditional on that risk draw.
60 """
61 n, boats = p.shape
62 means = np.zeros((n, boats))
63 ranks = np.full((n, boats), float(boats))
64 locked = set(fixed_order)
65 if fixed_order:
66 means[:, fixed_order[0]] = 1.0
67 no_finisher = np.zeros(n)
68 for place, index in enumerate(fixed_order, 1):
69 ranks[:, index] = place
70 free = [j for j in range(boats) if j not in locked]
71 total = p[:, free].sum(axis=1)
72 for index in free:
73 ranks[:, index] = boats * (1 - p[:, index]) + p[:, index] * (
74 len(fixed_order) + 1 + 0.5 * (total - p[:, index]))
75 else:
76 no_finisher = np.prod(1.0 - p, axis=1)
77 total = p.sum(axis=1)
78 for index in range(boats):
79 count_distribution = np.ones((n, 1))
80 for other in range(boats):
81 if other == index:
82 continue
83 previous = count_distribution
84 count_distribution = np.zeros((n, previous.shape[1] + 1))
85 count_distribution[:, :-1] += previous * (1 - p[:, other, None])
86 count_distribution[:, 1:] += previous * p[:, other, None]
87 means[:, index] = p[:, index] * (count_distribution / np.arange(1, boats + 1)).sum(axis=1)
88 ranks[:, index] = boats * (1 - p[:, index]) + p[:, index] * (
89 1 + 0.5 * (total - p[:, index]))
90 return means.mean(axis=0), no_finisher.mean(), ranks.mean(axis=0)

154def analytic_plackett_luce_expected_ranks(worth: Sequence[float] | np.ndarray,
155 completion_prob: Sequence[float] | np.ndarray) -> np.ndarray:
156 """Exact deterministic expected tied-N ranks under Plackett-Luce order and independent completion.
157
158 expected_rank[i] = N * (1 - s[i]) + s[i] * (1 + sum(s[j] * w[j] / (w[i] + w[j]) for j != i))
159 """
160 w = np.asarray(worth, dtype=float)
161 s = np.asarray(completion_prob, dtype=float)
162 if w.shape != s.shape or w.ndim != 1 or len(w) == 0:
163 raise ValueError("worth and completion_prob must be 1D arrays of the same nonempty length")
164 if np.any(w <= 0) or np.any(~np.isfinite(w)):
165 raise ValueError("Plackett-Luce worth values must be positive and finite")
166 if np.any(s < 0) or np.any(s > 1) or np.any(~np.isfinite(s)):
167 raise ValueError("Completion probabilities must lie in [0, 1]")
168 n = len(w)
169 if n == 1:
170 return np.array([1.0])
171 pairwise = w[None, :] / (w[:, None] + w[None, :])
172 np.fill_diagonal(pairwise, 0.0)
173 rival_ahead = pairwise @ s
174 return n * (1.0 - s) + s * (1.0 + rival_ahead)

462 - float((state or {}).get(b, {}).get("remaining_fraction", 1.))) / total_legs
463 for b in boats])
464 lead = progress - progress.mean()
465 state_features = dict(zip(boats, lead / np.sqrt(np.maximum(1 - progress, .01)))) if state else None
466 scale = (discrimination_scale if discrimination_scale is not None
467 else getattr(strength, "discrimination_scale", 1.0))
468 if hasattr(strength, "sample_log_worth") and parameter_uncertainty:
469 logits = strength.sample_log_worth(boats, rng, n, wind_knots, state_features,
470 state_coefficient=state_coefficient)
471 elif hasattr(strength, "log_worth"):
472 logits = strength.log_worth(boats, wind_knots, state_features, state_coefficient=state_coefficient)[None, :]
473 else:
474 # Also accepts the past-fitted conditional PL benchmark.
475 nested = {b: {"lead_fraction": float(lead[j]), "remaining_fraction": float(1-progress[j])}
476 for j, b in enumerate(boats)}
477 p = strength.predict_win_probabilities(boats, state_features=nested)
478 logits = np.log([max(float(p[b]), 1e-300) for b in boats])[None, :]
479 scaled_logits = logits * scale
480 worth = np.exp(np.clip(scaled_logits - np.max(scaled_logits, axis=1, keepdims=True), -700, 0))
481 clocks = rng.exponential(size=(n, len(boats))) / worth
482 assert base.finish_times_s is not None and base.ranks is not None and base.winners is not None
483 clocks[~np.isfinite(base.finish_times_s)] = math.inf
484 locked = sorted([j for j, b in enumerate(boats)
485 if str((state or {}).get(b, {}).get("terminal_status", "")).lower() == "finished"],
486 key=lambda j: (state or {})[boats[j]].get("finished_at_s", math.inf))
487 if known_winner is not None and boats.index(known_winner) not in locked:
488 locked = [boats.index(known_winner), *locked]
489 for place, j in enumerate(locked):
490 clocks[:, j] = -len(locked) + place
491 ranks, winners = _ranks_and_winners(clocks, rng.random(clocks.shape))
492 win = np.bincount(winners[winners >= 0], minlength=len(boats)) / n
493 no_finisher = float(np.mean(winners < 0))
494 expected = ranks.mean(axis=0)
495 probabilities = {b: float((1 - TAIL_WEIGHT) * win[j] + TAIL_WEIGHT * base.win_prob_by_boat[b])
496 for j, b in enumerate(boats)}
497 positions = {b: float((1 - TAIL_WEIGHT) * expected[j] + TAIL_WEIGHT * base.expected_position_by_boat[b])
498 for j, b in enumerate(boats)}
499 none = (1 - TAIL_WEIGHT) * no_finisher + TAIL_WEIGHT * base.no_finisher_prob
500 mixture = rng.random(n) < TAIL_WEIGHT
501 assumptions = [
502 "Pooled ridge PL complete-order reference; identity calibration",
503 "Laplace-posterior strength draws" if parameter_uncertainty else "MAP strength (no parameter uncertainty)",
504 "Nonfinish scored N; sampled complete orders share the completion posterior",
505 "Course-exposure survival conditioning; no predicted travel-time clock",
506 f"Diffusion-lead live term, coefficient {state_coefficient:g}" if state else "Pre-race static strength",
507 f"Calibrated discrimination scale {scale:g}" if scale != 1.0 else "Unit PL dispersion",
508 f"Joint uniform-reference mixture weight {TAIL_WEIGHT:g}",
509 ]
510 return RaceSimulation(True, base.status, None, boats, probabilities, float(none), positions,
511 n, seed, 1.96 * math.sqrt(.25/n), conditional_complete,
512 assumptions,
513 np.where(mixture[:, None], base.ranks, ranks) if return_samples else None,
514 None, None,
515 base.failure_course_fractions if return_samples else None,
516 np.where(mixture, base.winners, winners) if return_samples else None)
