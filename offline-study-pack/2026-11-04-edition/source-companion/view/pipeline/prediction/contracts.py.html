24def atomic_json(path: Path, value):
25 path.parent.mkdir(parents=True, exist_ok=True)
26 temporary = None
27 try:
28 with tempfile.NamedTemporaryFile(dir=path.parent, prefix="."+path.name+".", suffix=".tmp", delete=False) as stream:
29 temporary = Path(stream.name)
30 stream.write(canonical(value)+b"\n")
31 stream.flush()
32 os.fsync(stream.fileno())
33 for attempt in range(8):
34 try:
35 os.replace(temporary, path)
36 break
37 except PermissionError:
38 # Windows readers/antivirus can briefly deny replacement. Never
39 # fall back to truncating the last successfully published file.
40 if attempt == 7:
41 raise
42 time.sleep(min(.1*2**attempt, 1.))
43 finally:
44 if temporary is not None:
45 temporary.unlink(missing_ok=True)

56 def require(condition, message):
57 if not condition:
58 raise ValueError("Cup scenario evidence: " + message)
59 def integer(number):
60 return isinstance(number, int) and not isinstance(number, bool) and number >= 0
61 def probability(number):
62 return isinstance(number, (int, float)) and not isinstance(number, bool) and math.isfinite(number) and 0 <= number <= 1
63 require(isinstance(analysis, dict) and analysis.get("schema_version") == "cup-scenarios-1", "unsupported schema")
64 teams = {team["id"] for team in value["teams"]}
65 for field in ("probabilities", "win_prob_by_team"):
66 headline = value.get(field)
67 require(isinstance(headline, dict) and set(headline) == teams
68 and all(number is None for number in headline.values()),
69 "headline Cup probability containers must include every team with null values")
70 from .cup import CupScenario
71 lineage = analysis["model_lineage"]
72 require(lineage.get("model_ref") == value["header"].get("model_ref"), "header/model lineage mismatch")
73 if lineage.get("model_ref") is not None:
74 require(re.fullmatch(r"[a-f0-9]{64}", lineage["model_ref"]) is not None
75 and lineage.get("source_manifest_hash") == value["header"]["source_manifest_hash"], "invalid pinned model lineage")
76 records = analysis.get("records", [])
77 ids = [record["scenario_id"] for record in records]
78 require(bool(records) and len(set(ids)) == len(ids), "missing or duplicate scenarios")
79 require(ids == value.get("evaluated_scenarios") == value.get("scenarios"), "scenario index mismatch")
80 require(analysis["baseline_scenario_id"] in ids, "baseline absent from grid")
81 budget = analysis["build_budget"]
82 require(integer(budget["trials_per_scenario"]) and budget["trials_per_scenario"] > 0
83 and budget["scenario_count"] == len(records) and integer(budget["seed"]), "invalid build budget")
84 for record in records:

86 scenario.validate()
87 require(scenario.scenario_id == record["scenario_id"], "scenario ID/parameter mismatch")
88 n, resolved, unresolved = record["trials"], record["resolved_trials"], record["unresolved_trials"]
89 require(integer(n) and n > 0 and integer(resolved) and integer(unresolved) and resolved + unresolved == n, "trial counts do not reconcile")
90 require(n == budget["trials_per_scenario"] and record["seed"] == budget["seed"], "scenario budget mismatch")
91 require(abs(record["resolved_fraction"] - resolved / n) < 1e-12
92 and abs(record["unresolved_fraction"] - unresolved / n) < 1e-12, "trial fractions do not reconcile")
93 require(set(record["resolved_win_counts"]) == teams and all(integer(c) for c in record["resolved_win_counts"].values())
94 and sum(record["resolved_win_counts"].values()) == resolved, "resolved wins do not reconcile")
95 first = record["first_unresolved_stage_counts"]
96 require(all(integer(count) and count > 0 for count in first.values()) and sum(first.values()) == unresolved, "first-stage counts are not mutually exclusive")
97 require(all(integer(count) and 0 < count <= n for count in record["overlapping_reason_counts"].values()), "invalid overlapping reason counts")
98 require(sum(record["overlapping_reason_counts"].values()) >= unresolved, "missing unresolved reason attribution")
99 require(record["status"] == ("rule_bounds" if unresolved else "conditional_scenario"), "incorrect scenario status")
100 require(set(record["rule_bounds"]) == teams and set(record["probabilities"]) == teams, "incomplete scenario field")
101 for team in teams:
102 bounds = record["rule_bounds"][team]
103 count = record["resolved_win_counts"][team]
104 require(probability(bounds["lower"]) and probability(bounds["upper"])
105 and abs(bounds["lower"] - count / n) < 1e-12
106 and abs(bounds["upper"] - min(1, (count + unresolved) / n)) < 1e-12, "bounds do not reconstruct from counts")
107 require(record["probabilities"][team] is None if unresolved else
108 probability(record["probabilities"][team]) and abs(record["probabilities"][team] - count / n) < 1e-12,
109 "unresolved point estimate or invalid probability")
110 numerical = record["numerical_error"]
111 require(abs(numerical["worst_case_95_half_width"] - 1.96 * math.sqrt(.25 / n)) < 1e-12
112 and "excludes model/rule" in numerical["scope"], "invalid Monte Carlo precision")
113 require(re.fullmatch(r"[a-f0-9]{64}", record["input_hash"]) is not None, "invalid scenario input hash")
114 require(content_hash(dict(analysis["simulation_inputs"], scenario=record["scenario"],
115 seed=record["seed"], trials=n)) == record["input_hash"], "scenario input hash mismatch")
116 require(record["source_manifest_hash"] == value["header"]["source_manifest_hash"], "scenario source mismatch")
117 require(record.get("model_ref") == analysis["model_lineage"].get("model_ref"), "scenario model mismatch")
118 baseline = next(record for record in records if record["scenario_id"] == analysis["baseline_scenario_id"])
119 for team in teams:
120 for side in ("lower", "upper"):
121 extreme = (min if side == "lower" else max)(record["rule_bounds"][team][side] for record in records)
122 require(value["scenario_envelope_by_team"][team][side] == round(extreme, 4), "envelope does not reconstruct")
123 require(value["rule_bounds_by_team"][team][side] == round(baseline["rule_bounds"][team][side], 4), "baseline bounds do not reconstruct")
124 expected = [record["scenario_id"] for record in records if record["rule_bounds"][team][side] == extreme]
125 require(analysis["extrema_by_team"][team][side + "_scenario_ids"] == expected, "incorrect endpoint attribution")
126 mapping = analysis["mapping_provenance"]["teams"]
127 require({entry["team_id"] for entry in mapping} == teams and len(mapping) == len(teams), "incomplete team mapping")

129 require(isinstance(entry["observed"], bool), "invalid observed mapping flag")

138def validate_race(value):
139 h, pre, live = value["header"], value["pre_race"], value["live"]
140 boats = h["boat_ids"]
141 if h["schema_version"] != "prediction-1" or h["boat_class"] != "AC40" or not boats or len(set(boats)) != len(boats):
142 raise ValueError("Invalid race/field/schema")
143 if "replay_boat_ids" in h:
144 replay=h["replay_boat_ids"]
145 roster=h.get("scheduled_entries",[])
146 if len(replay)!=len(set(replay)) or not set(replay)<=set(boats) or len(roster)!=len(boats) or {r["id"] for r in roster}!=set(boats):
147 raise ValueError("Scheduled roster/replay field mismatch")
148 for key in ("source_manifest_hash", "model_ref", "scenario_hash"):
149 if not isinstance(h.get(key),str) or not re.fullmatch(r"[0-9a-f]{64}",h[key]):
150 raise ValueError("Missing input/model hash")
151 if h.get("status_dictionary") != STATUSES or pre.get("status") not in STATUSES:
152 raise ValueError("Unknown status or status dictionary")
153 if h.get("pre_race_available_at_ms") is not None and h.get("train_cutoff_ms",0)>h["pre_race_available_at_ms"]:
154 raise ValueError("Pre-race training leakage")
155 if set(pre["expected_position_by_boat"]) != set(boats) or pre["nonfinish_score"] != "N":
156 raise ValueError("Incomplete expected-position field")
157 usable = pre["status"] in STATUSES[:2]
158 for x in pre["expected_position_by_boat"].values():
159 if (usable and (x is None or not math.isfinite(x) or not 1 <= x <= len(boats))) or (not usable and x is not None):
160 raise ValueError("Invalid expected-position availability")
161 count = len(live["state_timestamps_ms"])
162 flat = ("published_at_ms", "progress_band", "no_finisher_prob", "status_code", "numerical_error", "reason_code")
163 nested = ("win_prob_by_boat", "source_age_ms_by_boat", "position_uncertainty_summary_by_boat", "quality_code_by_boat")
164 if any(len(live[k]) != count for k in flat):
165 raise ValueError("Unequal checkpoint columns")
166 for key in nested:
167 if set(live[key]) != set(boats) or any(len(x) != count for x in live[key].values()):
168 raise ValueError("Incomplete checkpoint field")
169 for i, t in enumerate(live["state_timestamps_ms"]):
170 publication = live["published_at_ms"][i]
171 if publication < t or (i and (t <= live["state_timestamps_ms"][i-1] or publication < live["published_at_ms"][i-1])):
172 raise ValueError("Noncausal chronology")
173 if h.get("train_cutoff_ms", 0) > t:
174 raise ValueError("Training leakage")
175 if h.get("result_cutoff_ms",0) > t:
176 raise ValueError("Result-label leakage")
177 code=live["status_code"][i]
178 if type(code) is not int or not 0<=code<len(STATUSES):
179 raise ValueError("Unknown checkpoint status code")
180 status = h["status_dictionary"][code]
181 probabilities = [live["win_prob_by_boat"][b][i] for b in boats]+[live["no_finisher_prob"][i]]
182 if status in STATUSES[:2]:
183 if any(p is None or not math.isfinite(p) or not 0<=p<=1 for p in probabilities) or abs(sum(probabilities)-1)>1e-5:
184 raise ValueError("Incomplete joint probability mass")
185 elif any(p is not None for p in probabilities):
186 raise ValueError("Unavailable probabilities must be null")
187 error=live["numerical_error"][i]
188 if error is not None and (not math.isfinite(error) or not 0<=error<=1):
189 raise ValueError("Invalid numerical precision")
190 for boat in boats:
191 age=live["source_age_ms_by_boat"][boat][i]
192 if age is not None and (not isinstance(age,(int,float)) or not math.isfinite(age) or age<0):
193 raise ValueError("Invalid source age")
194 quality=live["quality_code_by_boat"][boat][i]
195 if type(quality) is not int or not 0<=quality<len(STATUSES):
196 raise ValueError("Unknown boat quality code")
197 if status in STATUSES[:2] and quality>=2:
198 raise ValueError("Usable forecast omits an unresolved competitor")
199 radius=live["position_uncertainty_summary_by_boat"][boat][i]
200 if radius is not None and (not isinstance(radius,(float,int)) or not math.isfinite(radius) or radius<0):
201 raise ValueError("Invalid position uncertainty")
202 canonical(value)
203 return True

206def verify_public(root: Path, *, index=None):
207 directory = root/"public"/"data"/"predictions"
208 index = index or json.loads((directory/"index.json").read_text())
209 if index.get("schema_version") != "prediction-index-1":
210 raise ValueError("Unsupported prediction index schema")
211 if len({x["race_id"] for x in index["races"]}) != len(index["races"]):
212 raise ValueError("Duplicate indexed race")
213 def safe_path(relative):
214 path=(directory/relative.removeprefix("predictions/")).resolve()
215 if not path.is_relative_to(directory.resolve()) or path.suffix != ".json":
216 raise ValueError("Unsafe artifact path")
217 return path
218 manifest_path=safe_path(index["source_manifest_path"])
219 manifest=json.loads(manifest_path.read_text())
220 manifest_hash=content_hash(manifest)
221 if manifest_hash != index["source_manifest_hash"] or manifest_path.stem != manifest_hash:
222 raise ValueError("Source manifest content hash mismatch")
223 paths = [safe_path(entry["path"]) for entry in index["races"]]
224 model_paths,scenario_paths=set(),set()
225 for path, entry in zip(paths, index["races"]):
226 artifact = json.loads(path.read_text())
227 validate_race(artifact)
228 if artifact["header"]["race_id"] != entry["race_id"] or artifact["header"]["source_manifest_hash"] != manifest_hash:
229 raise ValueError("Race/manifest scope mismatch")
230 for key in ("boat_ids","replay_boat_ids"):
231 if key in entry and entry[key]!=artifact["header"].get(key):
232 raise ValueError("Index/scheduled roster mismatch")
233 if not path.stem.endswith(content_hash(artifact)):
234 raise ValueError("Race artifact content hash mismatch")
235 for key in ("source_manifest_hash", "model_ref", "scenario_hash"):
236 if artifact["header"][key] != entry[key]:
237 raise ValueError("Index/artifact hash mismatch")
238 model_path=safe_path("models/"+entry["model_ref"]+".json")
239 card=json.loads(model_path.read_text())
240 if content_hash(card) != entry["model_ref"] or card["source_manifest_hash"] != manifest_hash:
241 raise ValueError("Model card content hash mismatch")
242 model_paths.add(model_path)
243 for preset in entry.get("presets",[]):
244 if not preset.get("path"):
245 continue
246 scenario_path=safe_path(preset["path"])
247 scenario=json.loads(scenario_path.read_text())
248 validate_race(scenario)
249 sh=scenario["header"]
250 if scenario_path.stem != content_hash(scenario) or sh.get("scope") != "pre_race" or sh["race_id"] != entry["race_id"]:
251 raise ValueError("Pre-race scenario content/scope mismatch")
252 for key in ("source_manifest_hash","model_ref","scenario_hash"):
253 if sh[key] != preset[key]:
254 raise ValueError("Preset reference mismatch")
255 if sh.get("conditions") != {k:preset[k] for k in ("tws_kt","direction_offset_deg")}:
256 raise ValueError("Preset conditions mismatch")
257 scenario_manifest=json.loads(safe_path("manifests/"+sh["source_manifest_hash"]+".json").read_text())
258 scenario_model_path=safe_path("models/"+sh["model_ref"]+".json")
259 scenario_model=json.loads(scenario_model_path.read_text())
260 if content_hash(scenario_manifest) != sh["source_manifest_hash"] or content_hash(scenario_model) != sh["model_ref"]:
261 raise ValueError("Preset source/model content mismatch")
262 if scenario_model["source_manifest_hash"] != sh["source_manifest_hash"]:
263 raise ValueError("Preset model source mismatch")
264 model_paths.add(scenario_model_path)
265 scenario_paths.add(scenario_path)
266 cup_path=safe_path(index["cup_path"])
267 cup=json.loads(cup_path.read_text())
268 if (cup.get("header",{}).get("schema_version") != "cup-1" or cup["header"].get("boat_class") != "AC75"
269 or cup["header"].get("source_manifest_hash") != manifest_hash or len(cup.get("teams",[])) != 7
270 or len({t["id"] for t in cup.get("teams",[])}) != 7 or cup.get("status") not in STATUSES):
271 raise ValueError("Cup schema, class, field or provenance mismatch")
272 if cup_path.stem != content_hash(cup):
273 raise ValueError("Cup artifact content hash mismatch")
274 validate_cup(cup)
275 raw = sum(p.stat().st_size for p in paths)+len(canonical(index))+cup_path.stat().st_size
276 compressed = sum(len(gzip.compress(p.read_bytes(), mtime=0)) for p in paths)
277 if raw > 3*1024*1024:
278 raise ValueError(f"Forecast baseline exceeds 3 MiB hard budget: {raw}")
279 scenario_bytes=sum(p.stat().st_size for p in scenario_paths)
280 if scenario_bytes>3*1024*1024:
281 raise ValueError("Lazy pre-race scenarios exceed their separate 3 MiB budget")
282 return {"races": len(paths), "baseline_bytes": raw, "race_gzip_bytes": compressed,
283 "referenced_model_bytes":sum(p.stat().st_size for p in model_paths),
284 "source_manifest_bytes":manifest_path.stat().st_size,
285 "lazy_scenario_bytes":scenario_bytes,
286 "target_1_5_MiB_passed": raw <= 1.5*1024*1024, "hard_3_MiB_passed": True}
