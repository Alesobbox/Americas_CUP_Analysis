75 describe('3-Point Affine Synchronizer', () => {
76 // Simulated MapLibre camera model
77 function createSimulatedMapProject(
78 centerMetric: Point,
79 scalePxPerM: number,
80 bearingDeg: number,
81 viewportWidth = 1000,
82 viewportHeight = 800
83 ) {
84 const rad = (bearingDeg * Math.PI) / 180
85 const cx = viewportWidth / 2
86 const cy = viewportHeight / 2
87
88 return (lngLat: [number, number]): ScreenPoint => {
89 const [mx, my] = lngLatToMetric(lngLat[0], lngLat[1], NAPLES_ORIGIN, NAPLES_PROJECTION)
90 const dx = mx - centerMetric[0]
91 const dy = my - centerMetric[1]
92
93 // Camera bearing clockwise rotation causes counter-clockwise world rotation
94 const screenX = cx + (Math.cos(rad) * dx - Math.sin(rad) * dy) * scalePxPerM
95 const screenY = cy - (Math.sin(rad) * dx + Math.cos(rad) * dy) * scalePxPerM
96
97 return { x: screenX, y: screenY }
98 }
99 }
100
