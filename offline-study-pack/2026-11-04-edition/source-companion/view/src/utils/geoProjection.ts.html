45/**
46 * Closed-form inverse equirectangular projection:
47 * Inverts local metric Cartesian coordinates (x, y) in metres relative to course origin
48 * into WGS84 geographic coordinates [longitude, latitude].
49 * (Returns [lon, lat] adhering to GeoJSON and MapLibre conventions).
50 */
51export function metricToLngLat(
52 x: number,
53 y: number,
54 origin: GeoOrigin,
55 projection: GeoProjectionParams
56): [number, number] {
57 const lon = origin.lon + x / projection.k_lon
58 const lat = origin.lat + y / projection.k_lat
59 return [lon, lat]
60}

62/**
63 * Forward projection:
64 * Converts WGS84 geographic coordinates [longitude, latitude] into local metric Cartesian coordinates (x, y) in metres.
65 */
66export function lngLatToMetric(
67 lon: number,
68 lat: number,
69 origin: GeoOrigin,
70 projection: GeoProjectionParams
71): Point {
72 const x = (lon - origin.lon) * projection.k_lon
73 const y = (lat - origin.lat) * projection.k_lat
74 return [x, y]
75}

77/**
78 * Computes the 2D affine matrix from screen projections of 3 anchor points:
79 * - Origin (0, 0) m -> s0
80 * - East (distanceM, 0) m -> sx
81 * - North (0, distanceM) m -> sy
82 */
83export function computeAffineMatrixFromScreenPoints(
84 s0: ScreenPoint,
85 sx: ScreenPoint,
86 sy: ScreenPoint,
87 distanceM = 1000
88): AffineMatrix2D {
89 if (distanceM <= 0) {
90 throw new Error('Anchor distance must be greater than zero')
91 }
92 return {
93 a11: (sx.x - s0.x) / distanceM,
94 a12: (sy.x - s0.x) / distanceM,
95 tx: s0.x,
96 a21: (sx.y - s0.y) / distanceM,
97 a22: (sy.y - s0.y) / distanceM,
98 ty: s0.y,
99 }
100}

102/**
103 * 3-Point Affine Anchor calculation:
104 * Computes 2D affine matrix (a11, a12, tx, a21, a22, ty) from MapLibre camera projections
105 * of Origin (0, 0), East (1000, 0), and North (0, 1000).
106 *
107 * Allows the 2D canvas overlay to map metric metres directly to screen pixels
108 * in O(1) constant time with sub-pixel (< 0.01 px) accuracy.
109 *
110 * ZERO DOM and ZERO MapLibre runtime imports.
111 */
112export function computeAffineScreenTransform(
113 projectFn: (lngLat: [number, number]) => ScreenPoint,
114 origin: GeoOrigin,
115 projection: GeoProjectionParams,
116 anchorDistanceM = 1000
117): AffineScreenTransform {
118 // 1. Origin (0, 0) m
119 const s0 = projectFn([origin.lon, origin.lat])
120
121 // 2. East (anchorDistanceM, 0) m
122 const eastLngLat = metricToLngLat(anchorDistanceM, 0, origin, projection)
123 const sx = projectFn(eastLngLat)
124
125 // 3. North (0, anchorDistanceM) m
126 const northLngLat = metricToLngLat(0, anchorDistanceM, origin, projection)
127 const sy = projectFn(northLngLat)
128
129 const matrix = computeAffineMatrixFromScreenPoints(s0, sx, sy, anchorDistanceM)
130
131 // Scale in pixels per metre
132 const scaleX = Math.hypot(matrix.a11, matrix.a21)
133 const scaleY = Math.hypot(matrix.a12, matrix.a22)
134 const scale = (scaleX + scaleY) / 2
135
136 // Determinant for 2x2 inverse
137 const det = matrix.a11 * matrix.a22 - matrix.a12 * matrix.a21
138
139 // Rotation: East vector on screen has angle atan2(a21, a11)
140 // When bearing = 0, East vector points right (0 rad).
141 // Clockwise camera bearing rotates the on-screen East vector counter-clockwise.
142 const eastAngleRad = Math.atan2(matrix.a21, matrix.a11)
143 let rotationDeg = -eastAngleRad * (180 / Math.PI)
144 while (rotationDeg < 0) rotationDeg += 360
145 while (rotationDeg >= 360) rotationDeg -= 360
146 const rotationRad = (rotationDeg * Math.PI) / 180
147
148 const toScreen = ([x, y]: Point): Point => [
149 matrix.a11 * x + matrix.a12 * y + matrix.tx,
150 matrix.a21 * x + matrix.a22 * y + matrix.ty,
151 ]
152
153 const vectorToScreen = ([dx, dy]: Point): Point => [
154 matrix.a11 * dx + matrix.a12 * dy,
155 matrix.a21 * dx + matrix.a22 * dy,
156 ]
157
158 const toWorld = ([X, Y]: Point): Point => {
159 if (Math.abs(det) < 1e-12) return [0, 0]
160 const dx = X - matrix.tx
161 const dy = Y - matrix.ty
162 return [
163 (matrix.a22 * dx - matrix.a12 * dy) / det,
164 (-matrix.a21 * dx + matrix.a11 * dy) / det,
165 ]
166 }
167
168 return {
169 matrix,
170 toScreen,
171 vectorToScreen,
172 toWorld,
173 originScreen: [s0.x, s0.y],
174 scaleX,
175 scaleY,
176 scale,
177 rotationRad,
178 rotationDeg,
179 }
