35export function statusAt(replay: ReplayManifest, competitor: Competitor, timestamp: number): RaceStatus {
36 if (timestamp < replay.chronology.gun_time) return 'PRE START'
37 if ((competitor.start.ocs && competitor.start.crossing_time > 0 && competitor.start.crossing_time <= timestamp) ||
38 (competitor.classification === 'RET' && timestamp >= replay.chronology.window_end)) return 'OCS / RET'
39 if (competitor.classification === 'DNF' && timestamp >= replay.chronology.window_end) return 'DNF'
40 if (!(competitor.start.crossing_time > 0) || competitor.start.crossing_time > timestamp) return 'PRE START'
41 const finish = finishTime(replay, competitor)
42 if (finish !== null && timestamp >= finish) return 'FINISHED'
43 if (competitor.classification === 'DNF' && timestamp >= replay.chronology.window_end) return 'DNF'
44 return 'RACING'
45}

48export function getStandings(replay: ReplayManifest, frame: ReplayFrame): Standing[] {
49 const entries = replay.competitors
50 .filter((competitor) => frame.boats[competitor.id])
51 .map((competitor) => ({ competitor, frame: frame.boats[competitor.id], rank: null, status: statusAt(replay, competitor, frame.timestamp), gapM: null, gapS: null, gapIsEstimate: false }) as Standing)
52
53 const priority: Record<RaceStatus, number> = { FINISHED: 0, RACING: 1, 'PRE START': 2, DNF: 3, 'OCS / RET': 4 }
54 entries.sort((a, b) => {
55 if (priority[a.status] !== priority[b.status]) return priority[a.status] - priority[b.status]
56 if (a.status === 'FINISHED') return (finishTime(replay, a.competitor) ?? Infinity) - (finishTime(replay, b.competitor) ?? Infinity)
57 if (a.status === 'RACING') return b.frame.progress_m - a.frame.progress_m || a.competitor.team_code.localeCompare(b.competitor.team_code)
58 return a.competitor.team_code.localeCompare(b.competitor.team_code)
59 })
60
61 const leader = entries.find((entry) => entry.status === 'FINISHED' || entry.status === 'RACING')
62 let placed = 0
63 for (const entry of entries) {
64 if (entry.status === 'PRE START' || entry.status === 'OCS / RET' || entry.status === 'DNF') continue
65 entry.rank = ++placed
66 if (!leader || entry === leader) { entry.gapM = 0; entry.gapS = 0; continue }
67 entry.gapM = Math.max(0, leader.frame.progress_m - entry.frame.progress_m)
68 if (leader.status === 'FINISHED' && entry.status === 'FINISHED') {
69 entry.gapS = Math.max(0, ((finishTime(replay, entry.competitor) ?? 0) - (finishTime(replay, leader.competitor) ?? 0)) / 1000)
70 } else {
71 entry.gapS = leader.frame.sog > 0 ? entry.gapM / (leader.frame.sog * KNOTS_TO_MPS) : null
72 entry.gapIsEstimate = entry.gapS !== null
73 }
74 }
75
76 // Pre-start separation for 2-boat match race
77 if (entries.length === 2 && entries.every((e) => e.status === 'PRE START')) {
78 const dist = Math.hypot(entries[0].frame.x - entries[1].frame.x, entries[0].frame.y - entries[1].frame.y)
79 entries[0].gapM = 0
80 entries[1].gapM = dist
81 }
82
83 return entries
84}

86export function compareBoats(a: BoatFrame, b: BoatFrame, axisDeg: number): Comparison {
87 const eastM = b.x - a.x
88 const northM = b.y - a.y
89 const separationM = Math.hypot(eastM, northM)
90 const axis = axisDeg * Math.PI / 180
91 const direction = a.leg > 0 && a.leg % 2 === 0 ? -1 : 1
92 const alongLegM = direction * (eastM * Math.sin(axis) + northM * Math.cos(axis))
93 const lateralM = eastM * Math.cos(axis) - northM * Math.sin(axis)
94 const velocity = (boat: BoatFrame): Point => [Math.sin(boat.cog * Math.PI / 180) * boat.sog * KNOTS_TO_MPS, Math.cos(boat.cog * Math.PI / 180) * boat.sog * KNOTS_TO_MPS]
95 const av = velocity(a)
96 const bv = velocity(b)
97 const closureMps = separationM < 0.001 ? 0 : -((bv[0] - av[0]) * eastM + (bv[1] - av[1]) * northM) / separationM
98 return { separationM, eastM, northM, alongLegM, lateralM, progressDeltaM: b.progress_m - a.progress_m, speedDeltaKt: b.sog - a.sog, vmgDeltaKt: b.vmg - a.vmg, closureMps }
99}

101export function windAt(replay: ReplayManifest, timestamp: number) {
102 const timeline = replay.wind.timeline
103 if (!timeline.length || timeline[0].t > timestamp) return { t: null, twd: Number.NaN, tws: Number.NaN }
104 let lo = 0
105 let hi = timeline.length - 1
106 while (lo < hi) { const mid = Math.ceil((lo + hi) / 2); if (timeline[mid].t <= timestamp) lo = mid; else hi = mid - 1 }
