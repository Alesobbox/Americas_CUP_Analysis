7/** Supports the current regattas catalog and the older flat races catalog. */
8export function parseRaceCatalog(value: unknown): RaceListing[] {
9 if (!value || typeof value !== 'object') throw new Error('Race catalog is missing')
10 const root = value as { regattas?: unknown; races?: unknown; regatta?: unknown; venue?: unknown }
11 const groups: RawRegatta[] = Array.isArray(root.regattas) ? root.regattas : [{ id: 'legacy', name: root.regatta, venue: root.venue, races: root.races }]
12 const listings: RaceListing[] = []
13 for (const group of groups) {
14 if (!Array.isArray(group.races)) continue
15 for (const candidate of group.races) {
16 const race = candidate as RawRace
17 const path = race.replay_file ?? race.file
18 if (typeof race.id !== 'string' || typeof race.name !== 'string' || typeof race.date !== 'string' || typeof path !== 'string') continue
19 if (!/^replays\/[a-zA-Z0-9_-]+\.json$/.test(path)) continue
20 listings.push({ id: race.id, name: race.name, date: race.date, replayFile: path, regattaId: String(group.id ?? 'regatta'), regattaName: String(group.name ?? 'Regatta'), venue: String(group.venue ?? '') })
21 }
22 }
23 if (!listings.length) throw new Error('The race catalog contains no loadable races')
24 return listings
25}
