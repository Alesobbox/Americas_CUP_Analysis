216/**
217 * Calculates render scale factor (screen pixels per metric metre).
218 * True mode preserves physical map scale. Enhanced mode applies the documented
219 * minimum pixel floor so distant boats remain visible.
220 */
221export function computeBoatRenderScale(
222 boatClass: BoatClass | undefined,
223 viewportScalePxPerM: number,
224 mode: BoatScaleMode = 'enhanced',
225): number {
226 const metricScale = Number.isFinite(viewportScalePxPerM) ? Math.max(0, viewportScalePxPerM) : 0
227 if (mode === 'true') return metricScale
228 const dims = getBoatDimensions(boatClass)
229 const minScaleFromFloor = dims.minPixelFloor / dims.totalLengthM
230 return Math.max(metricScale, minScaleFromFloor)
231}

233/**
234 * Helper to trace a closed polygon in a canvas 2D context.
235 * In rotated boat-local coordinates:
236 * local forward (v) is on the x-axis, and local starboard (u) is on the y-axis.
237 */
238export function tracePolygonInCanvas(
239 ctx: CanvasRenderingContext2D,
240 polygon: Point2D[],
241 scale: number,
242): void {
243 if (polygon.length === 0) return
244 ctx.beginPath()
245 // pt[0] is u (starboard), pt[1] is v (forward)
246 ctx.moveTo(polygon[0][1] * scale, polygon[0][0] * scale)
247 for (let i = 1; i < polygon.length; i++) {
248 ctx.lineTo(polygon[i][1] * scale, polygon[i][0] * scale)
249 }
250 ctx.closePath()
251}

253/**
254 * Draws class-specific parametric foiling monohull in boat-local canvas space.
255 * Assumes context is translated to boat position and rotated to screen heading.

257export function renderFoilingBoatModel(
258 ctx: CanvasRenderingContext2D,
259 boatClass: BoatClass | undefined,
260 teamColor: string,
261 scale: number,
262 selected: boolean,
263 mode: BoatScaleMode = 'enhanced',
264): void {
265 const isAc75 = boatClass === 'AC75'
266
267 const hullPoly = isAc75 ? AC75_HULL_POLYGON : AC40_HULL_POLYGON
268 const cockpitStbd = isAc75 ? AC75_COCKPIT_STARBOARD : AC40_COCKPIT_STARBOARD
269 const cockpitPort = isAc75 ? AC75_COCKPIT_PORT : AC40_COCKPIT_PORT
270 const foilArmPort = isAc75 ? AC75_FOIL_ARM_PORT : AC40_FOIL_ARM_PORT
271 const foilArmStbd = isAc75 ? AC75_FOIL_ARM_STARBOARD : AC40_FOIL_ARM_STARBOARD
272 const foilWingPort = isAc75 ? AC75_FOIL_WING_PORT : AC40_FOIL_WING_PORT
273 const foilWingStbd = isAc75 ? AC75_FOIL_WING_STARBOARD : AC40_FOIL_WING_STARBOARD
274 const rudderElevator = isAc75 ? AC75_RUDDER_ELEVATOR : AC40_RUDDER_ELEVATOR
275
276 ctx.save()
277
278 // At course zoom, true-metric hulls are only a few pixels long. Suppress
279 // fixed-pixel fittings so they cannot masquerade as a much larger boat.
280 if (mode === 'true' && scale < 0.5) {
281 ctx.fillStyle = teamColor
282 tracePolygonInCanvas(ctx, foilWingPort, scale)
283 ctx.fill()
284 tracePolygonInCanvas(ctx, foilWingStbd, scale)
285 ctx.fill()
286 ctx.fillStyle = '#ffffff'
287 tracePolygonInCanvas(ctx, hullPoly, scale)
288 ctx.fill()
289 ctx.restore()
290 return
291 }
292
293 // 1. Rudder elevator assembly (drawn under hull)
294 ctx.fillStyle = '#1e293b'
295 ctx.strokeStyle = '#0f172a'
296 ctx.lineWidth = Math.max(0.75, 0.1 * scale)
297 tracePolygonInCanvas(ctx, rudderElevator, scale)
298 ctx.fill()
299 ctx.stroke()
300
301 // Rudder stock connecting line
302 ctx.beginPath()
303 ctx.moveTo(0, 0)
304 ctx.lineTo((isAc75 ? -0.3 : -0.18) * scale, 0)
305 ctx.strokeStyle = '#334155'
306 ctx.lineWidth = Math.max(1, 0.18 * scale)
307 ctx.stroke()
308
309 // 2. Canting Foil Arms (structural carbon grey)
310 ctx.fillStyle = '#27272a'
311 ctx.strokeStyle = '#18181b'
312 ctx.lineWidth = Math.max(0.75, 0.08 * scale)
313
314 tracePolygonInCanvas(ctx, foilArmPort, scale)
315 ctx.fill()
316 ctx.stroke()
317
318 tracePolygonInCanvas(ctx, foilArmStbd, scale)
319 ctx.fill()
320 ctx.stroke()
321
322 // 3. Canting Foil Wings / T-Foils (with team / foiling cyan accents)
323 ctx.fillStyle = '#0f172a'
324 ctx.strokeStyle = teamColor
325 ctx.lineWidth = Math.max(1.0, 0.12 * scale)
326
327 tracePolygonInCanvas(ctx, foilWingPort, scale)
328 ctx.fill()
329 ctx.stroke()
330
331 tracePolygonInCanvas(ctx, foilWingStbd, scale)
332 ctx.fill()
333 ctx.stroke()
334
335 // Wing tip winglet highlights
336 ctx.fillStyle = selected ? '#00d2ff' : teamColor
337 const tipSpan = isAc75 ? 8.0 : 5.415
338 const tipV = isAc75 ? 11.35 : 6.58
339 const tipR = Math.max(1.5, 0.22 * scale)
340
341 ctx.beginPath()
342 ctx.arc(tipV * scale, -tipSpan * scale, tipR, 0, Math.PI * 2)
343 ctx.arc(tipV * scale, tipSpan * scale, tipR, 0, Math.PI * 2)
344 ctx.fill()
345
346 // 4. Main Hull Structure (Crisp white monohull with aerodynamic taper)
347 tracePolygonInCanvas(ctx, hullPoly, scale)
348 ctx.fillStyle = '#ffffff'
349 ctx.fill()
350
351 // Hull outer perimeter outline
352 ctx.strokeStyle = selected ? '#00d2ff' : 'rgba(15, 23, 42, 0.95)'
353 ctx.lineWidth = selected ? Math.max(1.8, 0.22 * scale) : Math.max(1.0, 0.12 * scale)
354 ctx.stroke()
355
356 // 5. Team Livery Deck Accent Strip
357 const stemV = isAc75 ? 20.65 : 11.80
358 ctx.beginPath()
359 ctx.moveTo(0, 0)
360 ctx.lineTo(stemV * scale, 0)
361 ctx.strokeStyle = teamColor
362 ctx.lineWidth = Math.max(1.5, 0.25 * scale)
363 ctx.stroke()
364
365 // 6. Recessed Cockpit Trenches (carbon black aero cockpits)
366 ctx.fillStyle = '#090e17'
367 ctx.strokeStyle = '#1e293b'
368 ctx.lineWidth = Math.max(0.5, 0.06 * scale)
369
370 tracePolygonInCanvas(ctx, cockpitPort, scale)
371 ctx.fill()
372 ctx.stroke()
373
374 tracePolygonInCanvas(ctx, cockpitStbd, scale)
375 ctx.fill()
376 ctx.stroke()
377
378 // 7. Bowsprit Spar / Forward Instrumentation Wand
379 const bowspritTipV = isAc75 ? 21.50 : 12.963
380 ctx.beginPath()
381 ctx.moveTo(stemV * scale, 0)
382 ctx.lineTo(bowspritTipV * scale, 0)
383 ctx.strokeStyle = '#090e17'
384 ctx.lineWidth = Math.max(1.2, 0.16 * scale)
385 ctx.stroke()
386
387 // Foremost tip bead
388 ctx.fillStyle = selected ? '#ffffff' : '#94a3b8'
389 ctx.beginPath()
390 ctx.arc(bowspritTipV * scale, 0, Math.max(1.5, 0.18 * scale), 0, Math.PI * 2)
391 ctx.fill()
392
393 ctx.restore()
