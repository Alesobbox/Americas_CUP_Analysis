57 function findElementsByType(node: any, type: string): any[] {
58 const matches: any[] = []
59 if (!node) return matches
60 if (node.type === type) {
61 matches.push(node)
62 }
63 if (node.props?.children) {
64 const children = Array.isArray(node.props.children) ? node.props.children : [node.props.children]
65 for (const child of children) {
66 matches.push(...findElementsByType(child, type))
67 }
68 }
69 return matches
70 }
