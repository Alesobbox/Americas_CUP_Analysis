57 // Recursive element traversal helper
58 function findElementsByType(node: any, type: string): any[] {
59 const matches: any[] = []
60 if (!node) return matches
61 if (node.type === type) {
62 matches.push(node)
63 }
64 if (node.props?.children) {
65 const children = Array.isArray(node.props.children) ? node.props.children : [node.props.children]
66 for (const child of children) {
67 matches.push(...findElementsByType(child, type))
68 }
69 }
70 return matches
71 }
