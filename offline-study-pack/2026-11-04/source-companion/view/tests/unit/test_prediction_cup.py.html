20def fixed_results(*, cup_zero_races=0, selected="LA_ROCHE_POSAY"):
21 """Every qualifying result is fixed, so later known participants are valid."""
22 scores = {}
23 for round_number in range(1, 4):
24 for index_a, team_a in enumerate(TEAM_IDS):
25 for team_b in TEAM_IDS[index_a + 1:]:
26 scores[f"rr:{round_number}:{team_a}:{team_b}"] = MatchScore(team_a, team_b, 1, 0)
27
28 def wins(stage, team_a, team_b, count):
29 for race in range(1, count + 1):
30 scores[f"{stage}:{race}"] = MatchScore(team_a, team_b, 1, 0)
31
32 wins("repechage_1", "ARC_USA", "AUSTRALIA", 2)
33 wins("repechage_2", "LA_ROCHE_POSAY", "ARC_USA", 2)
34 wins("semifinal_1", "GB1", selected, 5)
35 other = "ALINGHI" if selected == "LA_ROCHE_POSAY" else "LA_ROCHE_POSAY"
36 wins("semifinal_2", "LUNA_ROSSA", other, 5)
37 wins("challenger_final", "GB1", "LUNA_ROSSA", 7)
38 for race in range(1, cup_zero_races + 1):
39 scores[f"cup_match:{race}"] = MatchScore("ETNZ", "GB1", 0, 0)
40 for race in range(cup_zero_races + 1, cup_zero_races + 8):
41 scores[f"cup_match:{race}"] = MatchScore("ETNZ", "GB1", 1, 0)
42 return CupRules(known_fleet_orders=(TEAM_IDS,) * 6, known_match_scores=scores,
43 official_selections={"GB1": selected} if selected != "LA_ROCHE_POSAY" else {})
