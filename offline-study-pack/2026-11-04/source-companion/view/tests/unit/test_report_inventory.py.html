18 with (DATA / "analysis" / (race["id"] + "_summary.csv")).open(newline="", encoding="utf-8") as stream:
19 rows = list(csv.DictReader(stream))
20 expected = {(boat["id"], key): metric for boat in detail["boats"] for key, metric in boat["metrics"].items()}
21 self.assertEqual(len(rows), len(expected), race["id"])
22 for row in rows:

24 self.assertEqual(row["value"], "" if metric["value"] is None else str(metric["value"]))
25 self.assertEqual(row["coverage_s"], "" if metric["coverage_s"] is None else str(metric["coverage_s"]))
26 self.assertEqual(row["quality"], metric["quality"])
27 self.assertEqual(row["reason"], metric.get("reason", ""))
28 self.assertEqual(row["method_version"], detail["method_version"])
