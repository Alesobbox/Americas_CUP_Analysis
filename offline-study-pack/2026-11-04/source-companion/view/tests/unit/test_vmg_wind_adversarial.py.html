19def srgb_to_linear(c_8bit: int) -> float:
20 s = c_8bit / 255.0
21 return s / 12.92 if s <= 0.04045 else math.pow((s + 0.055) / 1.055, 2.4)

24def relative_luminance(r: int, g: int, b: int) -> float:
25 return 0.2126 * srgb_to_linear(r) + 0.7152 * srgb_to_linear(g) + 0.0722 * srgb_to_linear(b)

28def wcag_contrast_ratio(rgb1: Tuple[int, int, int], rgb2: Tuple[int, int, int]) -> float:
29 l1 = relative_luminance(*rgb1)
30 l2 = relative_luminance(*rgb2)
31 lighter = max(l1, l2)
32 darker = min(l1, l2)
33 return (lighter + 0.05) / (darker + 0.05)

36def euclidean_rgb_distance(rgb1: Tuple[int, int, int], rgb2: Tuple[int, int, int]) -> float:
37 return math.sqrt(
38 (rgb1[0] - rgb2[0]) ** 2 +
39 (rgb1[1] - rgb2[1]) ** 2 +
40 (rgb1[2] - rgb2[2]) ** 2

157 def test_downwind_vector_projection_math(self):
158 """Verifies trigonometric projection for True North (0), East (90), South (180), West (270)."""
159 # North wind (TWD = 0) -> downwind = 180 -> flow travels South (dy < 0 in world, vy > 0 in screen)
160 # East wind (TWD = 90) -> downwind = 270 -> flow travels West (dx < 0 in world, vx < 0 in screen)
161 # South wind (TWD = 180) -> downwind = 0 -> flow travels North (dy > 0 in world, vy < 0 in screen)
162 # West wind (TWD = 270) -> downwind = 90 -> flow travels East (dx > 0 in world, vx > 0 in screen)
163 directions = [
164 (0, 180, 0.0, -1.0, 0.0, 1.0), # North: world=(0, -1), screen=(0, +1) -> Down
165 (90, 270, -1.0, 0.0, -1.0, 0.0), # East: world=(-1, 0), screen=(-1, 0) -> Left
166 (180, 0, 0.0, 1.0, 0.0, -1.0), # South: world=(0, +1), screen=(0, -1) -> Up
167 (270, 90, 1.0, 0.0, 1.0, 0.0), # West: world=(+1, 0), screen=(+1, 0) -> Right
168 ]
169
170 for twd, expected_downwind, exp_wx, exp_wy, exp_sx, exp_sy in directions:
171 downwind = (twd + 180) % 360
172 self.assertEqual(downwind, expected_downwind)
173 rad = math.radians(downwind)
174 wx = math.sin(rad)
175 wy = math.cos(rad)
176 self.assertAlmostEqual(wx, exp_wx, places=5)
177 self.assertAlmostEqual(wy, exp_wy, places=5)
178
179 # In North-Up screen projection: vx = wx, vy = -wy
180 sx = wx
181 sy = -wy
182 self.assertAlmostEqual(sx, exp_sx, places=5)
183 self.assertAlmostEqual(sy, exp_sy, places=5)

185 def test_course_up_rotation_downwind_alignment(self):
186 """In Course-Up mode (40° vertical), wind from 40° blows straight DOWN on the screen (vx=0, vy>0)."""
187 course_axis = 40.0
188 twd = 40.0 # Wind directly down the course axis
189 downwind = (twd + 180) % 360 # 220°
190 rad = math.radians(downwind)
191 world_vec = (math.sin(rad), math.cos(rad))
192
193 # World to screen rotation by 40°:
194 # vx = (cos(40) * wx - sin(40) * wy)
195 # vy = -(sin(40) * wx + cos(40) * wy)
196 rot_rad = math.radians(course_axis)
197 vx = math.cos(rot_rad) * world_vec[0] - math.sin(rot_rad) * world_vec[1]
198 vy = -(math.sin(rot_rad) * world_vec[0] + math.cos(rot_rad) * world_vec[1])
199
200 self.assertAlmostEqual(vx, 0.0, places=5)
201 self.assertGreater(vy, 0.0) # Straight down screen along leeward course direction
