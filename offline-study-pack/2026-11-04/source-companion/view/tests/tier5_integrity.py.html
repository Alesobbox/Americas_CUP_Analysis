107 def test_timestamps_and_boat_columns_align(self):
108 times = self.replay["telemetry"]["timestamps"]
109 self.assertGreater(len(times), 1_000)
110 self.assertEqual(times[0], self.replay["chronology"]["window_start"])
111 self.assertEqual(times[-1], self.replay["chronology"]["window_end"])
112 self.assertTrue(all(b - a == 100 for a, b in zip(times, times[1:])))
113 competitors = self.replay["competitors"]
114 self.assertEqual(set(self.replay["telemetry"]["boats"]), {boat["id"] for boat in competitors})
115 for boat in competitors:
116 columns = self.replay["telemetry"]["boats"][boat["id"]]
117 for name, values in columns.items():
118 self.assertEqual(len(values), len(times), (boat["team_code"], name))
119 self.assertTrue(all(math.isfinite(value) for value in values), (boat["team_code"], name))
120 for name in ("heading", "cog", "twd"):
121 self.assertTrue(all(0 <= angle < 360 for angle in columns[name]), (boat["team_code"], name))
122 self.assertTrue(all(b >= a for a, b in zip(columns["progress_m"], columns["progress_m"][1:])), boat["team_code"])

133 def test_events_are_ordered_and_classification_is_consistent(self):
134 window = self.replay["chronology"]
135 ranks = []
136 for boat in self.replay["competitors"]:
137 gates = [gate["crossing_time"] for gate in boat["gates"]]
138 manoeuvres = [event["timestamp"] for event in boat["manoeuvres"]]
139 self.assertEqual(gates, sorted(gates), boat["team_code"])
140 self.assertEqual(manoeuvres, sorted(manoeuvres), boat["team_code"])
141 self.assertTrue(all(window["window_start"] <= t <= window["window_end"] for t in gates + manoeuvres))
142 if boat["classification"].startswith("RANK_"):
143 self.assertGreater(boat["summary"]["elapsed_s"], 0)
144 ranks.append(boat["summary"]["finish_rank"])
145 else:
146 self.assertEqual(boat["summary"]["elapsed_s"], 0)
147 self.assertEqual(ranks, list(range(1, len(ranks) + 1)))
