233 def test_dynamic_marks_physics_and_rounding_sla(self):
234 """Verifies RM2 position at t = 1790079233000 (frame 4130) is within 30m of FRA rounding."""
235 replay_path = WORKSPACE_ROOT / "public" / "data" / "replays" / "naples_pr1.json"

242 t_rounding = 1790079233000
243 timestamps = manifest["telemetry"]["timestamps"]

245 frame_idx = timestamps.index(t_rounding)
246 self.assertEqual(frame_idx, 4130)

249 rm2_asset_id = manifest["course"]["marks"]["RM2"]["asset_id"]

256 rm2_y = marks_telemetry[rm2_asset_id]["y"][frame_idx]
257 self.assertAlmostEqual(rm2_x, 1103.60, delta=0.2)
258 self.assertAlmostEqual(rm2_y, 537.55, delta=0.2)
259
260 # Find FRA
261 fra_comp = next(c for c in manifest["competitors"] if c["team_code"] == "FRA")
262 fra_boat = manifest["telemetry"]["boats"][fra_comp["id"]]
263 fra_x = fra_boat["x"][frame_idx]
264 fra_y = fra_boat["y"][frame_idx]
265

267 self.assertGreater(fra_x, 1000.0)
268 self.assertLess(fra_x, 1100.0)
269 self.assertGreater(fra_y, 500.0)
270 self.assertLess(fra_y, 600.0)
271
272 # Compute Euclidean distance

274 self.assertLess(distance_m, 30.0, f"Distance {distance_m:.2f}m exceeds 30m SLA")

279 static_dist = math.hypot(static_target[0] - fra_x, static_target[1] - fra_y)

282 def test_dynamic_marks_full_fleet_roundings_leg1_and_leg3(self):
283 """Verifies full fleet rounding alignment across Leg 1 and Leg 3 against dynamic marks."""
284 replay_path = WORKSPACE_ROOT / "public" / "data" / "replays" / "naples_pr1.json"
285 if not replay_path.exists():
286 self.skipTest("naples_pr1.json not found")
287
288 with open(replay_path, "r", encoding="utf-8") as f:
289 manifest = json.load(f)
290
291 timestamps = manifest["telemetry"]["timestamps"]
292 marks_telemetry = manifest["telemetry"]["marks"]
293 rm2_stream = marks_telemetry[manifest["course"]["marks"]["RM2"]["asset_id"]]
294 rm4_stream = marks_telemetry[manifest["course"]["marks"]["RM4"]["asset_id"]]
295
296 # Leg 1 Fleet Roundings
297 for comp in manifest["competitors"]:
298 boat = manifest["telemetry"]["boats"][comp["id"]]
299 legs = boat["leg"]
300
301 # Find transition from leg 1 to leg 2
302 leg1_trans = next((i for i in range(len(legs) - 1) if legs[i] == 1 and legs[i + 1] == 2), None)
303 if leg1_trans is None:
304 continue
305
306 # Closest approach in +/- 5s window
307 w_start = max(0, leg1_trans - 50)
308 w_end = min(len(legs) - 1, leg1_trans + 50)
309 min_dist = float("inf")
310 for j in range(w_start, w_end + 1):
311 bx, by = boat["x"][j], boat["y"][j]
312 d2 = math.hypot(rm2_stream["x"][j] - bx, rm2_stream["y"][j] - by)
313 d4 = math.hypot(rm4_stream["x"][j] - bx, rm4_stream["y"][j] - by)
314 min_dist = min(min_dist, d2, d4)
315
316 self.assertLess(min_dist, 35.0, f"Team {comp['team_code']} Leg 1 closest mark dist {min_dist:.2f}m exceeds 35m")
317
318 # Leg 3 Fleet Roundings
319 for comp in manifest["competitors"]:
320 boat = manifest["telemetry"]["boats"][comp["id"]]
321 legs = boat["leg"]
322
323 leg3_trans = next((i for i in range(len(legs) - 1) if legs[i] == 3 and legs[i + 1] == 4), None)
324 if leg3_trans is None:
325 continue
326
327 w_start = max(0, leg3_trans - 50)
328 w_end = min(len(legs) - 1, leg3_trans + 50)
329 min_dist = float("inf")
330 for j in range(w_start, w_end + 1):
331 bx, by = boat["x"][j], boat["y"][j]
332 d2 = math.hypot(rm2_stream["x"][j] - bx, rm2_stream["y"][j] - by)
333 d4 = math.hypot(rm4_stream["x"][j] - bx, rm4_stream["y"][j] - by)
334 min_dist = min(min_dist, d2, d4)
335
336 self.assertLess(min_dist, 45.0, f"Team {comp['team_code']} Leg 3 closest mark dist {min_dist:.2f}m exceeds 45m")
