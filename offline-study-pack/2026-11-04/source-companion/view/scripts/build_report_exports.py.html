20def _write_csv(path, fields, rows):
21 stream = io.StringIO(newline="")
22 writer = csv.DictWriter(stream, fieldnames=fields, extrasaction="ignore", lineterminator="\n")
23 writer.writeheader()
24 writer.writerows(rows)
25 content = stream.getvalue()
26 path = Path(path)
27 path.parent.mkdir(parents=True, exist_ok=True)
28 if path.exists() and path.read_text(encoding="utf-8") == content:
29 return
30 temporary = path.with_suffix(path.suffix + ".tmp")
31 temporary.write_text(content, encoding="utf-8", newline="")
32 temporary.replace(path)

35def export_report_files(detail, analysis_dir):
36 """Copy exported values, quality and definitions verbatim into tidy CSV rows."""
37 race = detail["race"]
38 summary, performance = [], []
39 for boat in detail["boats"]:
40 common = {"race_id": race["id"], "race_date": race.get("date", ""), "boat_id": boat["id"],
41 "team_code": boat["team_code"], "classification": boat["classification"],
42 "reconstructed_rank": boat.get("official_rank"),
43 "reconstructed_elapsed_s": boat.get("official_elapsed_s"),
44 "source_conflicts": "; ".join(boat.get("source_conflicts") or [])}
45 for key, metric in sorted(boat.get("metrics", {}).items()):
46 definition = detail.get("definitions", {}).get(key, {})
47 summary.append({**common, "metric": key, **metric,
48 "window": definition.get("window", ""),
49 "denominator": definition.get("denominator", ""),
50 "source": definition.get("source", ""),
51 "method_version": detail["method_version"]})
52 profile = boat.get("performance") or {}
53 for dimension in profile.get("dimensions", []):
54 base = {**common, "metric": dimension["id"], "unit": dimension["unit"],
55 "window": dimension["window"], "denominator": dimension["denominator"],
56 "source": dimension["source"], "method_version": profile["method_version"]}
57 performance.append({**base, **{k: dimension.get(k) for k in PERFORMANCE_FIELDS if k in dimension},
58 "scope": "all_comparable_legs", "peer_ids": ";".join(sorted({peer
59 for leg in dimension.get("leg_evidence", []) for peer in leg["peer_ids"]}))})
60 for leg in dimension.get("leg_evidence", []):
61 performance.append({**base, **leg, "scope": "leg",
62 "quality": leg.get("quality", dimension["quality"]),
63 "peer_ids": ";".join(leg["peer_ids"])})
64 folder = Path(analysis_dir)
65 paths = (folder / f"{race['id']}_summary.csv", folder / f"{race['id']}_performance.csv")
66 _write_csv(paths[0], SUMMARY_FIELDS, summary)
67 _write_csv(paths[1], PERFORMANCE_FIELDS, performance)
68 return paths

71def build_reports(root=ROOT):
72 folder = Path(root) / "public" / "data" / "analysis"
73 details = sorted(path for path in folder.glob("*.json") if path.name != "index.json")
74 for path in details:
75 export_report_files(json.loads(path.read_text(encoding="utf-8")), folder)
76 return {"race_count": len(details), "csv_files": len(details) * 2}
