2
3Run this before opening multi-megabyte replay or source files in a prompt.

8from pathlib import Path

11def snapshot(root: Path, race_id: str | None = None) -> dict:
12 analysis = root / "public" / "data" / "analysis"
13 index = json.loads((analysis / "index.json").read_text(encoding="utf-8"))
14 result = {"schema_version": index["schema_version"],
15 "race_count": len(index["races"]),
16 "boat_race_count": sum(r["boat_count"] for r in index["races"]),
17 "races": [{"id": r["id"], "date": r["date"], "type": r["race_type"],
18 "class": r["boat_class"], "boats": r["boat_count"],
19 "limitations": len(r.get("limitations", []))} for r in index["races"]]}
20 if race_id:
21 if race_id not in {r["id"] for r in index["races"]}:
22 raise ValueError(f"race is absent from index: {race_id}")
23 detail = json.loads((analysis / f"{race_id}.json").read_text(encoding="utf-8"))
24 result["selected_race"] = {"id": race_id,
25 "metric_method": detail["method_version"],
26 "wind_samples": detail["wind"].get("sample_count", 0),
27 "pairs": len(detail["pairs"]),
28 "boats": [{"code": b["team_code"], "status": b["classification"],
29 "complete_legs": sum(bool(l["complete"]) for l in b["legs"]),
30 "manoeuvres": len(b["manoeuvres"]),
31 "unavailable_metrics": sum(m["value"] is None for m in b["metrics"].values()),
32 "partial_metrics": sum(m["quality"] == "partial" for m in b["metrics"].values())}
33 for b in detail["boats"]]}
34 return result
