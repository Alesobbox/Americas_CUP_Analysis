13function formatRelativeClock(deltaMs: number): string {
14 const sign = deltaMs >= 0 ? '+' : '−'
15 const totalSec = Math.floor(Math.abs(deltaMs) / 1000)
16 const mm = Math.floor(totalSec / 60)
17 const ss = totalSec % 60
18 return `${sign}${String(mm).padStart(2, '0')}:${String(ss).padStart(2, '0')}`
19}

21function lowerBound(values: number[], target: number): number {
22 let low = 0
23 let high = values.length
24 while (low < high) {
25 const mid = (low + high) >>> 1
26 if (values[mid] < target) low = mid + 1
27 else high = mid
28 }
29 return low
30}
