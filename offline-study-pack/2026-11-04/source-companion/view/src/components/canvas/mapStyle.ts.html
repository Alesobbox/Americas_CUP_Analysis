43export function createMaritimeMapStyle(customConfig?: Partial<MapTileConfig>): StyleSpecification {
44 const config = { ...DEFAULT_MAP_TILE_CONFIG, ...customConfig }
45
46 const attributionHtml = config.attributionLink

48 : config.attribution
49
50 const layers: StyleSpecification['layers'] = [
51 {
52 id: 'background-ocean',
53 type: 'background',
54 paint: {
55 'background-color': config.backgroundColor,
56 },
57 },
58 ]
59
60 const sources: StyleSpecification['sources'] = {}
61
62 if (config.enabled) {
63 sources['osm-raster-source'] = {
64 type: 'raster',
65 tiles: [config.urlTemplate],
66 tileSize: config.tileSize,
67 maxzoom: config.maxZoom,
68 attribution: attributionHtml,
69 }
70
71 layers.push({
72 id: 'osm-raster-layer',
73 type: 'raster',
74 source: 'osm-raster-source',
75 minzoom: 0,
76 maxzoom: config.maxZoom,
77 paint: {
78 'raster-opacity': config.rasterOpacity,
79 'raster-saturation': config.rasterSaturation,
80 'raster-contrast': config.rasterContrast,
81 'raster-brightness-min': config.rasterBrightnessMin,
82 'raster-brightness-max': config.rasterBrightnessMax,
83 },
84 })
85 }

90 layers,
91 }
