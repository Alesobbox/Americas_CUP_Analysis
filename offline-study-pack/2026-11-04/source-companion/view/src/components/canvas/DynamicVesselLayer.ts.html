50export function drawWindFlowField(
51 context: CanvasRenderingContext2D,
52 width: number,
53 height: number,
54 replay: ReplayManifest,
55 timestamp: number,
56 viewport: Viewport,
57): void {
58 const wind = windAt(replay, timestamp)
59 if (!Number.isFinite(wind.twd) || !Number.isFinite(wind.tws)) return
60 // Atmospheric airflow travels downwind towards (wind.twd + 180) % 360
61 const flowRad = (((wind.twd + 180) % 360) * Math.PI) / 180
62 const [vx, vy] = worldVectorToScreen(viewport, [Math.sin(flowRad), Math.cos(flowRad)])
63 const mag = Math.hypot(vx, vy) || 1
64 const dx = vx / mag
65 const dy = vy / mag
66
67 const spacing = 85
68 // Directional arrows describe the recorded wind without decorative drift.
69 const drift = 0
70 const shaftLength = 12 + Math.min(6, wind.tws * 0.35)
71 const barbLength = 3.5
72 const barbAngle = 0.45
73 const angle = Math.atan2(dy, dx)
74 const cosMinus = Math.cos(angle - barbAngle)
75 const sinMinus = Math.sin(angle - barbAngle)
76 const cosPlus = Math.cos(angle + barbAngle)
77 const sinPlus = Math.sin(angle + barbAngle)
78
79 context.save()
80 context.strokeStyle = COURSE_PALETTE.airflow
81 context.lineWidth = 1.0
82 context.beginPath()
83
84 for (let y = -spacing; y < height + spacing; y += spacing) {
85 const rowOffset = (Math.abs(Math.floor(y / spacing)) % 2) * (spacing * 0.5)
86 for (let x = -spacing; x < width + spacing; x += spacing) {
87 const px = x + rowOffset + dx * drift
88 const py = y + dy * drift
89 if (px < -15 || px > width + 15 || py < -15 || py > height + 15) continue
90
91 const x2 = px + dx * shaftLength
92 const y2 = py + dy * shaftLength
93
94 context.moveTo(px, py)
95 context.lineTo(x2, y2)
96 context.moveTo(x2, y2)
97 context.lineTo(x2 - barbLength * cosMinus, y2 - barbLength * sinMinus)
98 context.moveTo(x2, y2)
99 context.lineTo(x2 - barbLength * cosPlus, y2 - barbLength * sinPlus)
100 }
101 }
102
103 context.stroke()
104 context.restore()
105}

303function drawTrail(
304 context: CanvasRenderingContext2D,
305 replay: ReplayManifest,
306 boat: Competitor,
307 frame: ReplayFrame,
308 viewport: Viewport,
309 length: TrailLength,
310 colorMode: TrailColor,
311): void {
312 if (length === 'off') return
313 const data = replay.telemetry.boats[boat.id]
314 if (!data) return
315 let start = 0
316 if (length !== 'race') {
317 const seconds = length === 'leg' ? Infinity : Number.parseInt(length, 10)
318 if (length === 'leg') {
319 const leg = data.leg[frame.index]
320 start = frame.index
321 while (start > 0 && data.leg[start - 1] === leg) start -= 1
322 } else {
323 start = Math.max(0, frame.index - seconds * 10)
324 }
325 }
326 const stride = Math.max(1, Math.ceil((frame.index - start) / 500))
327 context.save()
328 context.lineWidth = 1.6
329 context.lineCap = 'round'
330
331 for (let index = start; index < frame.index; index += stride) {
332 const next = Math.min(frame.index, index + stride)
333 const from = worldToScreen(viewport, [data.x[index], data.y[index]])
334 const to = worldToScreen(viewport, [data.x[next], data.y[next]])
335 const strokeColor =
336 colorMode === 'team'
337 ? boat.color
338 : colorMode === 'vmg'
339 ? getVmgTrailColor(Math.abs(data.vmg[index]))
340 : colorForValue(data.sog[index], 52)
341 const opacity = 0.4 + (0.5 * (index - start)) / Math.max(1, frame.index - start)
342 context.beginPath()
343 context.moveTo(from[0], from[1])
344 context.lineTo(to[0], to[1])
345 // A quiet under-stroke keeps pale team colors legible over water and tiles.
346 context.strokeStyle = COURSE_PALETTE.traceOutline
347 context.globalAlpha = opacity
348 context.lineWidth = 3.2
349 context.stroke()
350 context.strokeStyle = strokeColor
351 context.lineWidth = 1.6
352 context.stroke()
353 }
354 context.restore()
355}

361 state: { x: number; y: number; heading: number },
362 viewport: Viewport,
363 border: BorderPoint[],
364 renderScale: number,
365): Point[] {
366 const center = worldToScreen(viewport, [state.x, state.y])
367 const scaledViewport = { ...viewport, scale: renderScale }
368 return transformLocalToWorld(border, state.x, state.y, state.heading).map((worldPoint) => {
369 const displacement = worldVectorToScreen(scaledViewport, [worldPoint[0] - state.x, worldPoint[1] - state.y])
370 return [center[0] + displacement[0], center[1] + displacement[1]]
371 })

622

624 const pB = worldToScreen(viewport, [boatB.x, boatB.y])

626 const midY = (pA[1] + pB[1]) / 2
627

630 context.strokeStyle = '#ff3b30'
631 context.lineWidth = 2.0
632 context.setLineDash([4, 3])
633 context.beginPath()

636 context.stroke()
637 context.setLineDash([])

641 context.font = '600 12px "DM Sans", sans-serif'
642 const tw = context.measureText(alertLabel).width

646 context.fillStyle = COURSE_PALETTE.label
647 context.strokeStyle = '#ff3b30'
648 context.lineWidth = 1.5

652 context.stroke()

661 // 7. Head-to-head measurement line (Match Race or comparison mode)
662 const isMatchRace =
663 replay.competitors.length === 2 &&
664 Boolean(frame.boats[replay.competitors[0].id] && frame.boats[replay.competitors[1].id])

674

676 context.setLineDash([5, 4])
677 context.strokeStyle = isMatchRace ? COURSE_PALETTE.action : 'rgba(95, 99, 104, 0.6)'
678 context.lineWidth = 1.5
679 context.beginPath()
680 context.moveTo(...p1)
681 context.lineTo(...p2)
682 context.stroke()
683 context.setLineDash([])
684

686 const midY = (p1[1] + p2[1]) / 2
687 const label = `${Math.round(distM)} m`
688 context.font = '500 12px "DM Sans", sans-serif'
689 const textWidth = context.measureText(label).width
690 const badgeW = textWidth + 16
691 const badgeH = 24
692
693 context.fillStyle = COURSE_PALETTE.label
694 context.strokeStyle = COURSE_PALETTE.labelBorder
695 context.lineWidth = 1
696 context.beginPath()
697 context.rect(midX - badgeW / 2, midY - badgeH / 2, badgeW, badgeH)
698 context.fill()
699 context.stroke()
700
701 context.fillStyle = COURSE_PALETTE.ink
702 context.textAlign = 'center'
703 context.textBaseline = 'middle'
704 context.fillText(label, midX, midY)
705 context.restore()
