73function fitMapToPoints(
74 map: MapLibreMap,
75 points: Point[],
76 replay: ReplayManifest,
77 targetBearing: number
78) {
79 if (!points || !points.length) return
80 const lngLats = points.map(([x, y]) =>
81 metricToLngLat(x, y, replay.course.origin, replay.course.projection)

83 const minLng = Math.min(...lngLats.map(([lng]) => lng))
84 const maxLng = Math.max(...lngLats.map(([lng]) => lng))
85 const minLat = Math.min(...lngLats.map(([, lat]) => lat))
86 const maxLat = Math.max(...lngLats.map(([, lat]) => lat))
87
88 map.fitBounds(
89 [
90 [minLng, minLat],
91 [maxLng, maxLat],
92 ],
93 {
94 padding: { top: 48, bottom: TOOLBELT_CLEARANCE_PX, left: 48, right: 48 },
95 bearing: targetBearing,
96 maxZoom: 16.5,
97 duration: 600,
98 }
99 )
