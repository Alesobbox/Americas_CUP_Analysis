10function initialView(params: URLSearchParams): DashboardView {
11 const requested = params.get('view') ?? params.get('tab')
12 if (params.get('modal') === 'analysis') return 'analysis'
13 if (params.get('modal') === 'summary') return 'reports'
14 if (requested === 'duel') return 'compare'
15 if (requested === 'start') return 'analysis'
16 return requested === 'analysis' || requested === 'compare' || requested === 'reports' || requested === 'forecast' ? requested : 'replay'
17}
