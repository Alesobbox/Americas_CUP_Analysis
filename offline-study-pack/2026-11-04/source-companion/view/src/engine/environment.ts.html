37export function parseRaceEnvironment(value: unknown, raceId: string): RaceEnvironment {
38 if (!record(value) || value.schema_version !== 'environment-1' || value.context !== 'historical_replay_only') invalid('unsupported historical data format.')
39 if (value.race_id !== raceId) invalid('the data belongs to another race.')
40 if (!location(value.requested_location)) invalid('invalid race location.')
41 if (!record(value.units) || value.units.time_ms !== 'Unix milliseconds UTC' || value.units.air_temperature_c !== '°C' || value.units.wave_height_m !== 'm' || value.units.water_temperature_c !== '°C') invalid('unexpected time or measurement units.')
42 if (!record(value.hourly) || !Array.isArray(value.hourly.time_ms)) invalid('missing hourly times.')
43 const times = value.hourly.time_ms
44 if (times.some((time, index) => !finite(time) || !Number.isSafeInteger(time) || !Number.isFinite(new Date(time).getTime()) || time % HOUR_MS !== 0 || (index > 0 && time <= times[index - 1]))) invalid('hourly times must be ordered UTC hours.')
45 for (const field of fields) {
46 const column = value.hourly[field]
47 if (!Array.isArray(column) || column.length !== times.length || column.some((item) => item !== null && (!finite(item) || (field === 'wave_height_m' && item < 0)))) invalid(`invalid ${field} hourly values.`)
48 }
49 if (!record(value.sources) || !record(value.errors)) invalid('missing source information.')
50 for (const product of ['air', 'marine'] as const) {
51 const error = value.errors[product]
52 if (error !== undefined && !nonempty(error)) invalid(`invalid ${product} source error.`)
53 const source = value.sources[product]
54 if (source !== undefined) {
55 if (!record(source) || source.provider !== 'Open-Meteo' || source.product !== 'historical_model' || !nonempty(source.request_url) || !/^https:\/\/[a-z0-9-]+\.open-meteo\.com\//i.test(source.request_url) || typeof source.response_sha256 !== 'string' || !/^[a-f0-9]{64}$/.test(source.response_sha256) || !nonempty(source.retrieved_at) || !Number.isFinite(Date.parse(source.retrieved_at)) || !location(source.grid_location) || source.response_file !== `responses/${source.response_sha256}.json`) invalid(`invalid ${product} model provenance.`)
56 } else if (fields.filter((field) => productFor(field) === product).some((field) => (value.hourly as RaceEnvironment['hourly'])[field].some((item) => item !== null))) {
57 invalid(`${product} values have no model provenance.`)
58 }
59 }
60 return value as unknown as RaceEnvironment
61}

63export async function loadRaceEnvironment(raceId: string, signal?: AbortSignal): Promise<RaceEnvironment> {
64 if (!/^[a-zA-Z0-9][a-zA-Z0-9_-]*$/.test(raceId)) invalid('invalid race identifier.')
65 const response = await fetch(`/data/environment/${encodeURIComponent(raceId)}.json`, { signal, cache: 'no-cache' })
66 if (!response.ok) invalid(`historical data request returned HTTP ${response.status}.`)
67 if (!response.headers.get('content-type')?.toLowerCase().includes('application/json')) invalid('historical data request did not return JSON.')
68 return parseRaceEnvironment(await response.json(), raceId)
69}

71/** The valid UTC hour contains the playhead; no future samples or gap filling. */
72export function environmentHourIndex(environment: RaceEnvironment, timestamp: number): number | null {
73 if (!Number.isFinite(timestamp)) return null
74 const times = environment.hourly.time_ms
75 let lower = 0
76 let upper = times.length
77 while (lower < upper) {
78 const middle = (lower + upper) >>> 1
79 if (times[middle] <= timestamp) lower = middle + 1
80 else upper = middle
81 }
82 const index = lower - 1
83 return index >= 0 && timestamp - times[index] < HOUR_MS ? index : null
84}

86export function environmentalReading(environment: RaceEnvironment | null, field: EnvironmentField, timestamp: number, error?: string | null): EnvironmentalReading {
87 if (!environment) return { text: error ? 'unavailable' : 'loading…', title: error || 'Loading historical environmental model data for this race.', modeled: false }
88 const product = productFor(field)
89 const sourceError = environment.errors[product]
90 if (sourceError) return { text: 'unavailable', title: sourceError, modeled: false }
91 const source = environment.sources[product]
92 if (!source) return { text: 'unavailable', title: 'No historical model response is available for this source.', modeled: false }
93 const quantity = field === 'wave_height_m' ? 'Significant wave height' : field === 'air_temperature_c' ? 'Air temperature' : 'Water temperature'
94 const index = environmentHourIndex(environment, timestamp)
95 const value = index === null ? null : environment.hourly[field][index]
96 if (index === null || value === null) return { text: 'unavailable', title: `Open-Meteo historical model has no ${quantity.toLowerCase()} value for this UTC hour. ${source.request_url}`, modeled: false }
97 const validHour = new Date(environment.hourly.time_ms[index]).toISOString().slice(0, 16).replace('T', ' ')
98 return {
99 text: `${value.toFixed(1)} ${field === 'wave_height_m' ? 'm' : '°C'}`,
100 title: `${quantity}: Open-Meteo historical model. Hourly model sample at ${validHour} UTC; displayed only within that UTC hour. Model grid ${source.grid_location.latitude}, ${source.grid_location.longitude}; not a boat measurement. ${source.request_url}`,
101 modeled: true,
102 }
