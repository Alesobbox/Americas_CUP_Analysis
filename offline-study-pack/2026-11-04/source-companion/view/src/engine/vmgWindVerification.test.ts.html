6/**
7 * WCAG 2.1 relative luminance calculation
8 * https://www.w3.org/WAI/GL/wiki/Relative_luminance
9 */
10function sRgbToLinear(channel8Bit: number): number {
11 const s = channel8Bit / 255.0
12 return s <= 0.04045 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4)
13}
14
15function relativeLuminance(r: number, g: number, b: number): number {
16 return 0.2126 * sRgbToLinear(r) + 0.7152 * sRgbToLinear(g) + 0.0722 * sRgbToLinear(b)
17}
18
19function contrastRatio(rgb1: [number, number, number], rgb2: [number, number, number]): number {
20 const l1 = relativeLuminance(...rgb1)
21 const l2 = relativeLuminance(...rgb2)
22 const lighter = Math.max(l1, l2)
23 const darker = Math.min(l1, l2)
24 return (lighter + 0.05) / (darker + 0.05)
25}
26
27function euclideanRgbDistance(c1: [number, number, number], c2: [number, number, number]): number {
28 return Math.hypot(c1[0] - c2[0], c1[1] - c2[1], c1[2] - c2[2])
29}
