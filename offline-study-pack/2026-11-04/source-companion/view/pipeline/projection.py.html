20def wgs84_to_metric(
21 lat: float,
22 lon: float,
23 lat0: float = LAT0,
24 lon0: float = LON0,
25 k_lat: float = K_LAT,
26 k_lon: float = K_LON,
27) -> Tuple[float, float]:
28 """Projects WGS84 coordinates (lat, lon) in degrees to local metric (x, y) in metres.
29
30 Convention:
31 x: Easting in metres relative to (lat0, lon0)
32 y: Northing in metres relative to (lat0, lon0)
33 """
34 x = (lon - lon0) * k_lon
35 y = (lat - lat0) * k_lat
36 return (x, y)

39def metric_to_wgs84(
40 x: float,
41 y: float,
42 lat0: float = LAT0,
43 lon0: float = LON0,
44 k_lat: float = K_LAT,
45 k_lon: float = K_LON,
46) -> Tuple[float, float]:
47 """Inverts local metric coordinates (x, y) in metres to WGS84 (lat, lon) in degrees."""
48 lon = lon0 + (x / k_lon)
49 lat = lat0 + (y / k_lat)
50 return (lat, lon)

53def project_polygon_coords(
54 points: List[Union[Dict[str, float], Tuple[float, float]]]
55) -> List[Tuple[float, float]]:
56 """Projects a list of boundary coordinate points to local metric (x, y).
57
58 Accepts points as:
59 - dict with 'x' (lon) and 'y' (lat) [America's Cup session polygon convention]
60 - dict with 'latitude' and 'longitude'
61 - tuple of (lat, lon)
62 """
63 projected: List[Tuple[float, float]] = []
64 for pt in points:
65 if isinstance(pt, dict):
66 if "latitude" in pt and "longitude" in pt:
67 lat = float(pt["latitude"])
68 lon = float(pt["longitude"])
69 elif "x" in pt and "y" in pt:
70 # In session polygons, x = longitude, y = latitude
71 lon = float(pt["x"])
72 lat = float(pt["y"])
73 else:
74 raise ValueError(f"Unrecognized coordinate dict keys: {pt.keys()}")
75 elif isinstance(pt, (tuple, list)):
76 lat = float(pt[0])
77 lon = float(pt[1])
78 else:
79 raise TypeError(f"Unsupported point type: {type(pt)}")
80 projected.append(wgs84_to_metric(lat, lon))
81 return projected

84def haversine_distance(
85 lat1: float,
86 lon1: float,
87 lat2: float,
88 lon2: float,
89 radius: float = EARTH_RADIUS_LOCAL_M,
90) -> float:
91 """Computes great-circle distance between two points on a spherical Earth in metres."""
92 if any(math.isnan(value) for value in (lat1, lon1, lat2, lon2, radius)):
93 return math.nan
94 phi1 = math.radians(lat1)
95 phi2 = math.radians(lat2)
96 dphi = math.radians(lat2 - lat1)
97 dlam = math.radians(lon2 - lon1)
98
99 sin_half_dphi = math.sin(dphi / 2.0)
100 sin_half_dlam = math.sin(dlam / 2.0)
101
102 a = (sin_half_dphi * sin_half_dphi) + (
103 math.cos(phi1) * math.cos(phi2) * sin_half_dlam * sin_half_dlam
104 )
105 # Clamp 'a' to [0.0, 1.0] to prevent floating point domain errors
106 a = min(1.0, max(0.0, a))
107 c = 2.0 * math.atan2(math.sqrt(a), math.sqrt(1.0 - a))
108 return radius * c

111def euclidean_distance(x1: float, y1: float, x2: float, y2: float) -> float:
112 """Computes 2D Euclidean distance in metric Cartesian space."""
113 return math.hypot(x2 - x1, y2 - y1)

116def verify_gate_marks_projection_accuracy(
117 tolerance_pct: float = 0.1,
118 radius: float = EARTH_RADIUS_LOCAL_M,
119) -> Dict[str, Any]:
120 """Verifies that Euclidean metric distances between gate marks match Haversine distances.
121
122 Checks:
123 1. Bottom gate: RM8 to RM6
124 2. Top gate: RM4 to RM2
125
126 Returns detailed error breakdown and boolean pass/fail status.
127 """
128 rm8 = MARKS["RM8"]
129 rm6 = MARKS["RM6"]
130 rm4 = MARKS["RM4"]
131 rm2 = MARKS["RM2"]
132
133 # 1. Bottom Gate (RM8 -> RM6)
134 x8, y8 = wgs84_to_metric(rm8.target_lat, rm8.target_lon)
135 x6, y6 = wgs84_to_metric(rm6.target_lat, rm6.target_lon)
136 d_cart_bottom = euclidean_distance(x8, y8, x6, y6)
137 d_hav_bottom = haversine_distance(
138 rm8.target_lat, rm8.target_lon, rm6.target_lat, rm6.target_lon, radius=radius
139 )
140 err_bottom_pct = abs(d_cart_bottom - d_hav_bottom) / d_hav_bottom * 100.0
141
142 # 2. Top Gate (RM4 -> RM2)
143 x4, y4 = wgs84_to_metric(rm4.target_lat, rm4.target_lon)
144 x2, y2 = wgs84_to_metric(rm2.target_lat, rm2.target_lon)
145 d_cart_top = euclidean_distance(x4, y4, x2, y2)
146 d_hav_top = haversine_distance(
147 rm4.target_lat, rm4.target_lon, rm2.target_lat, rm2.target_lon, radius=radius
148 )
149 err_top_pct = abs(d_cart_top - d_hav_top) / d_hav_top * 100.0
150
151 passed_bottom = err_bottom_pct <= tolerance_pct
152 passed_top = err_top_pct <= tolerance_pct
153
154 return {
155 "all_passed": passed_bottom and passed_top,
156 "tolerance_pct": tolerance_pct,
157 "bottom_gate": {
158 "from_mark": "RM8",
159 "to_mark": "RM6",
160 "cartesian_distance_m": d_cart_bottom,
161 "haversine_distance_m": d_hav_bottom,
162 "abs_error_m": abs(d_cart_bottom - d_hav_bottom),
163 "error_pct": err_bottom_pct,
164 "passed": passed_bottom,
165 },
166 "top_gate": {
167 "from_mark": "RM4",
168 "to_mark": "RM2",
169 "cartesian_distance_m": d_cart_top,
170 "haversine_distance_m": d_hav_top,
171 "abs_error_m": abs(d_cart_top - d_hav_top),
172 "error_pct": err_top_pct,
173 "passed": passed_top,
174 },
175 }
