378def export_replay_artifacts(
379 manifest_dict: Dict[str, Any],
380 replays_dir: str = "public/data/replays",
381 manifests_file: str = "public/data/manifests.json",
382) -> Tuple[str, str]:
383 """Writes replay JSON and upserts race into manifests.json."""
384 os.makedirs(replays_dir, exist_ok=True)
385 race_id = manifest_dict.get("id", "naples_pr1")
386 replay_path = os.path.join(replays_dir, f"{race_id}.json")
387
388 with open(replay_path, "w", encoding="utf-8") as f:
389 json.dump(manifest_dict, f, separators=(",", ":"))
390
391 # Also build / upsert manifests.json
392 os.makedirs(os.path.dirname(manifests_file), exist_ok=True)
393 event_edition = manifest_dict.get("event_edition", "37th America's Cup Preliminary Regatta Naples 2026")
394 boat_class = manifest_dict.get("boat_class", "AC40")
395
396 manifest_index = {
397 "event_edition": event_edition,
398 "boat_class": boat_class,
399 "regatta": "37th America's Cup Preliminary Regatta Naples 2026",
400 "venue": manifest_dict.get("venue", "Gulf of Naples, Italy"),
401 "races": [],
402 "regattas": [
403 {
404 "id": "naples_2026",
405 "name": "America's Cup Preliminary Regatta Naples 2026",
406 "event_edition": event_edition,
407 "boat_class": boat_class,
408 "venue": manifest_dict.get("venue", "Gulf of Naples, Italy"),
409 "races": [],
410 }
411 ],
412 }
413
414 if os.path.exists(manifests_file):
415 try:
416 with open(manifests_file, "r", encoding="utf-8") as f:
417 existing = json.load(f)
418 if isinstance(existing, dict) and "races" in existing:
419 manifest_index = existing
420 manifest_index.setdefault("event_edition", event_edition)
421 manifest_index.setdefault("boat_class", boat_class)
422 except Exception:
423 pass
424
425 comps = manifest_dict.get("competitors", [])
426 winner = next((comp for comp in comps if comp.get("classification") == "RANK_1"
427 and comp.get("summary", {}).get("finish_rank") == 1), None)
428 winner_code = winner["team_code"] if winner else None
429 duration_s = winner["summary"]["elapsed_s"] if winner else None
430 wind_kt = manifest_dict.get("wind", {}).get("reference_tws_kt", 10.7)
431
432 # Upsert into flat races list
433 new_root_entry = {
434 "id": race_id,
435 "name": manifest_dict.get("name", race_id),
436 "event_edition": event_edition,
437 "boat_class": boat_class,
438 "date": manifest_dict.get("date", "2026-09-22"),
439 "file": f"replays/{race_id}.json",
440 "winner": winner_code,
441 "duration_s": duration_s,
442 "result_provenance": "reconstructed" if winner else "unavailable",
443 }
444 races_list = [r for r in manifest_index.get("races", []) if r.get("id") != race_id]
445 races_list.append(new_root_entry)
446 races_list.sort(key=lambda x: (x.get("date", ""), x.get("id", "")))
447 manifest_index["races"] = races_list
448
449 # Upsert into the matching regatta; new events must not be placed under Naples.
450 new_regatta_entry = {
451 "id": race_id,
452 "name": manifest_dict.get("name", race_id),
453 "event_edition": event_edition,
454 "boat_class": boat_class,
455 "date": manifest_dict.get("date", "2026-09-22"),
456 "replay_file": f"replays/{race_id}.json",
457 "competitors_count": len(comps),
458 "winner": winner_code,
459 "wind_speed_kt": wind_kt,
460 "result_provenance": "reconstructed" if winner else "unavailable",
461 }
462 target_regatta_id = "cagliari_2026" if "cag" in race_id.lower() or "cagliari" in event_edition.lower() or "cagliari" in manifest_dict.get("venue", "").lower() else "naples_2026"
463 target_regatta_name = "37th America's Cup Preliminary Regatta Cagliari 2026" if target_regatta_id == "cagliari_2026" else "37th America's Cup Preliminary Regatta Naples 2026"
464 target_venue = "Cagliari, Sardinia, Italy" if target_regatta_id == "cagliari_2026" else "Gulf of Naples, Italy"
465
466 if not manifest_index.get("regattas"):
467 manifest_index["regattas"] = []
468
469 matching = next((group for group in manifest_index["regattas"] if group.get("id") == target_regatta_id), None)
470 if matching is None:
471 matching = {
472 "id": target_regatta_id,
473 "name": target_regatta_name,
474 "event_edition": event_edition,
475 "boat_class": boat_class,
476 "venue": manifest_dict.get("venue", target_venue),
477 "races": [],
478 }
479 manifest_index["regattas"].append(matching)
480
481 # Remove an existing ID from another group if metadata changed on re-ingestion.
482 for group in manifest_index["regattas"]:
483 group["races"] = [r for r in group.get("races", []) if r.get("id") != race_id]
484 reg_races = matching["races"]
485 reg_races.append(new_regatta_entry)
486 reg_races.sort(key=lambda x: (x.get("date", ""), x.get("id", "")))
487 matching["races"] = reg_races
488
489 if {group.get("id") for group in manifest_index["regattas"]} >= {"cagliari_2026", "naples_2026"}:
490 manifest_index["event_edition"] = "37th America's Cup Preliminary Regattas 2026"
491 manifest_index["regatta"] = "37th America's Cup Preliminary Regattas 2026"
492 manifest_index["venue"] = "Cagliari and Naples, Italy"
493
494 with open(manifests_file, "w", encoding="utf-8") as f:
495 json.dump(manifest_index, f, indent=2)
496
497 # Sync to dist/data if dist exists
498 dist_replays = "dist/data/replays"
499 dist_manifest = "dist/data/manifests.json"
500 if os.path.exists("dist/data"):
501 try:
502 os.makedirs(dist_replays, exist_ok=True)
503 dist_rep_path = os.path.join(dist_replays, f"{race_id}.json")
504 with open(dist_rep_path, "w", encoding="utf-8") as f:
505 json.dump(manifest_dict, f, separators=(",", ":"))
506 with open(dist_manifest, "w", encoding="utf-8") as f:
507 json.dump(manifest_index, f, indent=2)
508 except Exception:
509 pass
510
511 return replay_path, manifests_file
