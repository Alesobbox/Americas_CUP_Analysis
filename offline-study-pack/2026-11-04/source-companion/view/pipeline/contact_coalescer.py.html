39def transform_local_to_world(
40 local_coords: List[Any],
41 boat_x: float,
42 boat_y: float,
43 heading_deg: float,
44) -> List[Tuple[float, float]]:
45 """Transforms local (u starboard, v forward) coordinates to world (x East, y North)."""
46 rad = math.radians(heading_deg)
47 cos_h = math.cos(rad)
48 sin_h = math.sin(rad)
49
50 world: List[Tuple[float, float]] = []
51 for pt in local_coords:
52 if isinstance(pt, dict):
53 u = float(pt["x"])
54 v = float(pt["y"])
55 elif isinstance(pt, (list, tuple)):
56 u = float(pt[0])
57 v = float(pt[1])
58 else:
59 continue
60 x = boat_x + u * cos_h + v * sin_h
61 y = boat_y - u * sin_h + v * cos_h
62 world.append((x, y))
63 return world

73def test_sat_overlap(
74 poly_a: List[Tuple[float, float]],
75 poly_b: List[Tuple[float, float]],
76) -> Tuple[bool, float]:
77 """Tests 2D collision between two strictly convex polygons using SAT.
78
79 Returns (has_overlap, penetration_depth_m).
80 """
81 verts_a = _strip_closing_vertex(poly_a)
82 verts_b = _strip_closing_vertex(poly_b)
83
84 if len(verts_a) < 3 or len(verts_b) < 3:
85 return False, 0.0
86
87 min_overlap = float("inf")
88 polygons = [verts_a, verts_b]
89
90 for p in range(2):
91 poly = polygons[p]
92 count = len(poly)
93 for i in range(count):
94 p1 = poly[i]
95 p2 = poly[(i + 1) % count]
96
97 ex = p2[0] - p1[0]
98 ey = p2[1] - p1[1]
99 edge_len = math.hypot(ex, ey)
100 if edge_len < 1e-9:
101 continue
102
103 nx = -ey / edge_len
104 ny = ex / edge_len
105
106 # Project poly_a
107 min_a = float("inf")
108 max_a = float("-inf")
109 for pt in verts_a:
110 dot = pt[0] * nx + pt[1] * ny
111 if dot < min_a:
112 min_a = dot
113 if dot > max_a:
114 max_a = dot
115
116 # Project poly_b
117 min_b = float("inf")
118 max_b = float("-inf")
119 for pt in verts_b:
120 dot = pt[0] * nx + pt[1] * ny
121 if dot < min_b:
122 min_b = dot
123 if dot > max_b:
124 max_b = dot
125
126 overlap = min(max_a, max_b) - max(min_a, min_b)
127 if overlap <= 0.0:
128 return False, 0.0
129
130 if overlap < min_overlap:
131 min_overlap = overlap
132
133 return True, min_overlap if min_overlap != float("inf") else 0.0

136def compute_polygon_area(poly: List[Tuple[float, float]]) -> float:
137 """Computes area of a 2D polygon using the Shoelace formula."""
138 n = len(poly)
139 if n < 3:
140 return 0.0
141 area = 0.0
142 for i in range(n):
143 j = (i + 1) % n
144 area += poly[i][0] * poly[j][1] - poly[j][0] * poly[i][1]
145 return abs(area) * 0.5

148def _clip_polygon_by_line(
149 subject: List[Tuple[float, float]],
150 p1: Tuple[float, float],
151 p2: Tuple[float, float],
152) -> List[Tuple[float, float]]:
153 result: List[Tuple[float, float]] = []
154 if not subject:
155 return result
156
157 def is_inside(pt: Tuple[float, float]) -> bool:
158 return (p2[0] - p1[0]) * (pt[1] - p1[1]) - (p2[1] - p1[1]) * (pt[0] - p1[0]) >= -1e-9
159
160 def intersection(s: Tuple[float, float], e: Tuple[float, float]) -> Tuple[float, float]:
161 a1 = e[1] - s[1]
162 b1 = s[0] - e[0]
163 c1 = a1 * s[0] + b1 * s[1]
164
165 a2 = p2[1] - p1[1]
166 b2 = p1[0] - p2[0]
167 c2 = a2 * p1[0] + b2 * p1[1]
168
169 det = a1 * b2 - a2 * b1
170 if abs(det) < 1e-11:
171 return s
172 return ((b2 * c1 - b1 * c2) / det, (a1 * c2 - a2 * c1) / det)
173
174 s = subject[-1]
175 for e in subject:
176 if is_inside(e):
177 if not is_inside(s):
178 result.append(intersection(s, e))
179 result.append(e)
180 elif is_inside(s):
181 result.append(intersection(s, e))
182 s = e
183 return result

186def compute_polygon_intersection_area(
187 poly_a: List[Tuple[float, float]],
188 poly_b: List[Tuple[float, float]],
189) -> float:
190 """Computes exact 2D intersection area (m^2) between two convex polygons."""
191 subject = list(_strip_closing_vertex(poly_a))
192 clipper = list(_strip_closing_vertex(poly_b))
193
194 if len(subject) < 3 or len(clipper) < 3:
195 return 0.0
196
197 # Ensure clipper is counter-clockwise
198 area_sum = 0.0
199 for i in range(len(clipper)):
200 j = (i + 1) % len(clipper)
201 area_sum += clipper[i][0] * clipper[j][1] - clipper[j][0] * clipper[i][1]
202 if area_sum < 0:
203 clipper.reverse()
204
205 for i in range(len(clipper)):
206 p1 = clipper[i]
207 p2 = clipper[(i + 1) % len(clipper)]
208 subject = _clip_polygon_by_line(subject, p1, p2)
209 if len(subject) < 3:
210 return 0.0
211
212 return compute_polygon_area(subject)

215def coalesce_virtual_contacts(
216 grid_timestamps_ms: List[int],
217 boats: Dict[str, Any],
218 keep_clear_border_coords: Optional[List[Any]] = None,
219 bounding_radius_m: float = 15.0,
220) -> List[Dict[str, Any]]:
221 """Evaluates pairwise Keep Clear Border overlaps across 100ms frames and coalesces incidents.
222
223 boats can be:
224 - Dict mapping code/id -> SynchronizedBoatData (with .x, .y, .heading, .code, .asset_id)
225 - Dict mapping id -> {"x": [...], "y": [...], "heading": [...], "code": "..."}
226 """
227 if keep_clear_border_coords is None:
228 keep_clear_border_coords = DEFAULT_KEEP_CLEAR_BORDER
229
230 n_frames = len(grid_timestamps_ms)
231 boat_keys = list(boats.keys())
232
233 # Extract arrays and metadata for each boat
234 boat_metadata: Dict[str, Dict[str, Any]] = {}
235 for key in boat_keys:
236 b = boats[key]
237 if hasattr(b, "x"):
238 # Dataclass SynchronizedBoatData
239 boat_metadata[key] = {
240 "id": getattr(b, "asset_id", key),
241 "code": getattr(b, "code", key),
242 "x": b.x,
243 "y": b.y,
244 "heading": b.heading,
245 }
246 elif isinstance(b, dict):
247 # Dict structure
248 tel = b.get("telemetry", b)
249 boat_metadata[key] = {
250 "id": b.get("id", key),
251 "code": b.get("code", b.get("team_code", key)),
252 "x": tel.get("x", []),
253 "y": tel.get("y", []),
254 "heading": tel.get("heading", []),
255 }
256
257 active_incidents: Dict[Tuple[str, str], Dict[str, Any]] = {}
258 closed_incidents: List[Dict[str, Any]] = []
259
260 # Broadphase radius threshold: 2 * 15m = 30m -> 30^2 = 900
261 bounding_threshold_sq = (2 * bounding_radius_m) ** 2
262
263 for f_idx in range(n_frames):
264 t_ms = grid_timestamps_ms[f_idx]
265 current_frame_contacts = set()
266
267 for i in range(len(boat_keys)):
268 k_a = boat_keys[i]
269 b_a = boat_metadata[k_a]
270 if f_idx >= len(b_a["x"]) or f_idx >= len(b_a["y"]) or f_idx >= len(b_a["heading"]):
271 continue
272 xa = b_a["x"][f_idx]
273 ya = b_a["y"][f_idx]
274 hdg_a = b_a["heading"][f_idx]
275
276 for j in range(i + 1, len(boat_keys)):
277 k_b = boat_keys[j]
278 b_b = boat_metadata[k_b]
279 if f_idx >= len(b_b["x"]) or f_idx >= len(b_b["y"]) or f_idx >= len(b_b["heading"]):
280 continue
281 xb = b_b["x"][f_idx]
282 yb = b_b["y"][f_idx]
283 hdg_b = b_b["heading"][f_idx]
284
285 # 1. Broadphase bounding filter
286 dx = xb - xa
287 dy = yb - ya
288 dist_sq = dx * dx + dy * dy
289 if dist_sq > bounding_threshold_sq:
290 continue
291
292 center_dist = math.sqrt(dist_sq)
293
294 # 2. Narrowphase SAT
295 poly_a = transform_local_to_world(keep_clear_border_coords, xa, ya, hdg_a)
296 poly_b = transform_local_to_world(keep_clear_border_coords, xb, yb, hdg_b)
297 has_contact, depth = test_sat_overlap(poly_a, poly_b)
298
299 pair_key = (k_a, k_b)
300 if has_contact:
301 current_frame_contacts.add(pair_key)
302 overlap_area = compute_polygon_intersection_area(poly_a, poly_b)
303
304 if pair_key not in active_incidents:
305 active_incidents[pair_key] = {
306 "id": f"vci-{b_a['code']}-{b_b['code']}-{t_ms}",
307 "boat_a": b_a["code"],
308 "boat_b": b_b["code"],
309 "boat_a_id": b_a["id"],
310 "boat_b_id": b_b["id"],
311 "boat_a_code": b_a["code"],
312 "boat_b_code": b_b["code"],
313 "start_ms": t_ms,
314 "end_ms": t_ms,
315 "duration_ms": 0,
316 "peak_ms": t_ms,
317 "min_center_distance_m": round(center_dist, 2),
318 "peak_overlap_m": round(depth, 3),
319 "peak_overlap_m2": round(overlap_area, 3),
320 "start_frame_idx": f_idx,
321 "end_frame_idx": f_idx,
322 "type": "KEEP_CLEAR_BORDER_CROSSING",
323 }
324 else:
325 inc = active_incidents[pair_key]
326 inc["end_ms"] = t_ms
327 inc["end_frame_idx"] = f_idx
328 inc["duration_ms"] = inc["end_ms"] - inc["start_ms"]
329
330 if center_dist < inc["min_center_distance_m"]:
331 inc["min_center_distance_m"] = round(center_dist, 2)
332
333 if depth > inc["peak_overlap_m"]:
334 inc["peak_overlap_m"] = round(depth, 3)
335 inc["peak_ms"] = t_ms
336
337 if overlap_area > inc["peak_overlap_m2"]:
338 inc["peak_overlap_m2"] = round(overlap_area, 3)
339
340 # Close incidents that ended on this frame
341 ended_pairs = [pair for pair in active_incidents if pair not in current_frame_contacts]
342 for pair in ended_pairs:
343 inc = active_incidents.pop(pair)
344 inc["duration_ms"] = inc["end_ms"] - inc["start_ms"]
345 closed_incidents.append(inc)
346
347 # Close any remaining active incidents at dataset conclusion
348 for inc in active_incidents.values():
349 inc["duration_ms"] = inc["end_ms"] - inc["start_ms"]
350 closed_incidents.append(inc)
351
352 closed_incidents.sort(key=lambda x: x["start_ms"])
353 return closed_incidents
