48def _write_bytes(path: Path, data: bytes):
49 path.parent.mkdir(parents=True, exist_ok=True)
50 temporary = None
51 try:
52 with tempfile.NamedTemporaryFile(dir=path.parent, prefix=".environment-", delete=False) as stream:
53 temporary = Path(stream.name)
54 stream.write(data)
55 stream.flush()
56 os.fsync(stream.fileno())
57 os.replace(temporary, path)
58 finally:
59 if temporary is not None and temporary.exists():
60 temporary.unlink()

67def _parse_retrieved(value):
68 if not isinstance(value, str):
69 raise ValueError("Missing response retrieval timestamp")

76def replay_metadata(path: Path):
77 """Decode only top-level metadata from a bounded prefix, before telemetry.
78
79 All current catalog metadata ends before 112 KiB. A layout exceeding the
80 1 MiB contract fails explicitly rather than loading replay telemetry arrays.
81 """
82 with path.open("rb") as stream:
83 prefix = stream.read(MAX_METADATA_BYTES).decode("utf-8")
84 decoder = json.JSONDecoder()
85 position = 0
86 result = {}
87 wanted = {"id", "course", "chronology"}
88 try:
89 position = len(prefix) - len(prefix.lstrip())
90 if prefix[position] != "{":
91 raise ValueError("Replay metadata must be an object")
92 position += 1
93 while wanted - result.keys():
94 while prefix[position].isspace() or prefix[position] == ",":
95 position += 1
96 key, position = decoder.raw_decode(prefix, position)
97 if not isinstance(key, str):
98 raise ValueError("Invalid replay metadata key")
99 if key == "telemetry":
100 raise ValueError("Required replay metadata must precede telemetry")
101 while prefix[position].isspace():
102 position += 1
103 if prefix[position] != ":":
104 raise ValueError("Invalid replay metadata separator")
105 position += 1
106 while prefix[position].isspace():
107 position += 1
108 value, position = decoder.raw_decode(prefix, position)
109 if key in wanted:
110 if key in result:
111 raise ValueError("Duplicate required replay metadata")
112 result[key] = value
113 except (IndexError, json.JSONDecodeError) as exc:
114 raise ValueError("Required replay metadata exceeds bounded prefix or is malformed") from exc
115 return validate_metadata(result)

118def validate_metadata(metadata):
119 race_id = metadata.get("id")
120 if not isinstance(race_id, str) or not re.fullmatch(r"[A-Za-z0-9_-]+", race_id):
121 raise ValueError("Invalid race ID")
122 origin = metadata.get("course", {}).get("origin", {})
123 latitude, longitude = origin.get("lat"), origin.get("lon")
124 if (not _finite(latitude) or not -90 <= latitude <= 90
125 or not _finite(longitude) or not -180 <= longitude <= 180):
126 raise ValueError("Replay course origin must contain valid latitude/longitude")
127 chronology = metadata.get("chronology", {})
128 start, end = chronology.get("window_start"), chronology.get("window_end")
129 if (not _finite(start) or not _finite(end) or int(start) != start or int(end) != end
130 or end < start or end - start > 7 * 86400000):
131 raise ValueError("Invalid replay UTC window")
132 try:
133 start_date = datetime.fromtimestamp(start / 1000, timezone.utc).date().isoformat()
134 end_date = datetime.fromtimestamp(end / 1000, timezone.utc).date().isoformat()
135 except (OverflowError, OSError, ValueError) as exc:
136 raise ValueError("Replay UTC window is out of range") from exc
137 return {"race_id": race_id, "latitude": latitude, "longitude": longitude,
138 "start_date": start_date, "end_date": end_date,
139 "window_start_ms": int(start), "window_end_ms": int(end)}

142def request_url(source, metadata):
143 parameters = {
144 "latitude": str(metadata["latitude"]), "longitude": str(metadata["longitude"]),
145 "start_date": metadata["start_date"], "end_date": metadata["end_date"],

154def fetch_bytes(url, timeout=TIMEOUT_SECONDS):
155 request = Request(url, headers={"Accept": "application/json", "User-Agent": "AC40Replay/1.0 historical-context"})
156 try:
157 with urlopen(request, timeout=timeout) as response:
158 content_type = response.headers.get("Content-Type", "").split(";", 1)[0].strip().lower()
159 if response.status != 200:
160 raise ValueError(f"Weather provider returned HTTP {response.status}")
161 if content_type != "application/json":
162 raise ValueError("Weather provider response is not JSON")
163 data = response.read(MAX_RESPONSE_BYTES + 1)
164 except HTTPError as exc:
165 raise ValueError(f"Weather provider returned HTTP {exc.code}") from exc
166 except (URLError, TimeoutError, OSError) as exc:
167 raise ValueError(f"Weather provider request failed: {exc}") from exc
168 if len(data) > MAX_RESPONSE_BYTES:
169 raise ValueError("Weather provider response exceeds size limit")
170 return data

173def parse_response(data: bytes, source: str, metadata):
174 if len(data) > MAX_RESPONSE_BYTES:
175 raise ValueError("Weather provider response exceeds size limit")
176 try:
177 body = json.loads(data)
178 except (json.JSONDecodeError, UnicodeDecodeError) as exc:
179 raise ValueError("Weather provider returned malformed JSON") from exc
180 if not isinstance(body, dict) or body.get("error"):
181 reason = body.get("reason", "Invalid provider object") if isinstance(body, dict) else "Invalid provider object"
182 raise ValueError(f"Weather provider error: {str(reason)[:240]}")
183 lat, lon = body.get("latitude"), body.get("longitude")
184 if (not _finite(lat) or not -90 <= lat <= 90 or not _finite(lon) or not -180 <= lon <= 180):
185 raise ValueError("Weather provider grid location is invalid")
186 if type(body.get("utc_offset_seconds")) is not int or body["utc_offset_seconds"] != 0:
187 raise ValueError("Weather provider response is not aligned to UTC")
188 hourly, units = body.get("hourly"), body.get("hourly_units")
189 if not isinstance(hourly, dict) or not isinstance(units, dict) or units.get("time") != "unixtime":
190 raise ValueError("Missing hourly data or Unix-time units")
191 times = hourly.get("time")
192 if not isinstance(times, list) or not times:
193 raise ValueError("Missing hourly timestamps")
194 start = int(datetime.fromisoformat(metadata["start_date"]).replace(tzinfo=timezone.utc).timestamp())
195 end = int(datetime.fromisoformat(metadata["end_date"]).replace(tzinfo=timezone.utc).timestamp()) + 86400
196 if (any(not isinstance(t, int) or isinstance(t, bool) or t % 3600 or not start <= t < end for t in times)
197 or any(a >= b for a, b in zip(times, times[1:]))):
198 raise ValueError("Hourly timestamps must be unique ascending UTC hours inside the requested days")
199 columns = {}
200 for variable, (field, unit) in VARIABLES[source].items():
201 values = hourly.get(variable)
202 if units.get(variable) != unit:
203 raise ValueError(f"Unexpected units for {variable}")
204 if not isinstance(values, list) or len(values) != len(times):
205 raise ValueError(f"Misaligned hourly values for {variable}")
206 if any(v is not None and (not _finite(v) or (variable == "wave_height" and v < 0)) for v in values):
207 raise ValueError(f"Invalid finite values for {variable}")
208 columns[field] = dict(zip((t * 1000 for t in times), values))
209 return columns, {"latitude": lat, "longitude": lon}

212def acquire_source(source, metadata, public_directory, cache_directory, *, fetcher=None, force=False, clock=None):
213 fetcher, clock = fetcher or fetch_bytes, clock or _utc_now
214 url = request_url(source, metadata)
215 cache_path = cache_directory / (hashlib.sha256(url.encode()).hexdigest() + ".json")
216 record, data = None, None
217 if not force and cache_path.exists():
218 record = json.loads(cache_path.read_bytes())
219 if record.get("request_url") != url or not re.fullmatch(r"[0-9a-f]{64}", record.get("response_sha256", "")):
220 raise ValueError("Historical weather cache request/hash mismatch; rebuild with --force")
221 _parse_retrieved(record.get("retrieved_at"))
222 data = (public_directory / "responses" / (record["response_sha256"] + ".json")).read_bytes()
223 if hashlib.sha256(data).hexdigest() != record["response_sha256"]:
224 raise ValueError("Historical weather cache content hash mismatch; rebuild with --force")
225 if data is None:
226 data = fetcher(url, timeout=TIMEOUT_SECONDS)
227 record = {"request_url": url, "response_sha256": hashlib.sha256(data).hexdigest(),
228 "retrieved_at": _parse_retrieved(clock())}
229 columns, grid = parse_response(data, source, metadata)
230 response_file = "responses/" + record["response_sha256"] + ".json"
231 _write_bytes(public_directory / response_file, data)
232 _write_bytes(cache_path, _json_bytes(record))
233 provenance = {"provider": "Open-Meteo", "product": "historical_model", **record,
234 "response_file": response_file, "grid_location": grid}
235 return columns, provenance

238def build_race(metadata, public_directory, cache_directory, *, fetcher=None, force=False, clock=None):
239 sources, errors, columns = {}, {}, {}
240 for source in VARIABLES:
241 try:
242 source_columns, provenance = acquire_source(source, metadata, public_directory, cache_directory,
243 fetcher=fetcher, force=force, clock=clock)
244 columns.update(source_columns)
245 sources[source] = provenance
246 except (ValueError, OSError, TypeError) as exc:
247 errors[source] = str(exc)[:360]
248 times = sorted({t for column in columns.values() for t in column})
249 artifact = {
250 "schema_version": "environment-1", "race_id": metadata["race_id"],
251 "context": "historical_replay_only",
252 "requested_location": {"latitude": metadata["latitude"], "longitude": metadata["longitude"]},
253 "replay_window": {"start_ms": metadata["window_start_ms"], "end_ms": metadata["window_end_ms"]},
254 "units": {"time_ms": "Unix milliseconds UTC", "air_temperature_c": "°C", "wave_height_m": "m",
255 "water_temperature_c": "°C"},
256 "hourly": {"time_ms": times, **{f: [columns.get(f, {}).get(t) for t in times] for f in OUTPUT_FIELDS}},
257 "sources": sources, "errors": errors,
258 }
259 _write_bytes(public_directory / (metadata["race_id"] + ".json"), _json_bytes(artifact))
260 return artifact

263def catalog_metadata(root, race_id=None):
264 data_directory = root / "public" / "data"
265 catalog = json.loads((data_directory / "manifests.json").read_bytes())
266 races = catalog.get("races") or [r for regatta in catalog.get("regattas", []) for r in regatta["races"]]
267 seen = set()
268 for race in races:
269 if race_id and race.get("id") != race_id:
270 continue
271 path = (data_directory / (race.get("replay_file") or race["file"])).resolve()
272 if not path.is_relative_to((data_directory / "replays").resolve()) or path.suffix != ".json":
273 raise ValueError("Catalog replay path must remain within public/data/replays")
274 metadata = replay_metadata(path)
275 if metadata["race_id"] != race["id"] or metadata["race_id"] in seen:
276 raise ValueError("Catalog replay ID mismatch or duplicate")
277 seen.add(metadata["race_id"])
278 yield metadata
279 if not seen:
280 raise ValueError("No matching replay races in catalog")

293def verify_archive(root: Path, *, race_id=None):
294 """Verify every catalog artifact and its original response bytes, offline."""
295 output = root / "public" / "data" / "environment"
296 counts = dict.fromkeys(OUTPUT_FIELDS, 0)
297 response_hashes, races, source_errors = set(), 0, {}
298 for metadata in catalog_metadata(root, race_id):
299 artifact = json.loads((output / (metadata["race_id"] + ".json")).read_bytes())
300 if (artifact.get("schema_version") != "environment-1" or artifact.get("race_id") != metadata["race_id"]
301 or artifact.get("context") != "historical_replay_only"
302 or artifact.get("requested_location") != {"latitude": metadata["latitude"], "longitude": metadata["longitude"]}
303 or artifact.get("replay_window") != {"start_ms": metadata["window_start_ms"], "end_ms": metadata["window_end_ms"]}
304 or artifact.get("units") != {"time_ms": "Unix milliseconds UTC", "air_temperature_c": "°C",
305 "wave_height_m": "m", "water_temperature_c": "°C"}):
306 raise ValueError(f"Environment metadata/unit mismatch: {metadata['race_id']}")
307 sources, errors = artifact.get("sources"), artifact.get("errors")
308 if (not isinstance(sources, dict) or not isinstance(errors, dict)
309 or set(sources) & set(errors) or set(sources) | set(errors) != set(VARIABLES)
310 or any(not isinstance(reason, str) or not reason for reason in errors.values())):
311 raise ValueError(f"Environment source/error contract mismatch: {metadata['race_id']}")
312 columns = {}
313 for source, provenance in sources.items():
314 if (not isinstance(provenance, dict) or provenance.get("provider") != "Open-Meteo"
315 or provenance.get("product") != "historical_model"
316 or provenance.get("request_url") != request_url(source, metadata)
317 or not re.fullmatch(r"[0-9a-f]{64}", provenance.get("response_sha256", ""))
318 or provenance.get("response_file") != "responses/" + provenance["response_sha256"] + ".json"):
319 raise ValueError(f"Environment response provenance mismatch: {metadata['race_id']}/{source}")
320 _parse_retrieved(provenance.get("retrieved_at"))
321 data = (output / provenance["response_file"]).read_bytes()
322 if hashlib.sha256(data).hexdigest() != provenance["response_sha256"]:
323 raise ValueError(f"Environment response hash mismatch: {metadata['race_id']}/{source}")
324 source_columns, grid = parse_response(data, source, metadata)
325 if provenance.get("grid_location") != grid:
326 raise ValueError(f"Environment grid location mismatch: {metadata['race_id']}/{source}")
327 columns.update(source_columns)
328 response_hashes.add(provenance["response_sha256"])
329 times = sorted({t for values in columns.values() for t in values})
330 expected = {"time_ms": times, **{f: [columns.get(f, {}).get(t) for t in times] for f in OUTPUT_FIELDS}}
331 if artifact.get("hourly") != expected:
332 raise ValueError(f"Environment hourly values differ from original responses: {metadata['race_id']}")
333 for field in OUTPUT_FIELDS:
334 counts[field] += any(v is not None for v in expected[field])
335 races += 1
336 if errors:
337 source_errors[metadata["race_id"]] = errors
338 return {"status": "pass", "races": races, "verified_response_objects": len(response_hashes),
339 "races_with_available_field": counts, "races_with_source_errors": source_errors}
