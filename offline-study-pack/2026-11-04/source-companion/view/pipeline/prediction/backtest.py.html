43 "empirical":empirical.predict_win_probabilities(order),

45 "condition_pl":condition.predict_win_probabilities(order,wind_knots=label.get("wind_knots"))} if label["order_valid"] else {}

48 diagnostic_p["correlated_mc"] = mc.win_prob_by_boat if mc.available else diagnostic_p["static_pl"]
49 for model,p in diagnostic_p.items():
50 values=np.array([p[b] for b in order],float)
51 # Fixed tail on the full conditional-order distribution, not a score repair.
52 if model != "correlated_mc" or not mc.available:
53 values=(1-.0001)*values+.0001/len(order)
54 diagnostics[model].append({"date":label["date"],"race_id":label["race_id"],

102 n=n,seed=42,global_finish_order_resolved=resolved)

104 global_finish_order_resolved=resolved)
105 if not reference.available:
106 reasons["live_reference_unavailable"]+=1
107 continue
108 for key,prediction in (("reference",reference),("candidate",candidate if candidate.available else reference)):
109 p=[prediction.win_prob_by_boat[b] for b in entries]+[prediction.no_finisher_prob]
110 winner=entries.index(order[0]) if order else len(entries)
111 row={"date":label["date"],"race_id":label["race_id"],"checkpoint":t,**scores(p,winner)}
112 primary[key].append(row)
113 progress_rows[key][band].append(row)

116 development={k:split_summary(v,DEVELOPMENT_DAYS) for k,v in primary.items()}
