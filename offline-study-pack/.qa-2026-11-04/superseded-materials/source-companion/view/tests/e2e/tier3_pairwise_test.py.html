250 def test_closest_point_of_approach_during_crossing(self):
251 """CPA calculation determines safe passing distance when boats cross on opposite tacks."""
252 # Boat A moving East at 15 m/s from (-50, 0)
253 # Boat B moving North at 15 m/s from (0, -50)
254 # CPA occurs at (0, 0) at t = 50 / 15 = 3.33s with distance 0m (collision trajectory)
255 pos_a = (-50.0, 0.0)
256 pos_b = (0.0, -50.0)
257 v_a = (15.0, 0.0)
258 v_b = (0.0, 15.0)
259
260 # Relative pos & vel
261 rx = pos_a[0] - pos_b[0] # -50
262 ry = pos_a[1] - pos_b[1] # +50
263 vx = v_a[0] - v_b[0] # +15
264 vy = v_a[1] - v_b[1] # -15
265
266 # t_cpa = - (r . v) / (v . v)
267 r_dot_v = rx * vx + ry * vy # -750 - 750 = -1500
268 v_sq = vx**2 + vy**2 # 225 + 225 = 450
269 t_cpa = -r_dot_v / v_sq # 1500 / 450 = 3.333s
270
271 self.assertAlmostEqual(t_cpa, 3.333, places=2)
