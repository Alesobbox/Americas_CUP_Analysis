40def run_tier(tier_num: int, verbosity: int = 2) -> Tuple[unittest.TestResult, float]:
41 """Runs tests for a specific tier and returns result and duration."""
42 module_name = TIER_MODULES[tier_num]
43 loader = unittest.defaultTestLoader
44 suite = loader.loadTestsFromName(module_name)
45
46 start_time = time.time()
47 runner = unittest.TextTestRunner(verbosity=verbosity)
48 result = runner.run(suite)
49 elapsed = time.time() - start_time
50
51 return result, elapsed

60def main():
61 parser = argparse.ArgumentParser(description="AC40 Regatta Replay E2E Test Runner")
62 parser.add_argument(
63 "--tier",
64 type=str,
65 default="all",
66 choices=["1", "2", "3", "4", "all"],
67 help="Test tier to execute (1, 2, 3, 4, or all)",

69 parser.add_argument(
70 "--verbose",

72 action="store_true",

75 args = parser.parse_args()
76
77 verbosity = 2 if args.verbose else 1
78
79 tiers_to_run = [1, 2, 3, 4] if args.tier == "all" else [int(args.tier)]
80
81 print_banner("AC40 AMERICA'S CUP 2D REGATTA REPLAY — E2E TEST RUNNER")
82 print(f"Workspace: {WORKSPACE_ROOT}")
83 print(f"Executing Tiers: {tiers_to_run}\n")
84
85 summary_rows = []
86 total_tests = 0
87 total_failures = 0
88 total_errors = 0
89 total_skipped = 0
90 total_time = 0.0
91
92 overall_success = True
93
94 for t in tiers_to_run:
95 print_banner(f"RUNNING {TIER_NAMES[t]}")
96 res, elapsed = run_tier(t, verbosity=verbosity)
97
98 tests_run = res.testsRun
99 fails = len(res.failures)
100 errs = len(res.errors)
101 skips = len(res.skipped)
102 passed = tests_run - fails - errs - skips
103
104 total_tests += tests_run
105 total_failures += fails
106 total_errors += errs
107 total_skipped += skips
108 total_time += elapsed
109
110 if fails > 0 or errs > 0:
111 overall_success = False
112
113 status = "PASS" if (fails == 0 and errs == 0) else "FAIL"
114 summary_rows.append({
115 "tier": f"Tier {t}",
116 "name": TIER_NAMES[t].split("—")[1].strip() if "—" in TIER_NAMES[t] else TIER_NAMES[t],
117 "total": tests_run,
118 "passed": passed,
119 "failed": fails,
120 "errors": errs,
121 "skipped": skips,
122 "time_s": f"{elapsed:.2f}s",
123 "status": status,
124 })
125 print()
126
127 # Final summary table
128 print_banner("E2E TEST SUITE EXECUTION SUMMARY")

131 print("-" * 88)
132 for row in summary_rows:
133 line = (

137 f"{row['passed']:<5} | "
138 f"{row['failed']:<5} | "
139 f"{row['time_s']:<8} | "
140 f"{row['status']}"
141 )
142 print(line)
143 print("-" * 88)
144 print(
145 f"TOTAL: {total_tests} tests executed in {total_time:.2f}s. "

150 print("=" * 80)
