66 def test_linear_symmetry_and_scaling(self):
67 """Displacements from origin must scale linearly with curvature coefficients."""
68 delta_deg = 0.01 # ~1.1 km
69 # Northing
70 _, y_pos = wgs84_to_metric(LAT0 + delta_deg, LON0)
71 _, y_neg = wgs84_to_metric(LAT0 - delta_deg, LON0)
72 self.assertAlmostEqual(y_pos, delta_deg * K_LAT, places=6)
73 self.assertAlmostEqual(y_neg, -delta_deg * K_LAT, places=6)
74 self.assertAlmostEqual(y_pos, -y_neg, places=9)
75
76 # Easting
77 x_pos, _ = wgs84_to_metric(LAT0, LON0 + delta_deg)
78 x_neg, _ = wgs84_to_metric(LAT0, LON0 - delta_deg)
79 self.assertAlmostEqual(x_pos, delta_deg * K_LON, places=6)
80 self.assertAlmostEqual(x_neg, -delta_deg * K_LON, places=6)
81 self.assertAlmostEqual(x_pos, -x_neg, places=9)

83 def test_haversine_properties(self):
84 """Haversine distance must satisfy identity, symmetry, and triangle inequality."""
85 p1 = (40.80293791, 14.22662080)
86 p2 = (40.80029953, 14.23084192)
87 p3 = (40.81800615, 14.24476797)
88
89 # Identity
90 self.assertAlmostEqual(haversine_distance(p1[0], p1[1], p1[0], p1[1]), 0.0, places=9)
91
92 # Symmetry
93 d12 = haversine_distance(p1[0], p1[1], p2[0], p2[1])
94 d21 = haversine_distance(p2[0], p2[1], p1[0], p1[1])
95 self.assertAlmostEqual(d12, d21, places=9)
96
97 # Triangle inequality: d(1, 3) <= d(1, 2) + d(2, 3)
98 d23 = haversine_distance(p2[0], p2[1], p3[0], p3[1])
99 d13 = haversine_distance(p1[0], p1[1], p3[0], p3[1])
100 self.assertLessEqual(d13, d12 + d23 + 1e-6)

102 def test_bottom_gate_projection_accuracy(self):
103 """Metric distance between RM8 and RM6 must match Haversine distance with <0.1% error."""
104 rm8 = MARKS["RM8"]
105 rm6 = MARKS["RM6"]
106
107 x8, y8 = wgs84_to_metric(rm8.target_lat, rm8.target_lon)
108 x6, y6 = wgs84_to_metric(rm6.target_lat, rm6.target_lon)
109 d_cart = euclidean_distance(x8, y8, x6, y6)
110 d_hav = haversine_distance(rm8.target_lat, rm8.target_lon, rm6.target_lat, rm6.target_lon)
111
112 error_pct = abs(d_cart - d_hav) / d_hav * 100.0
113
114 # Must be strictly under 0.1% (nominal ~0.028%)
115 self.assertLess(
116 error_pct,
117 0.1,
118 f"Bottom gate error {error_pct:.4f}% exceeds 0.1% limit",
119 )
120 self.assertAlmostEqual(error_pct, 0.028, delta=0.01)
