151 def test_ingestion_and_refresh_share_exact_raw_policy_and_hash(self):
152 from pipeline.config import OFFICIAL_WIND_ASSET_ID
153 from pipeline.ingest import IngestionPipeline
154 from scripts.refresh_team_analytics import read_source_wind
155 session = {"session": {"race_start": GUN, "window_start": GUN - 130_000,
156 "window_end": GUN + 40_000, "race_course": {"course_axis": 0, "course_length": 1000}}}
157 raw_records = [session]
158 for sample in samples():
159 field = "true_wind_direction" if "twd_deg" in sample else "true_wind_speed"
160 value = sample.get("twd_deg", sample.get("tws_kt"))
161 if field == "true_wind_speed":
162 from pipeline.config import MPS_TO_KNOTS
163 value /= MPS_TO_KNOTS
164 raw_records.append({"telemetry": {"asset_id": OFFICIAL_WIND_ASSET_ID,
165 "timestamp": sample["timestamp_ms"], field: {"value": value}}})
166 raw = ("\r\n".join(json.dumps(record) for record in raw_records) + "\r\n").encode("utf-8")
167 with tempfile.TemporaryDirectory() as directory:
168 source = Path(directory) / "test.jsonl"
169 source.write_bytes(raw)
170 ingestion = IngestionPipeline(str(source)).process()
171 observations, end, provenance = read_source_wind(source, Path(directory))
172 self.assertEqual(ingestion.official_wind.source_provenance["source_sha256"], provenance["source_sha256"])
173 self.assertEqual(ingestion.official_wind.source_provenance["source_size_bytes"], len(raw))
174 self.assertEqual(analyze(ingestion.official_wind.causal_samples)["snapshots"], analyze(observations)["snapshots"])
175 self.assertEqual(end, GUN + 40_000)
