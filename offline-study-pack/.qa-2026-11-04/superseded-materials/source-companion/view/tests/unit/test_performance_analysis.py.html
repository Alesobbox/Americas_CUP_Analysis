13def metric(value, coverage=100, quality="derived"):

17def boat(code, speed, retention=.8):
18 legs, events = [], []
19 for number in range(1, 5):
20 start = 1_000_000 + (number - 1) * 100_000
21 legs.append({"leg": number, "complete": True, "start_ms": start, "end_ms": start + 100_000,
22 "metrics": {"sog_mean_kt": metric(speed), "course_vmg_kt": metric(speed / 2),
23 "elapsed_s": metric(100, quality="reconstructed")}})
24 events.append({"type": "TACK" if number % 2 else "GYBE", "leg": number,
25 "timestamp": start + 50_000, "quality": "reconstructed",
26 "entry_speed_kt": 20, "min_speed_kt": 20 * retention,
27 "speed_loss_kt": 20 * (1 - retention)})
28 return {"id": code, "classification": "RANK_1", "source_conflicts": [],
29 "start": {"crossing_time_ms": 1_000_000, "ocs": False}, "legs": legs, "manoeuvres": events,
30 "source_coverage": {field: {"first_ms": 900_000, "last_ms": 1_500_000, "gaps": []}
31 for field in ("sog", "heading", "twd")}}
