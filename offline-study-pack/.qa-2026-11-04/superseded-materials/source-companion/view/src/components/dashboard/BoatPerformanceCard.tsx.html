82function MiniSparkline({
83 points,
84 color,
85 width = 36,
86 height = 14,
87}: {
88 points: number[]
89 color: string
90 width?: number
91 height?: number
92}) {
93 const pathD = useMemo(() => {
94 if (points.length < 2) return null
95 const min = Math.min(...points)
96 const max = Math.max(...points)
97 const range = max - min || 1
98 const coords = points.map((val, idx) => {
99 const x = (idx / (points.length - 1)) * width
100 const y = height - ((val - min) / range) * (height - 4) - 2
101 return `${x.toFixed(1)},${y.toFixed(1)}`
102 })
103 return `M ${coords.join(' L ')}`
104 }, [points, width, height])
105
106 if (!pathD) return null
107 return (
108 <svg width={width} height={height} viewBox={`0 0 ${width} ${height}`} className="mini-sparkline-svg">
109 <path d={pathD} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
110 </svg>
111 )
112}
