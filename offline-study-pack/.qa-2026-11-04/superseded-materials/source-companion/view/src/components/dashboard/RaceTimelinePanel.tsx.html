12export function eventColor(type: string): string {
13 const normalized = eventType(type)
14 if (normalized === 'gun' || normalized === 'start') return '#a855f7'

17 if (normalized === 'wind_shift') return '#38bdf8'

57function formatSignedCountdown(deltaMs: number): string {
58 const sign = deltaMs >= 0 ? '+' : '−'
59 const absSec = Math.floor(Math.abs(deltaMs) / 1000)
60 const mm = Math.floor(absSec / 60)
61 const ss = absSec % 60
62 return `${sign}${String(mm).padStart(2, '0')}:${String(ss).padStart(2, '0')}`
63}
