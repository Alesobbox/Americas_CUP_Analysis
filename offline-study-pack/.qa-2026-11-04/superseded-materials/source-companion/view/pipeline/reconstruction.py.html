103def compute_bow_position(
104 x_gps: float,
105 y_gps: float,
106 heading_deg: float,
107 bow_offset_m: float = AC40_BOW_OFFSET_M,
108) -> Tuple[float, float]:
109 """Computes the foremost bow coordinates in local metric metres."""
110 hdg_rad = math.radians(heading_deg)
111 # Heading: 0 deg = North (+Y), 90 deg = East (+X)
112 x_bow = x_gps + bow_offset_m * math.sin(hdg_rad)
113 y_bow = y_gps + bow_offset_m * math.cos(hdg_rad)
114 return x_bow, y_bow

117def intersect_segments_2d(
118 p_prev: Tuple[float, float],
119 p_curr: Tuple[float, float],
120 m_port: Tuple[float, float],
121 m_starboard: Tuple[float, float],
122 margin_epsilon: float = 0.20,
123) -> Tuple[bool, float, float, float, float]:
124 """Computes 2D trajectory segment intersection against a gate line segment.
125
126 Trajectory: B(u) = p_prev + u * (p_curr - p_prev), u in [0, 1]
127 Gate: G(s) = m_port + s * (m_starboard - m_port), s in [-margin, 1 + margin]
128
129 Returns:
130 (is_intersecting, u, s, x_cross, y_cross)
131 """
132 dx_p = p_curr[0] - p_prev[0]
133 dy_p = p_curr[1] - p_prev[1]
134 dx_m = m_starboard[0] - m_port[0]
135 dy_m = m_starboard[1] - m_port[1]
136
137 denom = dx_p * dy_m - dy_p * dx_m
138 if abs(denom) < 1e-9:
139 return False, 0.0, 0.0, 0.0, 0.0
140
141 orig_dx = m_port[0] - p_prev[0]
142 orig_dy = m_port[1] - p_prev[1]
143
144 u = (orig_dx * dy_m - orig_dy * dx_m) / denom
145 s = (orig_dx * dy_p - orig_dy * dx_p) / denom
146
147 is_intersecting = (0.0 <= u <= 1.0) and (-margin_epsilon <= s <= 1.0 + margin_epsilon)
148 x_cross = p_prev[0] + u * dx_p
149 y_cross = p_prev[1] + u * dy_p
150
151 return is_intersecting, u, s, x_cross, y_cross

154def compute_distance_to_start_line(
155 bow_pos: Tuple[float, float],
156 m_port: Tuple[float, float],
157 m_starboard: Tuple[float, float],
158 course_axis_deg: float = COURSE_AXIS_SETTLED_DEG,
159) -> float:
160 """Computes distance of bow to start line. Positive = on course side (OCS), negative = behind line."""
161 # Line vector from starboard (RM6) to port (RM8)
162 dx = m_port[0] - m_starboard[0]
163 dy = m_port[1] - m_starboard[1]
164 length = math.hypot(dx, dy)
165 if length < 1e-6:
166 return 0.0
167
168 # Unit normal pointing upwind (into course): rotated 90 deg clockwise from (dx, dy) or upwind normal
169 # For Naples: RM6 is SE, RM8 is NW. Vector RM6 -> RM8 has dx < 0, dy > 0.
170 # Normal pointing upwind into course (NE): nx = -dy / len, ny = dx / len
171 # Check course axis: (sin 44.39, cos 44.39) > 0.
172 nx = -dy / length
173 ny = dx / length
174 # Ensure normal points generally upwind along course axis (dot product with upwind > 0)
175 axis_rad = math.radians(course_axis_deg)
176 u_axis = (math.sin(axis_rad), math.cos(axis_rad))
177 if nx * u_axis[0] + ny * u_axis[1] < 0:
178 nx, ny = -nx, -ny
179
180 # Project vector from starboard mark to bow onto normal
181 rel_x = bow_pos[0] - m_starboard[0]
182 rel_y = bow_pos[1] - m_starboard[1]
183 return rel_x * nx + rel_y * ny

255 def reconstruct_geometric_crossings_for_boat(
256 self,
257 boat_x: List[float],
258 boat_y: List[float],
259 boat_heading: List[float],
260 timestamps: List[int],
261 marks: Dict[str, Any],
262 ) -> List[GateCrossing]:
263 """Detects gate crossings for a boat using trajectory segment intersection."""
264 n_frames = len(timestamps)
265 if n_frames < 2:
266 return []
267
268 # Build gate sequence dynamically from session sequences or default 4-leg sequence
269 gates_seq: List[Tuple[int, str, str, bool, str]] = []
270 if self.course_sequences:
271 total_legs = len(self.course_sequences) - 1
272 for leg_idx, seq in enumerate(self.course_sequences):
273 gate = seq.get("gate", {})
274 p_id = gate.get("port_asset_id")
275 s_id = gate.get("starboard_asset_id")
276 p_code = UUID_TO_MARK.get(p_id)
277 s_code = UUID_TO_MARK.get(s_id)
278 if not p_code:
279 for c, m in marks.items():
280 if getattr(m, "asset_id", None) == p_id:
281 p_code = c
282 break
283 if not p_code:
284 p_code = "RM4" if leg_idx % 2 == 1 else "RM8"
285 if not s_code:
286 for c, m in marks.items():
287 if getattr(m, "asset_id", None) == s_id:
288 s_code = c
289 break
290 if not s_code:
291 s_code = "RM2" if leg_idx % 2 == 1 else "RM6"
292 is_upwind = (leg_idx == 0) or (leg_idx % 2 == 1)
293 leg_role = "START" if leg_idx == 0 else ("FINISH" if leg_idx == total_legs else ("TOP" if leg_idx % 2 == 1 else "BOTTOM"))
294 gates_seq.append((leg_idx, p_code, s_code, is_upwind, leg_role))
295 else:
296 gates_seq = [
297 (0, "RM8", "RM6", True, "START"),
298 (1, "RM4", "RM2", True, "TOP"),
299 (2, "RM8", "RM6", False, "BOTTOM"),
300 (3, "RM4", "RM2", True, "TOP"),
301 (4, "RM8", "RM6", False, "FINISH"),
302 ]
303
304 crossings: List[GateCrossing] = []
305 next_leg_idx = 0
306
307 # Precompute bow positions
308 bow_x: List[float] = [0.0] * n_frames
309 bow_y: List[float] = [0.0] * n_frames
310 for i in range(n_frames):
311 bx, by = compute_bow_position(boat_x[i], boat_y[i], boat_heading[i])
312 bow_x[i] = bx
313 bow_y[i] = by
314
315 # Minimum time between consecutive gate crossings (e.g. 60 seconds)
316 min_gate_delta_ms = 60000
317
318 for i in range(n_frames - 1):
319 if next_leg_idx >= len(gates_seq):
320 break
321
322 leg_num, port_name, stbd_name, is_upwind, leg_role = gates_seq[next_leg_idx]
323
324 # Gate marks at frame i
325 m_port = marks.get(port_name)
326 m_stbd = marks.get(stbd_name)
327 if not m_port or not m_stbd:
328 continue
329 p_port = (m_port.x[i], m_port.y[i])
330 p_stbd = (m_stbd.x[i], m_stbd.y[i])
331
332 p_prev = (bow_x[i], bow_y[i])
333 p_curr = (bow_x[i + 1], bow_y[i + 1])
334
335 # Trajectory advance vector
336 dx_traj = p_curr[0] - p_prev[0]
337 dy_traj = p_curr[1] - p_prev[1]
338 advance = dx_traj * self.u_course[0] + dy_traj * self.u_course[1]
339
340 # Verify direction
341 if is_upwind and advance <= 0:
342 continue
343 if not is_upwind and advance >= 0:
344 continue
345
346 # Check start window: Leg 0 must be near gun time
347 if leg_num == 0 and timestamps[i] < self.gun_time_ms - 30000:
348 continue
349
350 crossed, u, s, cx, cy = intersect_segments_2d(p_prev, p_curr, p_port, p_stbd)
351 if crossed:
352 # Check minimum time since last crossing
353 t_cross = int(round(timestamps[i] + u * (timestamps[i + 1] - timestamps[i])))
354 if crossings and (t_cross - crossings[-1].crossing_time < min_gate_delta_ms):
355 continue
356
357 chosen_side = "PORT" if s < 0.50 else "STARBOARD"
358 chosen_mark = port_name if s < 0.50 else stbd_name
359
360 d_port = math.hypot(cx - p_port[0], cy - p_port[1])
361 d_stbd = math.hypot(cx - p_stbd[0], cy - p_stbd[1])
362
363 crossing = GateCrossing(
364 leg=leg_num,
365 crossing_time=t_cross,
366 mark_chosen=chosen_mark,
367 side=chosen_side,
368 gate_param_s=s,
369 dist_to_port_m=d_port,
370 dist_to_starboard_m=d_stbd,
371 is_official=False,
372 )
373 crossings.append(crossing)
374 next_leg_idx += 1
375
376 return crossings

599 "from_gate": "BOTTOM" if leg % 2 else "TOP",
