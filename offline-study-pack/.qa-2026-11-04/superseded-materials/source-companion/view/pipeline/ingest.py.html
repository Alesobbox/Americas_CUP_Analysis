54def interpolate_linear(
55 t: float,
56 t0: float,
57 t1: float,
58 v0: float,
59 v1: float,
60) -> float:
61 """Computes linear interpolation between two points."""
62 if t0 == t1:
63 return v0
64 u = (t - t0) / (t1 - t0)
65 return v0 + u * (v1 - v0)

68def interpolate_circular_deg(
69 t: float,
70 t0: float,
71 t1: float,
72 deg0: float,
73 deg1: float,
74) -> float:
75 """Computes shortest-arc circular interpolation between angles in degrees [0, 360)."""
76 if t0 == t1:
77 return deg0 % 360.0
78 u = (t - t0) / (t1 - t0)
79 delta = ((deg1 - deg0 + 180.0) % 360.0) - 180.0
80 return (deg0 + u * delta) % 360.0

83def compute_twa(twd_deg: float, heading_deg: float) -> float:
84 """Computes True Wind Angle (TWA) in range [-180.0, +180.0] degrees.
85
86 Positive = Starboard tack (wind coming from starboard side)
87 Negative = Port tack (wind coming from port side)
88 """
89 return ((twd_deg - heading_deg + 180.0) % 360.0) - 180.0

92def compute_vmg(sog_kt: float, cog_deg: float, target_axis_deg: float = COURSE_AXIS_DEG) -> float:
93 """Computes Velocity Made Good (VMG) in knots along a target course axis."""
94 angle_diff_rad = math.radians(cog_deg - target_axis_deg)
95 return sog_kt * math.cos(angle_diff_rad)

98def clean_and_sort_series(raw_series: List[Tuple[int, float]]) -> Tuple[List[int], List[float]]:
99 """Sorts timestamped series and deduplicates adjacent identical timestamps."""
100 if not raw_series:
101 return [], []
102 raw_series.sort(key=lambda item: item[0])
103 times: List[int] = []
104 values: List[float] = []
105 for t, v in raw_series:
106 if not times or t > times[-1]:
107 times.append(t)
108 values.append(v)
109 else:
110 # Overwrite duplicate timestamp with latest incoming value
111 values[-1] = v
112 return times, values

136def resample_scalar_series(
137 times: List[int],
138 values: List[float],
139 grid_timestamps: List[int],
140) -> List[float]:
141 """Resamples a scalar time series onto a regular timestamp grid using linear interpolation.
142
143 Performs an efficient single-pass two-pointer scan: O(N + M) complexity.
144 """
145 n_grid = len(grid_timestamps)
146 if not times:
147 return [0.0] * n_grid
148 if len(times) == 1:
149 return [values[0]] * n_grid
150
151 resampled: List[float] = [0.0] * n_grid
152 n_raw = len(times)
153 raw_idx = 0
154
155 for i, t in enumerate(grid_timestamps):
156 if t <= times[0]:
157 resampled[i] = values[0]
158 continue
159 if t >= times[-1]:
160 resampled[i] = values[-1]
161 continue
162
163 # Advance pointer to bracket [times[raw_idx], times[raw_idx + 1]]
164 while raw_idx + 1 < n_raw and times[raw_idx + 1] <= t:
165 raw_idx += 1
166
167 t0 = times[raw_idx]
168 t1 = times[raw_idx + 1]
169 v0 = values[raw_idx]
170 v1 = values[raw_idx + 1]
171
172 if t0 == t1:
173 resampled[i] = v0
174 else:
175 u = (t - t0) / (t1 - t0)
176 resampled[i] = v0 + u * (v1 - v0)
177
178 return resampled

191 if not times:

194 return [values[0] % 360.0] * n_grid

202 resampled[i] = values[0] % 360.0
203 continue
204 if t >= times[-1]:
205 resampled[i] = values[-1] % 360.0
206 continue

209 raw_idx += 1

536 stream.cog.append((t_ms, float(tel["course_over_ground"]["value"])))
537 elif "true_wind_speed" in tel:
538 stream.tws.append((t_ms, float(tel["true_wind_speed"]["value"])))
539 elif "true_wind_direction" in tel:
540 stream.twd.append((t_ms, float(tel["true_wind_direction"]["value"])))
541
542 # Route gate mark telemetry
543 elif asset_id in self.mark_uuids:
544 if "gnss_position" in tel:
545 pos = tel["gnss_position"]
546 mark_streams.setdefault(asset_id, []).append(

549

551 elif asset_id == self.official_wind_id:
552 if "true_wind_speed" in tel:

554 elif "true_wind_direction" in tel:

567 session_lines += 1

570 if raw_session:
571 # Dynamically discover active marks from session assets & sequences
572 for asset in raw_session.get("assets", []):

577 self.mark_uuids.add(aid)
578 self.tracked_uuids.add(aid)
579 mark_streams.setdefault(aid, [])
580 for seq in raw_session.get("race_course", {}).get("sequences", []):
581 gate = seq.get("gate", {})
582 pid = gate.get("port_asset_id")
583 sid = gate.get("starboard_asset_id")
584 if pid:
585 self.mark_uuids.add(pid)
586 self.tracked_uuids.add(pid)
587 mark_streams.setdefault(pid, [])
588 if sid:
589 self.mark_uuids.add(sid)
590 self.tracked_uuids.add(sid)
591 mark_streams.setdefault(sid, [])
592
593 race_points = raw_session.get("race_boundary", {}).get("polygon", {}).get("coordinates", [])
594 safety_points = raw_session.get("safety_boundary", {}).get("polygon", {}).get("coordinates", [])
595 if race_points and safety_points:
596 race_metric = [(wgs84_to_metric(float(p.get("y", p.get("latitude", 0))), float(p.get("x", p.get("longitude", 0))), lat0=self.lat0, lon0=self.lon0, k_lat=self.k_lat, k_lon=self.k_lon)) for p in race_points]
597 safety_metric = [(wgs84_to_metric(float(p.get("y", p.get("latitude", 0))), float(p.get("x", p.get("longitude", 0))), lat0=self.lat0, lon0=self.lon0, k_lat=self.k_lat, k_lon=self.k_lon)) for p in safety_points]
598 if not boundary_timeline or race_metric != boundary_timeline[-1]["race"] or safety_metric != boundary_timeline[-1]["safety"]:
599 boundary_timeline.append({"t": int(raw_session.get("timestamp", raw_session.get("window_start", 0))), "race": race_metric, "safety": safety_metric})

601 parse_time = time.perf_counter() - start_time
602
603 # 3. Validate and extract session metadata
604 if not raw_session:
605 raise ValueError("No session message found in telemetry file.")
606
607 race_start_ms = int(raw_session["race_start"])
608 window_start_ms = int(raw_session["window_start"])
609 window_end_ms = int(raw_session["window_end"])
610 course_axis = float(raw_session["race_course"]["course_axis"])
611 course_len = float(raw_session["race_course"]["course_length"])
612 rules = raw_session.get("rules", {})
613 course_seqs = raw_session.get("race_course", {}).get("sequences", [])
614 asset_targets = raw_session.get("asset_targets", [])
615
616 # Boundaries
617 race_boundary_pts = raw_session.get("race_boundary", {}).get("polygon", {}).get("coordinates", [])
618 safety_boundary_pts = raw_session.get("safety_boundary", {}).get("polygon", {}).get("coordinates", [])
619
620 def _proj_poly(coords):
621 pts = []
622 for p in coords:
623 if isinstance(p, dict):
624 lat = float(p.get("y", p.get("latitude", 0)))
625 lon = float(p.get("x", p.get("longitude", 0)))
626 elif isinstance(p, (tuple, list)):
627 lat, lon = float(p[0]), float(p[1])
628 else:
629 continue
630 pts.append(wgs84_to_metric(lat, lon, lat0=self.lat0, lon0=self.lon0, k_lat=self.k_lat, k_lon=self.k_lon))
631 return pts
632
633 race_boundary_metric = boundary_timeline[0]["race"] if boundary_timeline else _proj_poly(race_boundary_pts)
634 safety_boundary_metric = boundary_timeline[0]["safety"] if boundary_timeline else _proj_poly(safety_boundary_pts)
635
636 # Extract Platform Border & Keep Clear Border
637 platform_coords = (raw_session.get("platform_border") or {}).get("coordinates", [])
638 if platform_coords:
639 platform_border = [{"x": float(p["x"]), "y": float(p["y"])} for p in platform_coords]
640 else:
641 platform_border = list(DEFAULT_PLATFORM_BORDER_COORDS)
642
643 keep_clear_coords = (raw_session.get("keep_clear_border") or {}).get("coordinates", [])
644 if keep_clear_coords:
645 keep_clear_border = [{"x": float(p["x"]), "y": float(p["y"])} for p in keep_clear_coords]
646 else:
647 keep_clear_border = list(DEFAULT_KEEP_CLEAR_BORDER_COORDS)
648
649 is_cagliari = "cag" in self.file_path.lower() or "cagliari" in self.file_path.lower() or (
650 raw_session.get("name") and "cag" in raw_session.get("name", "").lower()
651 )
652 event_edition = "37th America's Cup Preliminary Regatta Cagliari 2026" if is_cagliari else "37th America's Cup Preliminary Regatta Naples 2026"
653
654 session_meta = SessionMetadata(
655 id=raw_session.get("id", ""),
656 name=raw_session.get("name", ""),
657 race_state=raw_session.get("race_state", ""),
658 race_type=raw_session.get("race_type", ""),
659 race_start_ms=race_start_ms,
660 window_start_ms=window_start_ms,
661 window_end_ms=window_end_ms,
662 official_wind_asset_id=raw_session.get("official_wind_asset_id", self.official_wind_id),
663 course_axis_deg=course_axis,
664 course_length_m=course_len,
665 rules=rules,
666 race_boundary=race_boundary_metric,
667 safety_boundary=safety_boundary_metric,
668 course_sequences=course_seqs,
669 asset_targets=asset_targets,
670 boundary_timeline=boundary_timeline,
671 platform_border=platform_border,
672 keep_clear_border=keep_clear_border,
673 event_edition=event_edition,
674 boat_class="AC40",
675 class_evidence={
676 "description": "Explicit ASSET_TYPE_AC40 and 12.963m length telemetry envelope",
677 "confidence": "CONFIRMED",
678 "source": "session.assets and session.platform_border",
679 },
680 origin_lat=self.lat0,
681 origin_lon=self.lon0,
682 k_lat=self.k_lat,
683 k_lon=self.k_lon,
684 )
685
686 # 4. Generate regular 10 Hz time grid
687 grid_timestamps = list(
688 range(window_start_ms, window_end_ms + 1, self.sample_interval_ms)
689 )
690
691 # 5. Resample competitor boat streams
692 synchronized_boats: Dict[str, SynchronizedBoatData] = {}
693 stream_coverage: Dict[str, Any] = {}
694 session_assets = {asset["id"]: asset for asset in raw_session.get("assets", []) if asset.get("id")}
695 for uuid in COMPETITORS:
696 boat_info = session_competitor_identity(uuid, session_assets)
697 code = boat_info.code
698 streams = boat_streams[uuid]
699 if not streams.gnss and not streams.sog:
700 continue
701
702 # Preserve source timestamp coverage before resampling creates held or
703 # long-gap interpolated values. A >2 s gap is not a valid observation.
704 def coverage(records):
705 stamps = sorted({int(row[0]) for row in records})
706 return {"first_ms": stamps[0] if stamps else None,
707 "last_ms": stamps[-1] if stamps else None,
708 "count": len(stamps),
709 "gaps": [[a, b] for a, b in zip(stamps, stamps[1:]) if b - a > 2000]}
710
711 stream_coverage[code] = {field_name: coverage(getattr(streams, field_name))
712 for field_name in ("gnss", "heading", "sog", "cog", "twd", "tws")}
713
714 # (a) Position projection & resampling
715 raw_pos_metric: List[Tuple[int, float, float]] = []
716 for t, lat, lon in streams.gnss:
717 x, y = wgs84_to_metric(lat, lon, lat0=self.lat0, lon0=self.lon0, k_lat=self.k_lat, k_lon=self.k_lon)
718 raw_pos_metric.append((t, x, y))
719
720 t_pos, pos_xs, pos_ys = clean_and_sort_xy_series(raw_pos_metric)
721 res_x, res_y = resample_xy_series(t_pos, pos_xs, pos_ys, grid_timestamps)
722
723 # (b) Heading
724 t_hdg, val_hdg = clean_and_sort_series(streams.heading)
725 res_hdg = resample_circular_series(t_hdg, val_hdg, grid_timestamps)
726
727 # (c) Speed Over Ground (SI m/s -> knots)
728 sog_knots_raw = [(t, v * MPS_TO_KNOTS) for t, v in streams.sog]
729 t_sog, val_sog = clean_and_sort_series(sog_knots_raw)
730 res_sog = resample_scalar_series(t_sog, val_sog, grid_timestamps)
731
732 # (d) Course Over Ground
733 t_cog, val_cog = clean_and_sort_series(streams.cog)
734 res_cog = resample_circular_series(t_cog, val_cog, grid_timestamps)
735
736 # (e) True Wind Speed (SI m/s -> knots)
737 tws_knots_raw = [(t, v * MPS_TO_KNOTS) for t, v in streams.tws]
738 t_tws, val_tws = clean_and_sort_series(tws_knots_raw)
739 res_tws = resample_scalar_series(t_tws, val_tws, grid_timestamps)
740
741 # (f) True Wind Direction
742 t_twd, val_twd = clean_and_sort_series(streams.twd)
743 res_twd = resample_circular_series(t_twd, val_twd, grid_timestamps)
744
745 # (g) Derived tactical metrics: TWA and VMG
746 res_twa = [compute_twa(twd, hdg) for twd, hdg in zip(res_twd, res_hdg)]
747 res_vmg = [compute_vmg(sog, cog, course_axis) for sog, cog in zip(res_sog, res_cog)]
748
749 synchronized_boats[code] = SynchronizedBoatData(
750 asset_id=uuid,
751 code=code,
752 name=boat_info.name,
753 official_name=boat_info.official_name,
754 color=boat_info.color,
755 x=res_x,
756 y=res_y,
757 heading=res_hdg,
758 cog=res_cog,
759 sog_kt=res_sog,
760 vmg_kt=res_vmg,
761 twa_deg=res_twa,
762 twd_deg=res_twd,
763 tws_kt=res_tws,
764 )
765
766 # 6. Resample Gate Marks (RM1 through RM8 and dynamic marks)
767 synchronized_marks: Dict[str, SynchronizedMarkData] = {}
768 session_marks_by_code: Dict[str, Dict[str, Any]] = {}
769 targets_by_id = {
770 t.get("asset_id"): (float(t["location"]["y"]), float(t["location"]["x"]))
771 for t in raw_session.get("asset_targets", [])
772 if isinstance(t.get("location"), dict) and t["location"].get("x") is not None and t["location"].get("y") is not None
773 }
774
775 for asset in raw_session.get("assets", []):
776 abbr = asset.get("abbreviation", "")
777 aid = asset.get("id")
778 if aid and (asset.get("type") == "ASSET_TYPE_RACE_MARK" or (abbr and abbr.startswith("RM"))):
779 code = abbr if (abbr and abbr.startswith("RM")) else UUID_TO_MARK.get(aid, abbr)
780 if code:
781 t_lat, t_lon = targets_by_id.get(aid, (0.0, 0.0))
782 session_marks_by_code[code] = {
783 "code": code,
784 "uuid": aid,
785 "name": asset.get("name", f"Race Mark {code}"),
786 "role": asset.get("official_name") or f"Mark {code}",
787 "color": asset.get("colour_code") or "#ff0000",
788 "target_lat": t_lat,
789 "target_lon": t_lon,
790 }
791
792 all_mark_codes = set(MARKS.keys()) | set(session_marks_by_code.keys())
793 for mark_code in sorted(all_mark_codes):
794 mark_info = MARKS.get(mark_code)
795 sess_info = session_marks_by_code.get(mark_code, {})
796 uuid = sess_info.get("uuid") or (mark_info.uuid if mark_info else None)
797 if not uuid:
798 continue
799 name = sess_info.get("name") or (mark_info.name if mark_info else f"Race Mark {mark_code}")
800 role = sess_info.get("role") or (mark_info.role if mark_info else f"Mark {mark_code}")
801 color = sess_info.get("color") or (mark_info.color if mark_info else "#ff0000")
802 t_lat = sess_info.get("target_lat") or (mark_info.target_lat if mark_info else 0.0)
803 t_lon = sess_info.get("target_lon") or (mark_info.target_lon if mark_info else 0.0)
804
805 target_x, target_y = wgs84_to_metric(t_lat, t_lon, lat0=self.lat0, lon0=self.lon0, k_lat=self.k_lat, k_lon=self.k_lon)
806
807 # Convert GNSS stream to metric
808 raw_mark_metric: List[Tuple[int, float, float]] = []
809 for t, lat, lon in mark_streams.get(uuid, []):
810 mx, my = wgs84_to_metric(lat, lon, lat0=self.lat0, lon0=self.lon0, k_lat=self.k_lat, k_lon=self.k_lon)
811 raw_mark_metric.append((t, mx, my))
812
813 if raw_mark_metric:
814 t_m, m_xs, m_ys = clean_and_sort_xy_series(raw_mark_metric)
815 res_mx, res_my = resample_xy_series(t_m, m_xs, m_ys, grid_timestamps)
816 if not sess_info.get("target_lat") and not (mark_info and mark_info.target_lat):
817 target_x = res_mx[0]
818 target_y = res_my[0]
819 else:
820 # Fallback to static target coordinate if no telemetry points
821 res_mx = [target_x] * len(grid_timestamps)
822 res_my = [target_y] * len(grid_timestamps)
823
824 synchronized_marks[mark_code] = SynchronizedMarkData(
825 mark_code=mark_code,
826 asset_id=uuid,
827 name=name,
828 role=role,
829 color=color,
830 target_x=target_x,
831 target_y=target_y,
832 x=res_mx,
833 y=res_my,
834 )
835
836 # 7. Resample Official Wind Telemetry
837 wind_tws_knots = [(t, v * MPS_TO_KNOTS) for t, v in official_wind_tws]
838 t_wtws, val_wtws = clean_and_sort_series(wind_tws_knots)
839 t_wtwd, val_wtwd = clean_and_sort_series(official_wind_twd)
840
841 res_wind_speed = resample_scalar_series(t_wtws, val_wtws, grid_timestamps)
842 res_wind_dir = resample_circular_series(t_wtwd, val_wtwd, grid_timestamps)
843
844 # Raw timeline for discrete wind inspection
845 raw_wind_timeline: List[Dict[str, Any]] = []
846 # Build paired timeline from matching or closest timestamps
847 for t, spd in zip(t_wtws, val_wtws):
848 # Find closest direction
849 closest_dir = resample_circular_series(t_wtwd, val_wtwd, [t])[0] if t_wtwd else 0.0
850 raw_wind_timeline.append({
851 "t": t,
852 "tws_kt": round(spd, 2),
853 "twd_deg": round(closest_dir, 1),
854 })
855
856 official_wind_data = OfficialWindData(
857 asset_id=self.official_wind_id,
858 raw_timeline=raw_wind_timeline,
859 speed_kt=res_wind_speed,
860 direction_deg=res_wind_dir,
861 causal_samples=(
862 [{"timestamp_ms": t, "twd_deg": v} for t, v in official_wind_twd]
863 + [{"timestamp_ms": t, "tws_kt": v * MPS_TO_KNOTS} for t, v in official_wind_tws]
864 ),
865 source_provenance={
866 "source_file": os.path.relpath(os.path.abspath(self.file_path),
867 os.path.dirname(os.path.dirname(os.path.abspath(__file__)))).replace("\\", "/"),
868 "source_sha256": source_hasher.hexdigest(),
869 "source_size_bytes": file_size_bytes,
870 "official_wind_asset_id": self.official_wind_id,
871 },
872 )
873
874 total_runtime = time.perf_counter() - start_time
875
876 stats = {
877 "file_size_mb": round(file_size_bytes / (1024 * 1024), 2),
878 "total_lines": total_lines,
879 "telemetry_lines": telemetry_lines,
880 "competitor_lines": competitor_lines,
881 "session_lines": session_lines,
882 "total_runtime_s": round(total_runtime, 3),
883 "parse_runtime_s": round(parse_time, 3),
884 "resampled_grid_frames": len(grid_timestamps),
885 "sample_rate_hz": SAMPLE_RATE_HZ,
886 "sample_interval_ms": self.sample_interval_ms,
887 }
888
889 return IngestionResult(
890 session=session_meta,
891 competitor_events=competitor_events,
892 grid_timestamps_ms=grid_timestamps,
893 boats=synchronized_boats,
894 marks=synchronized_marks,
895 official_wind=official_wind_data,
896 stats=stats,
897 stream_coverage=stream_coverage,
898 )
