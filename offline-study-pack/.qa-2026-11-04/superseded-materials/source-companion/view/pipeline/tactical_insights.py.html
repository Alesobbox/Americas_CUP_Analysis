51def _number(value):
52 return isinstance(value, (int, float)) and not isinstance(value, bool) and math.isfinite(value)

55def source_wind_observations(direction_series, speed_series):
56 """Preserve raw, unpaired field times; no interpolation or held pairing."""
57 observations = []
58 for field, series in (("twd_deg", direction_series), ("tws_kt", speed_series)):
59 for timestamp, value in series:
60 observations.append({"timestamp_ms": timestamp, field: value})
61 return sorted(observations, key=lambda item: item["timestamp_ms"])

64def _streams(samples):
65 streams = {"twd_deg": {}, "tws_kt": {}}
66 for sample in samples:
67 timestamp = sample.get("timestamp_ms", sample.get("t"))
68 if not isinstance(timestamp, int) or isinstance(timestamp, bool):
69 continue
70 for field in streams:
71 value = sample.get(field)
72 if (_number(value) and ((0 <= value < 360) if field == "twd_deg" else value >= 0)):
73 streams[field][timestamp] = float(value)
74 return {field: (sorted(values), [values[t] for t in sorted(values)])
75 for field, values in streams.items()}

78def _window(times, values, timestamp):
79 """One-second trailing grid using previous observations <= each grid time."""
80 start = timestamp - WINDOW_MS
81 grid = []
82 for t in range(start, timestamp + 1, 1000):
83 index = bisect_right(times, t) - 1
84 grid.append(values[index] if index >= 0 and t - times[index] <= MAX_SOURCE_AGE_MS else None)
85 first = bisect_right(times, start) - 1
86 last = bisect_right(times, timestamp)
87 active_times = times[max(0, first):last]
88 count = len(times[bisect_right(times, start - 1):last])
89 gaps = [(a, b) for a, b in zip(active_times, active_times[1:]) if b - a > MAX_SOURCE_AGE_MS]
90 # Reject the actual unsupported interval, even when the coarse one-second
91 # grid happens to land on two apparently fresh endpoints around a gap.
92 covered = sum(a is not None and b is not None
93 and not any(left < start + (i + 1) * 1000 and right > start + i * 1000
94 for left, right in gaps)
95 for i, (a, b) in enumerate(zip(grid, grid[1:])))
96 complete = (all(value is not None for value in grid) and count >= 60
97 and not gaps)
98 return grid, count, covered, complete

101def _unwrap(values):
102 unwrapped = [values[0]]
103 for a, b in zip(values, values[1:]):
104 unwrapped.append(unwrapped[-1] + (b - a + 180) % 360 - 180)
105 return unwrapped

108def _slope(values):
109 center = (len(values) - 1) / 2
110 average = sum(values) / len(values)
111 denominator = sum((i - center) ** 2 for i in range(len(values)))
112 slope = sum((i - center) * (value - average) for i, value in enumerate(values)) / denominator
113 residual = math.sqrt(sum((value - average - slope * (i - center)) ** 2
114 for i, value in enumerate(values)) / len(values))
115 return slope * 60, residual

118def _projection(values, rate):
119 """Conservative conditional extrapolation, deliberately often unavailable."""
120 first_rate, _ = _slope(values[:61])
121 last_rate, _ = _slope(values[60:])
122 _, residual = _slope(values)
123 net = values[-1] - values[0]
124 if (abs(net) < 1 or abs(rate) < 0.2 or abs(rate * 5) > 30
125 or first_rate * last_rate <= 0
126 or max(abs(first_rate - rate), abs(last_rate - rate)) > max(0.25, abs(rate) * 0.5)
127 or residual > max(1.0, abs(net) * 0.2)
128 or max(abs(b - a) for a, b in zip(values, values[1:])) > 3):
129 return None, "Recent direction is too flat, variable or steep for a bounded linear projection."
130 return round(rate * 5, 2), "Conditional on the last 120 s trend holding for 5 min; unvalidated linear extrapolation, not a calibrated forecast."

133def attach_tactical_insights(detail, source_wind_samples, *, window_end_ms=None, source_provenance=None):
134 """Attach compact diagnostics. No sample later than a snapshot is consulted."""
135 streams = _streams(source_wind_samples)
136 direction_times, direction_values = streams["twd_deg"]
137 speed_times, speed_values = streams["tws_kt"]
138 gun = detail["race"]["gun_time_ms"]
139 end = window_end_ms if window_end_ms is not None else detail["race"].get("window_end_ms")
140 if end is None:
141 end = max([gun, *direction_times, *speed_times])
142 snapshots = []
143 for timestamp in range(gun, end + 1, SNAPSHOT_INTERVAL_MS):
144 values, count, coverage, complete = _window(direction_times, direction_values, timestamp)
145 speeds, speed_count, speed_coverage, speed_complete = _window(speed_times, speed_values, timestamp)
146 snapshot = {"timestamp_ms": timestamp, "window_start_ms": timestamp - WINDOW_MS,
147 "window_end_ms": timestamp, "coverage_s": coverage, "sample_count": count,
148 "quality": "derived" if complete else "unavailable", "source": SOURCE,
149 "speed_coverage_s": speed_coverage, "speed_sample_count": speed_count,
150 "speed_quality": "derived" if speed_complete else "unavailable",
151 "speed_reason": None if speed_complete else "Need 120 s of fresh official speed observations, at least 60 samples and no source gap over 2 s.",
152 "direction_change_deg": None, "direction_rate_deg_per_min": None,
153 "speed_change_kt": None, "projection_change_deg": None,
154 "projection_reason": "A complete recent wind window is unavailable.", "insights": []}
155 if not complete:
156 snapshot["reason"] = "Need 120 s of fresh official direction observations, at least 60 samples and no source gap over 2 s."
157 else:
158 unwrapped = _unwrap(values)
159 change = unwrapped[-1] - unwrapped[0]
160 rate, _ = _slope(unwrapped)
161 projection, projection_reason = _projection(unwrapped, rate)
162 snapshot.update(direction_change_deg=round(change, 2), direction_rate_deg_per_min=round(rate, 3),
163 projection_change_deg=projection, projection_reason=projection_reason)
164 snapshot["insights"].append({"id": "wind_direction", "title": f"Wind change {change:+.1f}° in 2 min",
165 "detail": "Recorded official wind; positive is clockwise (veering), negative is counterclockwise (backing).",
166 "quality": "derived", "source": SOURCE, "kind": "observed_direction"})
167 if projection is not None:
168 snapshot["insights"].append({"id": "wind_projection", "title": f"Conditional 5 min projection: {projection:+.1f}°",
169 "detail": projection_reason, "quality": "conditional", "source": SOURCE, "kind": "conditional_projection"})
170 if speed_complete:
171 speed_change = speeds[-1] - speeds[0]
172 snapshot["speed_change_kt"] = round(speed_change, 2)
173 snapshot["insights"].append({"id": "wind_speed", "title": f"Wind speed {speed_change:+.1f} kt in 2 min",
174 "detail": "Difference between fresh official speed observations at the ends of the trailing 120 s window.",
175 "quality": "derived", "source": SOURCE, "kind": "observed_speed"})
176 snapshots.append(snapshot)
177 provenance = dict(source_provenance or {})
178 provenance.update(direction_sample_count=len(direction_times), speed_sample_count=len(speed_times),
179 first_measurement_ms=min([*direction_times, *speed_times], default=None),
180 last_measurement_ms=max([*direction_times, *speed_times], default=None))
181 detail["tactical_insights"] = {"schema_version": SCHEMA_VERSION, "method_version": METHOD_VERSION,
182 "race_id": detail["race"]["id"], "source": SOURCE, "snapshot_interval_ms": SNAPSHOT_INTERVAL_MS,
183 "max_snapshot_age_ms": MAX_SNAPSHOT_AGE_MS, "window_s": 120, "projection_horizon_s": 300,
184 "definitions": TACTICAL_DEFINITIONS,
185 "availability_basis": "Measurement timestamps only; retrospective, not arrival-time causal. Late records with earlier timestamps can revise history.",
186 "source_provenance": provenance, "snapshots": snapshots}
187 if end < gun:
188 detail["tactical_insights"]["reason"] = "Replay window ends before the source gun; no post-gun tactical interval."
189 return detail
