79 def __post_init__(self) -> None:

126 def validate(self) -> None:
127 if self.round_robins not in (1, 2, 3) or self.fleet_races != 6:
128 raise ValueError("configure 1-3 complete round robins and six fleet races")
129 if len(self.fleet_points) != 7 or len(self.challenger_carry) != 6:
130 raise ValueError("fleet and Challenger carry schedules must cover every entrant")
131 if self.overall_carry is not None and len(self.overall_carry) != 7:
132 raise ValueError("overall carry schedule must cover seven entrants")
133 if self.tie_policy not in ("unresolved", "fixture_lots", "priority"):
134 raise ValueError("unknown tie policy")
135 if self.tie_policy == "priority" and (set(self.tie_priority) != set(TEAM_IDS)
136 or len(self.tie_priority) != 7
137 or not self.tie_rule_reference):
138 raise ValueError("priority ties require a complete external order and rule reference")
139 if self.series_race_safety_limit < 14:
140 raise ValueError("a series cannot be capped at the usual 13 scheduled races")
141 if len(self.known_fleet_orders) > self.fleet_races:
142 raise ValueError("too many known fleet results")
143 for order in self.known_fleet_orders:
144 if len(order) != 7 or set(order) != set(TEAM_IDS):
145 raise ValueError("known complete fleet order must include all seven teams once")
146 if self.withdrawn and not set(self.withdrawn) <= set(TEAM_IDS):
147 raise ValueError("withdrawal references an unknown team")
148 for selection_a, selection_b in self.official_selections.items():
149 if selection_a not in TEAM_IDS[1:] or selection_b not in TEAM_IDS[1:] or selection_a == selection_b:
150 raise ValueError("official semifinal selections require distinct Challengers")
151 values = [*self.fleet_points, *self.challenger_carry, *(self.overall_carry or ()),
152 *self.weather_sensitivity.values()]
153 if any(not math.isfinite(v) for v in values):
154 raise ValueError("rules contain nonfinite scores or weather sensitivities")

162def hazard_probability(probability: float, multiplier: float) -> float:
163 if not 0 <= probability <= 1 or not math.isfinite(multiplier) or multiplier <= 0:
164 raise ValueError("invalid full-race hazard input")
165 return 1.0 if probability == 1 else -math.expm1(multiplier * math.log1p(-probability))

185def simulate_cup(mapped_strength: Mapping[str, float], *, scenario: CupScenario | None = None,
186 rules: CupRules | None = None, trials: int = 100_000, seed: int = 2027,
187 pooled_scale: float = 0.5, failure_probability: float = 0.0,
188 conditional_on_completed_scored_races: bool = True) -> dict:
189 """Vectorized complete-tournament continuations with persistent team effects.
190
191 A zero reliability input means an explicit no-failure scenario, not a learned
192 estimate. If a sampled failure needs missing official timing, that trajectory
193 is unresolved. A provisional internal stable tie order only lets simulation
194 continue; its winner is discarded from exact estimates and bounds use [0,1]
195 on every unresolved trajectory. Therefore no invented tie weights enter the
196 reported estimates. These outer bounds may be deliberately uninformative.
197 """
198 scenario, rules = scenario or CupScenario(), rules or CupRules()
199 scenario.validate()
200 rules.validate()
201 if trials < 1 or not isinstance(trials, int) or pooled_scale <= 0 or not math.isfinite(pooled_scale):
202 raise ValueError("positive integer trials and positive finite pooled scale required")
203 if any(team not in TEAM_IDS or not math.isfinite(value) for team, value in mapped_strength.items()):
204 raise ValueError("mapped strengths require registered teams and finite values")
205 if conditional_on_completed_scored_races and failure_probability != 0:
206 raise ValueError("completed-scored-race conditioning cannot simultaneously sample failures")
207 q = hazard_probability(failure_probability, scenario.hazard_multiplier)
208 rng = np.random.default_rng(seed)
209 rows = np.arange(trials)
210 means = np.array([mapped_strength.get(team, 0.0) for team in TEAM_IDS])
211 missing = np.array([team not in mapped_strength for team in TEAM_IDS])
212 population = rng.normal(0, pooled_scale, (trials, 7))
213 base = np.where(missing[None, :], population, means[None, :])
214 # One draw per team per whole tournament, never independently per match.
215 theta = scenario.rho * base + rng.normal(0, scenario.sigma_multiplier * pooled_scale, (trials, 7))
216 slopes = np.array([rules.weather_sensitivity.get(team, 0.0) for team in TEAM_IDS])
217 day_factors: dict[int, np.ndarray] = {}
218 unresolved = np.zeros(trials, dtype=bool)
219 reasons: set[str] = set()
220 reason_masks: dict[str, np.ndarray] = {}
221 first_unresolved_stage = np.full(trials, "", dtype=object)
222
223 def mark_unresolved(mask, stage, reason):
224 selected_mask = np.broadcast_to(np.asarray(mask, dtype=bool), (trials,))
225 if not np.any(selected_mask):
226 return
227 first_unresolved_stage[selected_mask & ~unresolved] = stage
228 unresolved[:] |= selected_mask
229 reasons.add(reason)
230 if reason not in reason_masks:
231 reason_masks[reason] = np.zeros(trials, dtype=bool)
232 reason_masks[reason] |= selected_mask
233
234 if rules.withdrawn:
235 mark_unresolved(True, "withdrawal", "Withdrawal scoring/advancement instructions are not supplied.")
236 rr_wins = np.zeros((trials, 7))
237 rr_played = np.zeros((trials, 7), dtype=np.int16)
238 max_series_races = 0

240 def day_theta(day: int) -> np.ndarray:
241 if day not in day_factors:
242 day_factors[day] = rng.normal(size=trials)
243 return theta + day_factors[day][:, None] * slopes[None, :]

245 def standing(points: np.ndarray, ids: np.ndarray, stage: str,
246 fleet_history: list[np.ndarray] | None = None) -> np.ndarray:
247 nonlocal unresolved
248 sorted_values = np.sort(points, axis=1)
249 ties = np.any(np.diff(sorted_values, axis=1) == 0, axis=1)
250 if fleet_history:
251 # RRSAC 107: latest common scheduled finishing places, then earlier
252 # races. Complete rank-only races have a distinct latest placing.
253 order = np.lexsort((*[ranks[:, ids] for ranks in fleet_history], -points), axis=1)
254 return np.broadcast_to(ids, points.shape)[rows[:, None], order]
255 if rules.tie_policy == "unresolved":
256 mark_unresolved(ties, stage, f"{stage}: official tie resolution is unresolved; outer bounds retain the entire tied path.")
257 secondary = np.broadcast_to(ids, points.shape)
258 elif rules.tie_policy == "fixture_lots":
259 secondary = rng.random(points.shape)
260 else:
261 priority = {TEAM_IDS.index(team): rank for rank, team in enumerate(rules.tie_priority)}
262 secondary = np.broadcast_to(np.array([priority[int(i)] for i in ids]), points.shape)
263 return np.broadcast_to(ids, points.shape)[rows[:, None], np.lexsort((secondary, -points), axis=1)]

265 def draw_match(a: np.ndarray, b: np.ndarray, day: int, stage="round_robin") -> tuple[np.ndarray, np.ndarray]:
266 nonlocal unresolved
267 abilities = day_theta(day)
268 # Joint uniform-reference mixture applied to the actual match draw.
269 p = (1 - 0.0001) * _logistic(scenario.match_concentration *
270 (abilities[rows, a] - abilities[rows, b])) + 0.0001 * 0.5
271 win_a = rng.random(trials) < p
272 if q:
273 finish_a, finish_b = rng.random(trials) >= q, rng.random(trials) >= q
274 mark_unresolved(~finish_a | ~finish_b, stage,
275 "Sampled nonfinish requires official timing/status scoring unavailable to the rank-only kernel.")
276 return ((finish_a & (win_a | ~finish_b)).astype(float),
277 (finish_b & (~win_a | ~finish_a)).astype(float))
278 return win_a.astype(float), (~win_a).astype(float)

280 def apply_known(key: str, a: np.ndarray, b: np.ndarray, pa: np.ndarray,
281 pb: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
282 nonlocal unresolved
283 known = rules.known_match_scores.get(key)
284 if known is None:
285 return pa, pb
286 ia, ib = TEAM_IDS.index(known.team_a), TEAM_IDS.index(known.team_b)
287 forward, reverse = (a == ia) & (b == ib), (a == ib) & (b == ia)
288 matched = forward | reverse
289 mark_unresolved(~matched, key.split(":")[0], f"{key}: observed participants require additional fixed earlier advancement results.")
290 if not known.resolved:
291 mark_unresolved(True, key.split(":")[0], f"{key}: observed score is unresolved.")
292 return (np.where(forward, known.points_a, np.where(reverse, known.points_b, pa)),
293 np.where(forward, known.points_b, np.where(reverse, known.points_a, pb)))

296 fleet_history = []
297 for race in range(rules.fleet_races):
298 if race < len(rules.known_fleet_orders):
299 order = np.broadcast_to([TEAM_IDS.index(team) for team in rules.known_fleet_orders[race]], (trials, 7))

304 order = np.argsort(-utilities, axis=1)
305 if q:
306 any_failure = np.any(rng.random((trials, 7)) < q, axis=1)
307 mark_unresolved(any_failure, "fleet", "Fleet nonfinish needs official status/timing points beyond rank-only observations.")
308 fleet_points[rows[:, None], order] += np.array(rules.fleet_points)[None, :]
309 ranks = np.empty((trials, 7), dtype=np.int8)
310 ranks[rows[:, None], order] = np.arange(7)[None, :]
311 fleet_history.append(ranks)
312 challenger_ids = np.arange(1, 7)
313 fleet_challenger_order = standing(fleet_points[:, 1:], challenger_ids, "fleet_challengers", fleet_history)
314 overall_fleet_order = standing(fleet_points, np.arange(7), "fleet_overall", fleet_history)
315 challenger_points = np.zeros((trials, 7))
316 challenger_points[rows[:, None], fleet_challenger_order] = np.array(rules.challenger_carry)[None, :]
317 overall_points = np.zeros((trials, 7))
318 if rules.overall_carry is not None:
319 overall_points[rows[:, None], overall_fleet_order] = np.array(rules.overall_carry)[None, :]
320 rr_race = 0
321 for round_number in range(1, rules.round_robins + 1):
322 for a_idx in range(7):
323 for b_idx in range(a_idx + 1, 7):
324 a, b = np.full(trials, a_idx), np.full(trials, b_idx)
325 key = f"rr:{round_number}:{TEAM_IDS[a_idx]}:{TEAM_IDS[b_idx]}"
326 pa, pb = apply_known(key, a, b, *draw_match(a, b, 10 + rr_race // 7))
327 overall_points[:, a_idx] += pa
328 overall_points[:, b_idx] += pb
329 if a_idx != 0: # Defender results never enter Challenger standings/policy.
330 challenger_points[:, a_idx] += pa
331 challenger_points[:, b_idx] += pb
332 policy_pa, policy_pb = pa, pb
333 known = rules.known_match_scores.get(key)
334 if known is not None and known.win_credit_a is not None:
335 forward = known.team_a == TEAM_IDS[a_idx]
336 policy_pa = np.full(trials, known.win_credit_a if forward else known.win_credit_b)
337 policy_pb = np.full(trials, known.win_credit_b if forward else known.win_credit_a)
338 elif known is not None and not (0 <= known.points_a <= 1 and 0 <= known.points_b <= 1):
339 mark_unresolved(True, "round_robin", f"{key}: penalized points need separate observed match win credits for opponent selection.")
340 policy_pa, policy_pb = np.zeros(trials), np.zeros(trials)
341 rr_wins[:, a_idx] += policy_pa
342 rr_wins[:, b_idx] += policy_pb
343 rr_played[:, a_idx] += 1
344 rr_played[:, b_idx] += 1
345 rr_race += 1
346 seeds = standing(challenger_points[:, 1:], challenger_ids, "round_robin_challengers")

348 def series(a: np.ndarray, b: np.ndarray, target: int, stage: str,
349 initial_a: float = 0, initial_b: float = 0, day: int = 100) -> np.ndarray:
350 nonlocal unresolved, max_series_races
351 points_a, points_b = np.full(trials, initial_a, dtype=float), np.full(trials, initial_b, dtype=float)
352 race_number = 0

354 active = (points_a < target) & (points_b < target)
355 race_number += 1
356 if race_number > rules.series_race_safety_limit:
357 mark_unresolved(active, stage, f"{stage}: continuation safety budget exhausted; no scheduled-race cap was substituted.")
358 break
359 pa, pb = apply_known(f"{stage}:{race_number}", a, b,
360 *draw_match(a, b, day + (race_number - 1) // 3, stage))
361 points_a += np.where(active, pa, 0)
362 points_b += np.where(active, pb, 0)
363 max_series_races = max(max_series_races, race_number)
364 mark_unresolved((points_a >= target) & (points_b >= target), stage,
365 f"{stage}: simultaneous threshold requires official resolution.")
366 return np.where(points_a >= target, a, b)

453def official_rules_gaps(rules: CupRules | None = None) -> list[str]:
454 """Publication checks; real digests/dates cannot be replaced by URL strings."""
455 rules = rules or CupRules()
456 gaps = []
457 for source in RULE_SOURCES:
458 version = source["version"]
459 digest = rules.document_hashes.get(version, "")
460 if len(digest) != 64 or any(char not in "0123456789abcdef" for char in digest):
461 gaps.append(f"Missing verified document SHA-256: {version}")
462 if not rules.document_effective_dates.get(version):
463 gaps.append(f"Missing verified effective date: {version}")
464 gaps.extend(clause for clause in UNRESOLVED_OFFICIAL_CLAUSES if clause not in rules.resolved_clauses)
465 if rules.tie_policy == "fixture_lots":
466 gaps.append("Synthetic fixture lottery rules cannot be published as official AC38 rules.")
467 if rules.overall_carry is None:
468 gaps.append("Overall-series carry-over points are not verified.")
469 return gaps

508def map_ac40_to_cup(ac40_effects: Mapping[str, float]) -> dict[str, float]:
509 """Map AC40 entry abilities to the 7 registered 2027 America's Cup teams.
510
511 Teams with multiple crews (e.g. B and G) are averaged.
512 Unobserved teams are omitted so the simulator draws their population prior;
513 an estimated zero remains a present, observed strength.
514 """
515 mapping = {
516 "ETNZ": ["NZL", "NZL B", "NZL G"],
517 "GB1": ["GBR", "GBR B", "GBR G"],
518 "LUNA_ROSSA": ["ITA", "ITA B", "ITA G"],
519 "ALINGHI": ["SUI"],
520 "LA_ROCHE_POSAY": ["FRA"],
521 "ARC_USA": ["USA"],
522 "AUSTRALIA": ["AUS", "AUSTRALIA"],
523 }
524 result = {}
525 for cup_team, ac40_codes in mapping.items():
526 matched = [float(ac40_effects[k]) for k in ac40_codes if k in ac40_effects]
527 if matched:
528 result[cup_team] = float(np.mean(matched))
529 return result
