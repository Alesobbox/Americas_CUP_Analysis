8def race_labels(metadata: dict) -> dict:
9 final = metadata["sessions"][-1]
10 sequences = final["race_course"]["sequences"]
11 final_leg = len(sequences)-1
12 timings, status, starts, availability = {}, {}, {}, {}
13 cleared_ocs, active_dsq, events_by_boat = {}, {}, {}
14 for event in metadata["competitor_events"]:
15 aid = event["asset_id"]
16 if aid not in metadata["boat_ids"]:
17 continue
18 events_by_boat.setdefault(aid, []).append(event)
19 status[aid] = event.get("classification", "UNSPECIFIED")
20 starts[aid] = event.get("start_state", {}).get("started")
21 penalties = event.get("penalties", [])
22 if any(p.get("reason") == "PENALTY_REASON_OCS" and p.get("state") == "PENALTY_STATE_CLEARED"
23 and p.get("cleared_at") for p in penalties):
24 cleared_ocs[aid] = True
25 active_dsq[aid] = any(p.get("type") == "PENALTY_TYPE_DISQUALIFIED"
26 and p.get("state") == "PENALTY_STATE_ACTIVE" for p in penalties)
27 boat = timings.setdefault(aid, {})
28 for timing in event.get("leg_timings", []):
29 if timing.get("gate_time"):
30 leg, t = int(timing["leg_number"]), int(timing["gate_time"])
31 if t > int(event["timestamp"]):
32 continue
33 if leg not in boat:
34 boat[leg] = t
35 availability[(aid, leg)] = event["available_at_ms"]
36 elif boat[leg] != t:
37 # Corrections supersede training labels only when available.
38 boat[leg] = t
39 availability[(aid, leg)] = event["available_at_ms"]
40 imputed = {}
41 for aid, boat in timings.items():
42 missing = set(range(final_leg+1)) - set(boat)
43 if len(missing) == 1:
44 k = next(iter(missing))
45 if 1 <= k < final_leg and (k-1 in boat) and (k+1 in boat) and boat[k-1] < boat[k+1]:
46 gate_marks = {sequences[k]["gate"].get("port_asset_id"),
47 sequences[k]["gate"].get("starboard_asset_id")}
48 evs = events_by_boat.get(aid, [])
49 zone_ev = next((ev for ev in evs
50 if (ev.get("mark_zone_state") or {}).get("inside") is True
51 and (ev.get("mark_zone_state") or {}).get("mark_id") in gate_marks
52 and boat[k-1] < int(ev["timestamp"]) < boat[k+1]), None)
53 leg_ev = next((ev for ev in evs
54 if (ev.get("course_position") or {}).get("leg_number") == k+1
55 and boat[k-1] < int(ev["timestamp"]) <= boat[k+1]), None)
56 if zone_ev is not None and leg_ev is not None:
57 boat[k] = int(zone_ev["timestamp"])
58 availability[(aid, k)] = zone_ev["available_at_ms"]
59 imputed.setdefault(aid, []).append(k)
60 def valid_finish(aid):
61 gates = timings.get(aid, {})
62 started_valid = starts.get(aid) is True or (0 in gates and cleared_ocs.get(aid, False))
63 return (set(gates) == set(range(final_leg+1)) and started_valid and not active_dsq.get(aid, False)
64 and all(gates[i] < gates[i+1] for i in range(final_leg)))
65 finishers = sorted((aid for aid in metadata["boat_ids"] if valid_finish(aid)),
66 key=lambda a: timings[a][final_leg])
67 ties = any(timings[a][final_leg] == timings[b][final_leg] for a,b in zip(finishers, finishers[1:]))
68 return {"race_id": metadata["race_id"], "date": metadata["date"],
69 "practice": metadata["practice"], "source_classification": status,
70 "source_started": starts, "timings": timings,
71 "imputed_gates": imputed, "cleared_ocs": cleared_ocs, "active_dsq": active_dsq,
72 "order": [metadata["identities"][a]["entry_id"] for a in finishers],
73 "finisher_ids": finishers, "order_tied":ties,"order_valid": not ties and len(finishers) >= 2,
74 "provenance": "source_gate_timings_with_nested_start_flag_and_cleared_ocs; not independently verified",
75 "official_order_verified": False,
76 "result_available_at_ms": max(availability.values(), default=metadata["window_end_ms"]+6000)}

79def source_completion_records(metadata: dict, label: dict) -> list[dict]:
80 """Scheduled-entry completion facts reconstructed from the same source feed.
81
82 These records describe the on-water target (did the boat record a valid
83 finish?). They are deliberately ``audited=False`` with
84 ``ascertainment='source'``: they are not an independent review and cannot
85 distinguish an exogenous breakdown from a retirement, a disqualification
86 or a recording gap. ``failure_type`` keeps that distinction visible.
87 """
88 if metadata["practice"]:
89 return []
90 final_leg = len(metadata["sessions"][-1]["race_course"]["sequences"])-1
91 finishers = set(label["finisher_ids"])
92 records = []
93 for aid in metadata["boat_ids"]:
94 gates = label["timings"].get(aid, {})
95 cls = str(label["source_classification"].get(aid) or "UNSPECIFIED").upper()
96 started = (label["source_started"].get(aid) is True
97 or (0 in gates and label["cleared_ocs"].get(aid, False)))
98 if cls.endswith("DID_NOT_START"):
99 started = False
100 completed = aid in finishers
101 record = {"asset_id": aid, "entry_id": metadata["identities"][aid]["entry_id"],
102 "race_id": metadata["race_id"], "date": metadata["date"],
103 "started": bool(started or completed), "completed": completed,
104 "audited": False, "ascertainment": "source", "competitive": True, "practice": False,
105 "available_at_ms": int(label["result_available_at_ms"])}
106 if record["started"] and not completed:
107 if label["active_dsq"].get(aid, False):
108 record["failure_type"] = "disqualified"
109 elif cls.endswith("RETIRED"):
110 record["failure_type"] = "retired"
111 else:
112 # No terminal classification: retirement or a recording gap.
113 record["failure_type"] = "unresolved"
114 record["last_gate"] = max(gates, default=-1)
115 record["final_gate"] = final_leg
116 records.append(record)
117 return records

120def leg_rows(metadata, rows, label):
121 """Valid intervals survive a later retirement. Gun-to-gate includes the start loss."""
122 final = metadata["sessions"][-1]
123 max_leg = len(final["race_course"]["sequences"])-1
124 # Winds are observed at entry, never a future race average.
125 winds = [r for r in rows if r["asset_id"] == final.get("official_wind_asset_id")
126 and r["field"] == "true_wind_speed" and r["value"] is not None]
127 result = []
128 for aid, gates in label["timings"].items():
129 imputed = set(label.get("imputed_gates", {}).get(aid, ()))
130 for leg in range(1, max_leg+1):
131 if leg in imputed or (leg-1) in imputed:
132 continue # zone-entry time is not a rounding time
133 entry = metadata["gun_ms"] if leg == 1 else gates.get(leg-1)
134 exit = gates.get(leg)
135 if not entry or not exit or exit <= entry:
136 continue
137 entry_session=next((s for s in reversed(metadata["sessions"]) if int(s["timestamp"]) <= entry),None)
138 if not entry_session or leg >= len(entry_session["race_course"]["sequences"]):
139 continue
140 distance=float(entry_session["race_course"]["course_length"])
141 wind = next((w["value"]*1.943844 for w in reversed(winds)
142 if w.get("bucket_closed_at_ms",w["timestamp_ms"]) <= entry), None)
143 result.append({"boat_id": metadata["identities"][aid]["entry_id"],
144 "leg_kind": "odd" if leg%2 else "even", "leg_index":leg-1,"distance_m": distance,
145 "duration_s": (exit-entry)/1000, "wind_knots": wind,
146 "race_id": metadata["race_id"], "date": metadata["date"],
147 "practice": metadata["practice"], "valid": True,
148 "target_censored": False, "target_source": "source_gate_interval",
149 "available_at_ms": label["result_available_at_ms"]})
150 return result

153def load_reviews(root: Path, datasets: list[dict], path: Path | None = None) -> list[dict]:
154 """Reject partial denominators and evidence-free review overrides.
155
156 Audited races must account for every scheduled entry, including nonstarters.
157 They need an evidence reference and an actual review availability timestamp.
158 """
159 path = path or root/"docs"/"forecast_completion_review.json"
160 if not path.exists():
161 return []
162 doc = json.loads(path.read_text(encoding="utf-8"))
163 if doc.get("schema_version") != "completion-review-1":
164 raise ValueError("Unsupported completion review schema")
165 by_race = {m["race_id"]: m for m in datasets}
166 result, seen = [], set()
167 for race in doc.get("races", []):
168 rid = race["race_id"]
169 if rid in seen or rid not in by_race:
170 raise ValueError("Duplicate or unknown reviewed race")
171 seen.add(rid)
172 meta = by_race[rid]
173 if meta["practice"]:
174 raise ValueError("Practice cannot establish competitive completion")
175 entries = race.get("entries", [])
176 if set(e["asset_id"] for e in entries) != set(meta["boat_ids"]) or len(entries) != len(meta["boat_ids"]):
177 raise ValueError("Reviewed denominator must include every scheduled entry exactly once")
178 if not race.get("evidence_ref") or not isinstance(race.get("available_at_ms"), int):
179 raise ValueError("Audited results require evidence and availability time")
180 for entry in entries:
181 if not isinstance(entry.get("started"), bool) or not isinstance(entry.get("completed"), bool):
182 raise ValueError("Unresolved completion is not an audited denominator")
183 if entry["completed"] and not entry["started"]:
184 raise ValueError("A nonstarter cannot be a valid finisher")
185 cause = entry.get("failure_type")
186 if entry["started"] and not entry["completed"] and cause not in ("exogenous", "time_rule"):
187 raise ValueError("An untyped DNF cannot establish random failure risk")
188 result.append(dict(entry, audited=True, competitive=True, practice=False,
189 race_id=rid, date=meta["date"], evidence_ref=race["evidence_ref"],
190 available_at_ms=race["available_at_ms"]))
191 return result
