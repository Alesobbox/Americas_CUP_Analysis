52def digest(path: Path) -> str:
53 h = hashlib.sha256()
54 with path.open("rb") as stream:
55 for block in iter(lambda: stream.read(4 * 1024 * 1024), b""):
56 h.update(block)
57 return h.hexdigest()

60def read_source(path: Path) -> tuple[dict, list[dict]]:
61 buckets, sessions, events = {}, [], []
62 h, maximum, disorder, count = hashlib.sha256(), 0, 0, 0
63 tracked = set()
64 with path.open("rb", buffering=4*1024*1024) as stream:
65 for ordinal, line in enumerate(stream, 1):
66 h.update(line)
67 count = ordinal
68 record = json.loads(line)
69 kind = next((k for k in ("session", "competitor", "telemetry") if record.get(k)), None)
70 if not kind:
71 continue
72 value = record[kind]
73 t = int(value.get("timestamp", 0))
74 if kind == "session":
75 tracked.update(c["asset_id"] for c in value.get("competitors", []))
76 tracked.update(a["id"] for a in value.get("assets", []) if a.get("type") == "ASSET_TYPE_RACE_MARK")
77 tracked.add(value.get("official_wind_asset_id"))
78 if kind == "telemetry" and value.get("asset_id") not in tracked:
79 continue
80 if kind == "telemetry":
81 disorder = max(disorder, maximum - t)
82 maximum = max(maximum, t)
83 if kind != "telemetry":
84 value = dict(value, source_ordinal=ordinal, available_at_ms=t+BUFFER_MS)
85 (sessions if kind == "session" else events).append(value)
86 continue
87 aid = value.get("asset_id")
88 field = next((f for f in FIELDS if f in value), None)
89 if not field:
90 continue
91 key = (aid, field, t//1000)
92 prior = buckets.get(key)
93 if prior and (prior["timestamp_ms"], prior["source_ordinal"]) >= (t, ordinal):
94 continue
95 measurement = value[field]
96 row = {"asset_id": aid, "field": field, "timestamp_ms": t,
97 "bucket_closed_at_ms": (t//1000+1)*1000,
98 "available_at_ms": (t//1000+1)*1000+BUFFER_MS, "source_ordinal": ordinal,
99 "value": None, "latitude": None, "longitude": None}
100 if field == "gnss_position":
101 row.update(latitude=measurement.get("latitude"), longitude=measurement.get("longitude"))
102 else:
103 row["value"] = measurement.get("value")
104 buckets[key] = row
105 if not sessions:
106 raise ValueError(f"No session metadata: {path.name}")
107 sessions.sort(key=lambda x: (int(x["timestamp"]), x["source_ordinal"]))
108 events.sort(key=lambda x: (int(x["timestamp"]), x["source_ordinal"]))
109 final = sessions[-1]
110 assets = {a["id"]: a for a in final["assets"]}
111 field = [c["asset_id"] for c in final["competitors"]]
112 marks = {g for s in final["race_course"]["sequences"] for g in
113 (s["gate"]["port_asset_id"], s["gate"]["starboard_asset_id"])}
114 keep = set(field) | marks | {final.get("official_wind_asset_id")}
115 rows = sorted((r for r in buckets.values() if r["asset_id"] in keep),
116 key=lambda r: (r["timestamp_ms"], r["source_ordinal"]))
117 gun = int(final["race_start"])
118 metadata = infer_race_metadata(str(path))
119 metadata.update(date=datetime.fromtimestamp(gun/1000, timezone.utc).date().isoformat(),
120 source_file=path.name, sha256=h.hexdigest(), bytes=path.stat().st_size, parser_version=PARSER_VERSION,
121 records=count, max_disorder_ms=disorder,
122 reorder_contract_valid=disorder <= BUFFER_MS,
123 sessions=sessions, competitor_events=events,
124 boat_ids=field, identities={aid: resolve_identity(aid, assets,
125 timestamp_ms=int(final["timestamp"])) for aid in field},
126 gun_ms=gun, window_start_ms=int(final["window_start"]),
127 window_end_ms=int(final["window_end"]),
128 practice=any(x.startswith("pr") for x in path.stem.lower().split()) or "practice" in path.stem.lower(),
129 boat_class="AC40", time_basis="event-time reconstruction with modeled latency")
130 return metadata, rows

133def source_dataset(root: Path, *, race: str | None = None, force=False) -> dict:
134 import pyarrow as pa
135 import pyarrow.parquet as pq
136 cache = root / ".forecast" / "dataset"
137 cache.mkdir(parents=True, exist_ok=True)
138 entries = []
139 for path in sorted((root/"Data").glob("*.jsonl")):
140 rid = infer_race_metadata(str(path))["race_id"]
141 if race and race != rid:
142 continue
143 meta_path, data_path = cache/f"{rid}.json", cache/f"{rid}.parquet"
144 stat = path.stat()
145 cached = json.loads(meta_path.read_text()) if meta_path.exists() else None
146 # Authenticate both source identity and the actual derived bytes.
147 # Old unsealed caches are rebuilt by this explicit dataset command;
148 # ordinary load_dataset never silently seals/trusts them.
149 reusable = False
150 if (not force and cached and cached.get("parser_version") == PARSER_VERSION and data_path.exists()
151 and cached.get("bytes") == stat.st_size and cached.get("sha256") == digest(path)):
152 try:
153 _validate_cache(cached, data_path)
154 reusable = True
155 except ValueError:
156 pass
157 if reusable:
158 metadata = cached
159 else:
160 metadata, rows = read_source(path)
161 pq.write_table(pa.Table.from_pylist(rows), data_path, compression="zstd")
162 metadata = _seal_metadata(dict(metadata, cache_seal_method="source_dataset_raw_parse"), data_path)
163 atomic_json(meta_path, metadata)
164 entries.append(_index_entry(rid, metadata))
165 if race and not entries:
166 raise ValueError(f"Unknown source race: {race}")
167 result = {"schema_version": "prediction-dataset-1", "races": entries}
168 if not race:
169 atomic_json(cache/"index.json", result)
170 elif (cache/"index.json").exists():
171 # A single-race rebuild must not leave an existing archive index bound
172 # to the previous version of that race's cache.
173 prior = json.loads((cache/"index.json").read_text())
174 replacements = {entry["race_id"]: entry for entry in entries}
175 merged = [replacements.pop(entry["race_id"], entry) for entry in prior["races"]]
176 merged.extend(replacements.values())
177 atomic_json(cache/"index.json", dict(prior, races=sorted(merged, key=lambda entry: entry["race_id"])))
178 return result

181def load_dataset(root: Path, rid: str) -> tuple[dict, list[dict]]:
182 import pyarrow.parquet as pq
183 base = root/".forecast"/"dataset"
184 metadata = json.loads((base/f"{rid}.json").read_text())
185 expected = None
186 if (base/"index.json").exists():
187 index = json.loads((base/"index.json").read_text())
188 expected = next((entry for entry in index["races"] if entry["race_id"] == rid), None)
189 if expected is None:
190 raise ValueError("Requested race is not part of the sealed forecasting dataset index")
191 _validate_cache(metadata, base/f"{rid}.parquet", expected)
192 rows=pq.read_table(base/f"{rid}.parquet").to_pylist()
193 # Deterministic availability migration for caches made before closed-bin
194 # publication was explicit. Values/ordinals/raw timestamps are unchanged.
195 for row in rows:
196 row["bucket_closed_at_ms"]=(row["timestamp_ms"]//1000+1)*1000
197 row["available_at_ms"]=row["bucket_closed_at_ms"]+BUFFER_MS
198 return metadata, rows

201def seal_existing_dataset(root: Path) -> dict:
202 """Explicit migration of a just-generated, trusted pre-seal conversion.
203
204 This function must never run implicitly on ordinary loads. The caller must
205 know the derived files came from the completed conversion in this session.
206 All raw hashes and cache structure are checked before any metadata/index is
207 rewritten. This avoids a second full JSON parsing pass while preserving an
208 auditable distinction from caches generated normally with sealing enabled.
209 """
210 import pyarrow.parquet as pq
211 base = root/".forecast"/"dataset"
212 index = json.loads((base/"index.json").read_text())
213 raw_paths = {infer_race_metadata(str(path))["race_id"]: path for path in (root/"Data").glob("*.jsonl")}
214 if set(raw_paths) != {entry["race_id"] for entry in index["races"]}:
215 raise ValueError("Explicit sealing requires a complete index matching every current raw source")
216 staged = []
217 for entry in index["races"]:
218 rid = entry["race_id"]
219 meta_path, table_path = base/f"{rid}.json", base/f"{rid}.parquet"
220 metadata = json.loads(meta_path.read_text())
221 raw_path = raw_paths[rid]
222 if (metadata.get("race_id") != rid or metadata.get("source_file") != raw_path.name
223 or metadata.get("sha256") != entry.get("sha256")
224 or metadata.get("bytes") != raw_path.stat().st_size
225 or metadata.get("sha256") != digest(raw_path)):
226 raise ValueError(f"Cannot seal {rid}: current raw source identity does not match the conversion")
227 schema = pq.read_schema(table_path)
228 if not {"asset_id", "field", "timestamp_ms", "source_ordinal"} <= set(schema.names):
229 raise ValueError(f"Cannot seal {rid}: malformed derived table")
230 sealed = _seal_metadata(dict(metadata, cache_seal_method="explicit_fresh_conversion_migration_raw_hash_verified"), table_path)
231 staged.append((rid, meta_path, sealed))
232 entries = []
233 for rid, meta_path, sealed in staged:
234 atomic_json(meta_path, sealed)
235 entries.append(_index_entry(rid, sealed))
236 atomic_json(base/"index.json", dict(index, races=entries))
237 return {"races": len(entries), "source_bytes_verified": sum(entry["bytes"] for entry in entries),
238 "seal_method": "explicit_fresh_conversion_migration_raw_hash_verified"}
