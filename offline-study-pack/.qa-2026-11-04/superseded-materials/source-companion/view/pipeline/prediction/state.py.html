8def transition(dt: float, q: float) -> tuple[np.ndarray, np.ndarray]:
9 f = np.eye(4)
10 f[:2, 2:] = np.eye(2)*dt
11 covariance = q*np.block([[np.eye(2)*dt**3/3, np.eye(2)*dt**2/2],
12 [np.eye(2)*dt**2/2, np.eye(2)*dt]])
13 return f, covariance

20 def predict(self, timestamp_ms):
21 if self.t is None or timestamp_ms < self.t:
22 raise ValueError("Prediction requires a past filtered observation")
23 f, noise = transition((timestamp_ms-self.t)/1000, self.q)
24 return f@self.mean, f@self.cov@f.T+noise

26 def observe(self, timestamp_ms, xy):
27 if self.t is None:
28 self.mean = np.r_[xy, [0., 0.]]
29 self.cov = np.diag([self.r, self.r, 400., 400.])
30 else:
31 m, p = self.predict(timestamp_ms)
32 h = np.eye(2, 4)
33 gain = np.linalg.solve(h@p@h.T+self.r*np.eye(2), h@p).T
34 self.mean = m+gain@(np.asarray(xy)-h@m)
35 # Joseph form preserves positive covariance under rounding.
36 a = np.eye(4)-gain@h
37 self.cov = a@p@a.T+self.r*gain@gain.T
38 self.t = timestamp_ms

209def fit_filter_parameters(races):
210 """Conservative pooled innovation estimates from strictly earlier raw positions.
211
212 GNSS noise and real acceleration cannot be fully separated at 1 Hz. Upper-quantile
213 racing accelerations ensure 95% filter ellipses cover manoeuvres and short outages.
214 """
215 differences = []
216 dates = set()
217 for metadata, rows in races:
218 if metadata.get("practice"):
219 continue
220 dates.add(metadata["date"])
221 gun_ms = int(metadata.get("gun_ms", 0))
222 end_ms = int(metadata.get("window_end_ms", 1 << 60))
223 streams={}
224 for row in rows:
225 t_ms = int(row["timestamp_ms"])
226 if (gun_ms <= t_ms <= end_ms and row["field"]=="gnss_position" and row["asset_id"] in metadata["boat_ids"]
227 and row.get("latitude") is not None and row.get("longitude") is not None
228 and math.isfinite(row["latitude"]) and math.isfinite(row["longitude"])):
229 streams.setdefault(row["asset_id"],[]).append(row)
230 for aid in metadata["boat_ids"]:
231 stream=streams.get(aid,[])
232 if len(stream)<3:
233 continue
234 times=np.array([r["timestamp_ms"] for r in stream],float)/1000
235 origin=(stream[0]["latitude"],stream[0]["longitude"])
236 xy=np.array([project(r["latitude"],r["longitude"],origin) for r in stream])
237 dt=np.diff(times)
238 valid=(dt[:-1]>=.5)&(dt[1:]>=.5)&(dt[:-1]<=1.5)&(dt[1:]<=1.5)
239 velocity=np.diff(xy,axis=0)/np.maximum(dt[:,None],1e-6)
240 accel=np.diff(velocity,axis=0)/np.maximum((dt[:-1]+dt[1:])[:,None]/2,1e-6)
241 differences.extend((np.sum(accel[valid]**2,axis=1)/2).tolist())
242 scale = float(np.percentile(differences, 96)) if differences else 8.4
243 return {"q": max(.2, min(20., scale)), "r": max(.25, min(25., scale/4)),
244 "training_dates": sorted(dates), "innovation_count": len(differences),
245 "method": "pooled-p96-racing-second-difference; noise-motion-confounded; mask certification required"}

248def certify_outages(races, *, horizons=(2, 5, 10, 20, 30, 60, 90), parameters=None):
249 """Nonoverlapping position masks. Independent race dates and strata remain explicit.
250
251 Only a contiguous passing prefix is certified. Unobserved strata never inherit
252 the steady-state certificate. Natural outages are excluded from the truth set.
253 """
254 records = {s: {h: [] for h in horizons} for s in ("steady", "turn", "gate", "recovery")}
255 by_date = {day: fit_filter_parameters([(m,r) for m,r in races if m["date"] < day])
256 for day in sorted({m["date"] for m,r in races})}
257 for metadata, rows in races:
258 if metadata.get("practice"):
259 continue
260 # The hidden mask/date cannot fit its own process/noise parameters.
261 prior_parameters = by_date[metadata["date"]]
262 race = CausalRace(metadata, rows, **{k:v for k,v in prior_parameters.items() if k in ("q", "r")})
263 for aid in metadata["boat_ids"]:
264 states = race.positions.get(aid, [])
265 times = [s[0] for s in states]
266 raw_truth={r["timestamp_ms"]:r for r in race.by_stream.get((aid,"gnss_position"),[])
267 if r.get("latitude") is not None and r.get("longitude") is not None}
268 next_start = 0
269 for i in range(15, len(states)):
270 t, mean, covariance = states[i]
271 mask_start=race.position_available_times[aid][i]
272 if mask_start < next_start or mask_start < metadata["gun_ms"]:
273 continue
274 next_start = mask_start+max(horizons)*1000+10000
275 feature = race.features(aid, mask_start)
276 stratum = feature.get("stratum")
277 if not stratum:
278 continue
279 for h in horizons:
280 target=mask_start+h*1000
281 j = bisect.bisect_left(times, target)
282 if j >= len(times) or abs(times[j]-target) > 1100:
283 continue
284 # No observed gap inside the hidden block: genuine complete truth only.
285 if any(b-a > 2000 for a,b in zip(times[i:j], times[i+1:j+1])):
286 continue
287 f, q = transition((times[j]-t)/1000, race.q)
288 pred, p = f@mean, f@covariance@f.T+q
289 truth = raw_truth[times[j]]
290 delta = project(truth["latitude"], truth["longitude"], race.origin)-pred[:2]
291 metric = float(delta@np.linalg.solve(p[:2,:2]+race.r*np.eye(2), delta))
292 error = float(np.linalg.norm(delta))
293 records[stratum][h].append((metadata["date"], error, metric <= 5.991,
294 error/feature["leg_length_m"]))
295 result = {}
296 for stratum, by_h in records.items():
297 horizon, details, contiguous = 0, {}, True
298 for h, values in by_h.items():
299 count, dates = len(values), len({x[0] for x in values})
300 coverage = sum(x[2] for x in values)/count if count else None
301 error = float(np.median([x[1] for x in values])) if count else None
302 normalized = float(np.median([x[3] for x in values])) if count else None
303 passed = count >= 50 and dates >= 2 and coverage >= .9 and normalized < .1
304 if contiguous and passed:
305 horizon = h
306 else:
307 contiguous = False
308 details[str(h)] = {"count": count, "dates": dates, "ellipse_coverage": coverage,
309 "median_error_m": error, "median_normalized_error": normalized, "passed": passed}
310 result[stratum] = {"horizon_s": horizon, "masks": details,
311 "loss_modes": ["position", "position_and_velocity_channels"],
312 "velocity_measurement_updates": False}
313 return result
