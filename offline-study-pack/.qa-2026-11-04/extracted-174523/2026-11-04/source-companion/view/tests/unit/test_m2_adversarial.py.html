97 def test_tack_wrapping_across_true_north_0_deg(self):
98 """Tack turning across True North (330 deg to 30 deg, delta=60 deg) is detected."""
99 n = 100
100 timestamps = [1790079000000 + i * 100 for i in range(n)]
101 heading = [330.0] * n
102 twa = [-35.0] * n
103 sog = [26.0] * n
104
105 # Turn phase: frames 30 to 60 (turn from 330 to 30 across 360/0)
106 for i in range(30, 61):
107 alpha = (i - 30) / 30.0
108 # 330 + alpha * 60: at alpha=0 -> 330, alpha=0.5 -> 360/0, alpha=1.0 -> 30
109 raw_hdg = 330.0 + alpha * 60.0
110 heading[i] = raw_hdg % 360.0
111 twa[i] = -35.0 + alpha * 70.0 # -35 to +35 deg
112 sog_dip = 6.0 * (1.0 - abs(alpha - 0.5) * 2.0)
113 sog[i] = 26.0 - sog_dip
114
115 for i in range(61, n):
116 heading[i] = 30.0
117 twa[i] = 35.0
118 sog[i] = 25.5
119
120 manoeuvres = self.detector.detect_manoeuvres_for_boat(
121 timestamps, heading, sog, twa
122 )
123 self.assertEqual(len(manoeuvres), 1)
124 m = manoeuvres[0]
125 self.assertEqual(m.type, "TACK")
126 self.assertAlmostEqual(m.turn_delta_deg, 60.0, delta=5.0)
127 self.assertGreaterEqual(m.speed_loss_kt, 4.0)
