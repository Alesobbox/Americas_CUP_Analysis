20class TestMonotonicProgressAndCourseAxis(unittest.TestCase):
21 """Verifies continuous monotonic race progress calculation."""
22
23 def setUp(self):
24 self.analytics = AnalyticsEngine(
25 gun_time_ms=GUN_TIME_EPOCH_MS,
26 course_axis_deg=44.39,
27 nominal_leg_length_m=2400.0,
28 )
29
30 def test_course_axis_unit_vectors(self):
31 """Course axis unit vectors are orthonormal and oriented at 44.39 deg."""
32 rad = math.radians(44.39)
33 self.assertAlmostEqual(self.analytics.u_upwind[0], math.sin(rad), places=4)
34 self.assertAlmostEqual(self.analytics.u_upwind[1], math.cos(rad), places=4)
35 # Downwind is negative upwind
36 self.assertAlmostEqual(self.analytics.u_downwind[0], -self.analytics.u_upwind[0], places=4)
37 self.assertAlmostEqual(self.analytics.u_downwind[1], -self.analytics.u_upwind[1], places=4)
38 # Dot product with lateral vector is 0 (orthogonal)
39 dot = (
40 self.analytics.u_upwind[0] * self.analytics.u_lat[0]
41 + self.analytics.u_upwind[1] * self.analytics.u_lat[1]
42 )
43 self.assertAlmostEqual(dot, 0.0, places=6)
44
45 def test_progress_monotonicity_guarantee(self):
46 """Simulated progress along leg guarantees non-decreasing monotonic series."""
47 # Advances that fluctuate (e.g. tacking upwind away from mark then towards)
48 raw_advances = [100.0, 250.0, 220.0, 310.0, 290.0, 450.0, 600.0]
49 monotonic_prog = []
50 max_p = 0.0
51 for adv in raw_advances:
52 if adv > max_p:
53 max_p = adv
54 monotonic_prog.append(max_p)
55
56 for i in range(len(monotonic_prog) - 1):
57 self.assertGreaterEqual(monotonic_prog[i + 1], monotonic_prog[i])
