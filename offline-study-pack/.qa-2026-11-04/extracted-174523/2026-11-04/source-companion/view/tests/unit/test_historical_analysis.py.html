25def synthetic_race(*, mark_y=0.0, revise_boundary=True, official_wind=True):
26 """Small race with exact 1 s timestamps and four distinct race outcomes."""
27 gun = 1_000_000
28 times = list(range(gun - 30_000, gun + 20_001, 1000))
29 n = len(times)
30 boundary = [(-100.0, -100.0), (100.0, -100.0),
31 (100.0, 100.0), (-100.0, 100.0)]
32 revised = [(-100.0, -100.0), (100.0, -100.0),
33 (100.0, 20.0), (-100.0, 20.0)]
34 session = SimpleNamespace(
35 boat_class="AC40", race_type="FLEET", course_axis_deg=0.0,
36 race_boundary=boundary,
37 boundary_timeline=[{"t": gun + 5_000, "race": revised}]
38 if revise_boundary else [],
39 )
40 boats = {}
41 comps = {}
42 legs = {}
43 for code, status, x_offset in (("FIN", "FINISHED", 0.0),
44 ("DNF", "DNF", 20.0),
45 ("RET", "RET", 40.0),
46 ("OCS", "OCS", 60.0)):
47 rel_s = [(t - gun) / 1000 for t in times]
48 y = [-20.0 if s <= 1 else 5.0 * (s - 1) if s <= 6
49 else 25.0 - 5.0 * (s - 6) for s in rel_s]
50 sog = [10.0 if s < 4 else 5.0 for s in rel_s]
51 boats[code] = SimpleNamespace(
52 asset_id=code.lower(), name=f"Team {code}", color="#ffffff",
53 x=[x_offset] * n, y=y,
54 heading=[0.0 if s <= 6 else 180.0 for s in rel_s],
55 cog=[0.0 if s <= 6 else 180.0 for s in rel_s],
56 sog_kt=sog, twd_deg=[0.0] * n,
57 )
58 gate = SimpleNamespace(leg=1, crossing_time=gun + 6_000,
59 mark_chosen="RM8", side="PORT", is_official=True)
60 comps[code] = SimpleNamespace(
61 team_code=code, classification=status,
62 start=SimpleNamespace(crossing_time=gun + 1_000, ocs=status == "OCS"),
63 gates=[gate] if code == "FIN" else [],
64 finish_time_ms=gun + 12_000 if code == "FIN" else None,
65 official_rank=1 if code == "FIN" else None,
66 elapsed_s=11.0 if code == "FIN" else None,
67 )
68 legs[code] = [0 if s < 1 else 1 if s <= 6 else 2 for s in rel_s]
69 mark_y_series = [mark_y] * n
70 marks = {
71 "RM8": SimpleNamespace(x=[-10.0] * n, y=mark_y_series),
72 "RM6": SimpleNamespace(x=[10.0] * n, y=mark_y_series),
73 }
74 wind = ([{"t": gun, "twd_deg": 359.0, "tws_kt": 12.0},
75 {"t": gun + 1_000, "twd_deg": 1.0, "tws_kt": 12.0}]
76 if official_wind else [])
77 ingestion = SimpleNamespace(
78 session=session, grid_timestamps_ms=times, boats=boats, marks=marks,
79 official_wind=SimpleNamespace(raw_timeline=wind),
80 )
81 reconstruction = SimpleNamespace(
82 chronology=SimpleNamespace(gun_time=gun, window_end=times[-1]),
83 competitors=comps,
84 )
85 tack = SimpleNamespace(type="TACK", timestamp=gun + 4_000,
86 entry_speed_kt=10.0, min_speed_kt=5.0,
87 speed_loss_kt=5.0)
88 return ingestion, reconstruction, {"FIN": [tack]}, legs

91def analyze_synthetic(**kwargs):

93 return build_race_analysis(
94 ingestion, reconstruction, manoeuvres, legs,
95 "synthetic", "Synthetic race", "2026-10-08", "Test course",
96 )
