31def sample_mark(data: Dict[str, List[float]], index: int, fraction: float) -> Tuple[float, float]:
32 """Python reference equivalent to TypeScript ReplayEngine.sampleMark."""
33 xs = data["x"]
34 ys = data["y"]
35 n = len(xs)
36 next_idx = min(index + 1, n - 1)
37 x0 = xs[index]
38 x1 = xs[next_idx]
39 y0 = ys[index]
40 y1 = ys[next_idx]
41 return lerp(x0, x1, fraction), lerp(y0, y1, fraction)

44def verify_boundary_timeline(data: Dict[str, Any]) -> Dict[str, Any]:
45 """Verifies boundary timeline revisions, timestamps, and point counts."""
46 course = data.get("course", {})
47 timeline = course.get("boundary_timeline", [])
48 gun_time = data.get("chronology", {}).get("gun_time", 1790079000000)
49
50 results: Dict[str, Any] = {
51 "passed": False,
52 "revision_count": len(timeline),
53 "revisions": [],
54 "errors": [],
55 }
56
57 if not timeline:
58 results["errors"].append("course.boundary_timeline is empty or missing")
59 return results
60
61 if len(timeline) < 3:
62 results["errors"].append(f"Expected at least 3 boundary revisions, got {len(timeline)}")
63
64 # Check each revision
65 for i, rev in enumerate(timeline):
66 t = rev.get("t")
67 race_pts = rev.get("race", [])
68 safety_pts = rev.get("safety", [])
69 delta_gun_s = (t - gun_time) / 1000.0 if t is not None else None
70
71 rev_summary = {
72 "index": i,
73 "timestamp": t,
74 "delta_gun_s": round(delta_gun_s, 3) if delta_gun_s is not None else None,
75 "race_point_count": len(race_pts),
76 "safety_point_count": len(safety_pts),
77 }
78 results["revisions"].append(rev_summary)
79
80 # Validate points format
81 for pt in race_pts + safety_pts:
82 if not isinstance(pt, (list, tuple)) or len(pt) != 2:
83 results["errors"].append(f"Invalid polygon point {pt} in revision {i}")
84 elif not all(isinstance(v, (int, float)) and math.isfinite(v) for v in pt):
85 results["errors"].append(f"Non-finite polygon coord {pt} in revision {i}")
86
87 # Specific check 1: Pre-start revisions have 8-point race boundaries
88 rev0 = timeline[0]
89 if len(rev0.get("race", [])) != 8:
90 results["errors"].append(f"Revision 0 race boundary has {len(rev0.get('race', []))} points, expected 8")

92 # Specific check 2: Transition at T = +15s (1790079015154) collapses to 4-point corridor
93 t_plus_15_rev = next((r for r in timeline if abs((r.get("t", 0) - gun_time) / 1000.0 - 15.154) < 1.0), None)
94 if not t_plus_15_rev:
95 results["errors"].append("Missing boundary revision at T ≈ +15s")
96 else:
97 race_pts = t_plus_15_rev.get("race", [])
98 if len(race_pts) != 4:
99 results["errors"].append(f"Revision at T=+15s has {len(race_pts)} race points, expected 4 (active corridor)")
100 delta_s = (t_plus_15_rev["t"] - gun_time) / 1000.0
101 rev_summary["transition_delta_s"] = delta_s
102
103 # Specific check 3: Safety boundaries remain 8 points throughout
104 for i, rev in enumerate(timeline):
105 if len(rev.get("safety", [])) != 8:
106 results["errors"].append(f"Revision {i} safety boundary has {len(rev.get('safety', []))} points, expected 8")
107
108 results["passed"] = len(results["errors"]) == 0
109 return results

162def verify_sample_mark_frame_integrity(data: Dict[str, Any]) -> Dict[str, Any]:
163 """Exhaustively tests sampleMark across all 19,488 frames and sub-frame fractions."""
164 telemetry = data.get("telemetry", {})
165 marks = telemetry.get("marks", {})
166 timestamps = telemetry.get("timestamps", [])
167 n_frames = len(timestamps)
168
169 results: Dict[str, Any] = {
170 "passed": False,
171 "total_samples_evaluated": 0,
172 "nan_count": 0,
173 "null_count": 0,
174 "inf_count": 0,
175 "range_violations": 0,
176 "min_x": float("inf"),
177 "max_x": float("-inf"),
178 "min_y": float("inf"),
179 "max_y": float("-inf"),
180 "errors": [],
181 }
182
183 fractions = [0.0, 0.25, 0.5, 0.75]
184
185 for idx in range(n_frames):
186 for frac in fractions:
187 for k, stream in marks.items():
188 results["total_samples_evaluated"] += 1
189 try:
190 x, y = sample_mark(stream, idx, frac)
191 except Exception as e:
192 results["errors"].append(f"Exception sampling mark {k} at idx={idx}, frac={frac}: {e}")
193 continue
194
195 if x is None or y is None:
196 results["null_count"] += 1
197 elif math.isnan(x) or math.isnan(y):
198 results["nan_count"] += 1
199 elif math.isinf(x) or math.isinf(y):
200 results["inf_count"] += 1
201 else:
202 if x < results["min_x"]: results["min_x"] = x
203 if x > results["max_x"]: results["max_x"] = x
204 if y < results["min_y"]: results["min_y"] = y
205 if y > results["max_y"]: results["max_y"] = y
206
207 # Sanity bounds check for Gulf of Naples AC40 local coordinate frame
208 if not (-3000.0 <= x <= 3000.0 and -3000.0 <= y <= 3000.0):
209 results["range_violations"] += 1
210
211 if results["nan_count"] > 0:
212 results["errors"].append(f"Found {results['nan_count']} NaN coordinates in sampleMark")
213 if results["null_count"] > 0:
214 results["errors"].append(f"Found {results['null_count']} null coordinates in sampleMark")
215 if results["inf_count"] > 0:
216 results["errors"].append(f"Found {results['inf_count']} Inf coordinates in sampleMark")
217 if results["range_violations"] > 0:
218 results["errors"].append(f"Found {results['range_violations']} out-of-range coordinates (beyond [-3000, 3000] m)")
219
220 results["passed"] = len(results["errors"]) == 0
221 return results
