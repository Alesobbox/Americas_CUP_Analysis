25def _digamma_positive(value: float) -> float:
26 """Stable positive-argument digamma; avoids a mandatory scipy import."""
27 correction = 0.
28 while value < 8.:
29 correction -= 1. / value
30 value += 1.
31 inverse = 1. / value
32 return (correction + math.log(value) - .5 * inverse - inverse**2/12
33 + inverse**4/120 - inverse**6/252)

45@dataclass
46class StrengthModel:
47 """Ridge-MAP Plackett--Luce reference, fitted only to valid finisher orders.
48
49 Each row has ``order`` (best first). An optional ``wind_knots`` supports a
50 strongly pooled entry-specific response; a common wind intercept would
51 cancel. Nonfinishers must not be appended to ``order`` as weak finishers.
52 """
53
54 effects: dict[str, float] = field(default_factory=dict)
55 wind_effects: dict[str, float] = field(default_factory=dict)
56 wind_center: float = 0.0
57 wind_scale: float = 1.0
58 n_races: int = 0
59 strength_sd: float = 1.2
60 condition_sd: float = 0.25
61 discrimination_scale: float = 1.0
62 method: str = "ridge-plackett-luce-1"
63 covariance: list[list[float]] = field(default_factory=list)
64 boat_order: list[str] = field(default_factory=list)

66 @classmethod
67 def fit(cls, rows: Iterable[Mapping[str, Any]], *, condition_effects: bool = False,
68 strength_sd: float = 1.2, condition_sd: float = 0.25,
69 practice_weight: float = 0.0,
70 discrimination_scale: float = 1.8) -> "StrengthModel":
71 valid_rows = []
72 weights = []
73 for r in rows:
74 if len(r.get("order", [])) < 2 or r.get("order_valid", True) is False or r.get("competitive", True) is False:
75 continue
76 is_practice = bool(r.get("practice", False))
77 w = practice_weight if is_practice else 1.0
78 if w > 0:
79 valid_rows.append(dict(r))
80 weights.append(float(w))
81 for row in valid_rows:
82 if len(set(row["order"])) != len(row["order"]):
83 raise ValueError("An order cannot contain duplicate entries")
84 boats = sorted({str(b) for r in valid_rows for b in r["order"]})
85 if not boats:
86 return cls(strength_sd=strength_sd, condition_sd=condition_sd,
87 discrimination_scale=discrimination_scale)
88 winds = [float(r["wind_knots"]) for r in valid_rows if _positive(r.get("wind_knots"))]
89 center = float(np.mean(winds)) if winds else 0.0
90 scale = max(float(np.std(winds)), 1.0) if winds else 1.0
91 n = len(boats)
92 dimensions = n * (2 if condition_effects else 1)
93 lookup = {b: i for i, b in enumerate(boats)}
94 precision = np.full(dimensions, 1.0 / strength_sd ** 2)
95 if condition_effects:
96 precision[n:] = 1.0 / condition_sd ** 2
97 designs = []
98 for row in valid_rows:
99 order = [lookup[str(b)] for b in row["order"]]
100 wind = ((float(row["wind_knots"]) - center) / scale
101 if _positive(row.get("wind_knots")) else 0.0)
102 design = np.zeros((len(order), dimensions))
103 design[np.arange(len(order)), order] = 1.0
104 if condition_effects:
105 design[np.arange(len(order)), np.array(order) + n] = wind
106 designs.append(design)
107
108 def objective(beta: np.ndarray, derivatives: bool = True):
109 value = 0.5 * float(np.dot(precision * beta, beta))
110 gradient = precision * beta
111 hessian = np.diag(precision)
112 for w, design in zip(weights, designs):
113 for k in range(len(design) - 1):
114 x = design[k:]
115 logits = x @ beta
116 peak = float(np.max(logits))
117 weights_k = np.exp(logits - peak)
118 weights_k /= weights_k.sum()
119 value += w * (peak + math.log(float(np.exp(logits - peak).sum())) - logits[0])
120 if derivatives:
121 mean = weights_k @ x
122 gradient += w * (mean - x[0])
123 centered = x - mean
124 hessian += w * ((centered.T * weights_k) @ centered)
125 return value, gradient, hessian
126
127 beta = np.zeros(dimensions)
128 for _ in range(50):
129 value, gradient, hessian = objective(beta)
130 step = np.linalg.solve(hessian, gradient)
131 if float(np.linalg.norm(step)) < 1e-9:
132 break
133 fraction = 1.0
134 while fraction > 1e-7:
135 trial = beta - fraction * step
136 if objective(trial, False)[0] <= value:
137 beta = trial
138 break
139 fraction *= 0.5
140 else:
141 break
142 beta[:n] -= beta[:n].mean()
143 if condition_effects:
144 beta[n:] -= beta[n:].mean()
145 # Laplace approximation: Gaussian posterior with the MAP Hessian. The
146 # ridge prior makes the Hessian positive definite.
147 covariance = np.linalg.inv(objective(beta)[2])
148 covariance = (covariance + covariance.T) / 2
149 return cls(dict(zip(boats, map(float, beta[:n]))),
150 dict(zip(boats, map(float, beta[n:]))) if condition_effects else {},
151 center, scale, len(valid_rows), strength_sd, condition_sd,
152 discrimination_scale=discrimination_scale,
153 covariance=covariance.tolist(), boat_order=boats)

155 def sample_log_worth(self, boat_ids: Sequence[str], rng: np.random.Generator, n: int,
156 wind_knots: float | None = None,
157 state_features: Mapping[str, float] | None = None,
158 *, state_coefficient: float = 0.25) -> np.ndarray:
159 """Posterior draws of log-worth, shape (n, len(boat_ids)).
160
161 Fitted entries use the Laplace posterior (correlated). Unseen entries
162 draw independently from the ridge prior, so a new boat is uncertain
163 rather than average-with-certainty.
164 """
165 wind = (wind_knots - self.wind_center) / self.wind_scale if wind_knots is not None else 0.0
166 mean = self.log_worth(boat_ids, wind_knots, state_features, state_coefficient=state_coefficient)
167 draws = np.repeat(mean[None, :], n, axis=0)
168 known_indices = {b: i for i, b in enumerate(self.boat_order)}
169 specified = set(self.boat_order) | set(self.effects.keys())
170 m = len(self.boat_order)
171 conditional = bool(self.wind_effects)
172 if self.covariance and m:
173 cov = np.asarray(self.covariance, dtype=float)
174 # Linear map from parameters to the requested fitted entries.
175 columns = [j for j, b in enumerate(boat_ids) if b in known_indices]
176 if columns:
177 design = np.zeros((len(columns), cov.shape[0]))
178 for row, j in enumerate(columns):
179 design[row, known_indices[boat_ids[j]]] = 1.0
180 if conditional:
181 design[row, m + known_indices[boat_ids[j]]] = wind
182 local = design @ cov @ design.T
183 sampled_shifts = rng.multivariate_normal(np.zeros(len(columns)), local, size=n,
184 method="cholesky")
185 if len(columns) > 1:
186 sampled_shifts -= sampled_shifts.mean(axis=1, keepdims=True)
187 draws[:, columns] += sampled_shifts
188 unseen = [j for j, b in enumerate(boat_ids) if b not in specified]
189 if unseen:
190 sd = math.sqrt(self.strength_sd**2 + (wind * self.condition_sd)**2 if conditional else self.strength_sd**2)
191 draws[:, unseen] += rng.normal(0.0, sd, size=(n, len(unseen)))
192 return draws

194 def log_worth(self, boat_ids: Sequence[str], wind_knots: float | None = None,
195 state_features: Mapping[str, float] | None = None,
196 *, state_coefficient: float = 0.25) -> np.ndarray:
197 wind = (wind_knots - self.wind_center) / self.wind_scale if wind_knots is not None else 0.0
198 # A conservative fixed benchmark extension, not an estimated physical law.
199 return np.array([self.effects.get(b, 0.0) + wind * self.wind_effects.get(b, 0.0)
200 + state_coefficient * (state_features or {}).get(b, 0.0)
201 for b in boat_ids], dtype=float)

203 def predict_win_probabilities(self, boat_ids: Sequence[str], wind_knots: float | None = None,
204 state_features: Mapping[str, float] | None = None) -> dict[str, float]:
205 if not boat_ids:
206 return {}
207 logits = self.log_worth(boat_ids, wind_knots, state_features)
208 scaled = logits * self.discrimination_scale
209 worth = np.exp(scaled - scaled.max())
210 return dict(zip(boat_ids, map(float, worth / worth.sum())))

315@dataclass
316class ConditionalLogitModel:
317 """Small jointly regularized PL state benchmark, with past-only scaling.
318
319 Rows use ``order`` and ``state_features[boat][name]``. Each race's total
320 checkpoint weight must be one (``weight``); callers cannot manufacture
321 independent races by repeating ten-second observations. Missing feature
322 values use the training mean. The diffusion feature is derived exclusively
323 from current progress, never eventual race duration.
324 """
325
326 effects: dict[str, float] = field(default_factory=dict)
327 feature_names: list[str] = field(default_factory=lambda: ["diffusion_lead", "relative_vmg", "crosswind_fraction"])
328 feature_means: list[float] = field(default_factory=list)
329 feature_scales: list[float] = field(default_factory=list)
330 state_coefficients: list[float] = field(default_factory=list)
331 n_rows: int = 0
332 method: str = "conditional-ridge-plackett-luce-1"
333
334 @staticmethod
335 def _features(values: Mapping[str, Any]) -> dict[str, Any]:
336 output = dict(values)
337 if "diffusion_lead" not in output and "lead_fraction" in output:
338 output["diffusion_lead"] = float(output["lead_fraction"]) / math.sqrt(
339 max(float(output.get("remaining_fraction", 1.0)), 0.01))
340 return output

342 @classmethod
343 def fit(cls, rows: Iterable[Mapping[str, Any]], *, raw_lead_ablation: bool = False) -> "ConditionalLogitModel":
344 rows = [dict(r) for r in rows if _competitive(r) and len(r.get("order", [])) >= 2
345 and r.get("order_valid", True) is not False]
346 names = ["lead_fraction" if raw_lead_ablation else "diffusion_lead", "relative_vmg", "crosswind_fraction"]
347 if not rows:
348 return cls(feature_names=names, feature_means=[0.] * 3,
349 feature_scales=[1.] * 3, state_coefficients=[0.] * 3)
350 boats = sorted({str(b) for row in rows for b in row["order"]})
351 lookup = {b: j for j, b in enumerate(boats)}
352 observed = []
353 for row in rows:
354 for b in row["order"]:
355 features = cls._features(row.get("state_features", {}).get(b, {}))
356 observed.append([features.get(name, np.nan) for name in names])
357 observed = np.asarray(observed, dtype=float)
358 means = np.array([np.nanmean(observed[:, k]) if np.any(np.isfinite(observed[:, k])) else 0.
359 for k in range(3)])
360 scales = np.array([max(float(np.nanstd(observed[:, k])), 1e-6)
361 if np.any(np.isfinite(observed[:, k])) else 1. for k in range(3)])
362 precision = np.r_[np.full(len(boats), 4.), np.full(3, 16.)]
363 designs = []
364 # Default row weights normalize repeated checkpoints within race.
365 race_counts: dict[str, int] = {}
366 for index, row in enumerate(rows):
367 race_key = str(row.get("race_id", f"row:{index}"))
368 race_counts[race_key] = race_counts.get(race_key, 0) + 1
369 for index, row in enumerate(rows):
370 design = np.zeros((len(row["order"]), len(boats) + 3))
371 for j, boat in enumerate(row["order"]):
372 design[j, lookup[str(boat)]] = 1.
373 features = cls._features(row.get("state_features", {}).get(boat, {}))
374 values = np.array([features.get(name, means[k]) for k, name in enumerate(names)], dtype=float)
375 values[~np.isfinite(values)] = means[~np.isfinite(values)]
376 design[j, len(boats):] = (values - means) / scales
377 weight = float(row.get("weight", 1. / race_counts[str(row.get("race_id", f"row:{index}"))]))
378 if not 0 < weight <= 1:
379 raise ValueError("Checkpoint weight must lie in (0,1]")
380 designs.append((design, weight))
381 beta = np.zeros(len(precision))
382 for _ in range(40):
383 grad = precision * beta
384 hess = np.diag(precision)
385 for design, weight in designs:
386 for k in range(len(design) - 1):
387 x = design[k:]
388 score = x @ beta
389 p = np.exp(score - np.max(score))
390 p /= p.sum()
391 mean = p @ x
392 grad += weight * (mean - x[0])
393 centered = x - mean
394 hess += weight * (centered.T * p) @ centered
395 step = np.linalg.solve(hess, grad)
396 # The strongly regularized, standardized model has a bounded step;
397 # damping protects sparse near-separable examples.
398 step /= max(1., float(np.linalg.norm(step)))
399 beta -= step
400 if np.linalg.norm(step) < 1e-8:
401 break
402 return cls(dict(zip(boats, map(float, beta[:len(boats)]))), names,
403 list(map(float, means)), list(map(float, scales)),
404 list(map(float, beta[len(boats):])), len(rows))

406 def predict_win_probabilities(self, boat_ids: Sequence[str],
407 state_features: Mapping[str, Mapping[str, Any]] | None = None, **_: Any) -> dict[str, float]:
408 if not boat_ids:
409 return {}
410 means = np.array(self.feature_means or [0.] * 3)
411 scales = np.array(self.feature_scales or [1.] * 3)
412 beta = np.array(self.state_coefficients or [0.] * 3)
413 scores = []
414 for boat in boat_ids:
415 features = self._features((state_features or {}).get(boat, {}))
416 values = np.array([features.get(name, means[k]) for k, name in enumerate(self.feature_names)])
417 values = np.where(np.isfinite(values), values, means)
418 scores.append(self.effects.get(boat, 0.) + float(((values - means) / scales) @ beta))
419 p = np.exp(np.asarray(scores) - max(scores))
420 return dict(zip(boat_ids, map(float, p / p.sum())))

426@dataclass
427class CompletionModel:
428 """Pooled DNS and in-race non-completion posteriors over scheduled entries.
429
430 Two ascertainment tiers are accepted and counted separately:
431
432 * ``audited=True`` independent reviews. Started records require
433 ``completed=True`` or ``failure_type='exogenous'``; unknown/time-rule
434 outcomes block availability (no approved censoring treatment).
435 * ``ascertainment='source'`` reconstructions from the same feed. For the
436 first-valid-on-water-finisher target every started non-completion
437 (retired, disqualified or unresolved) removes the boat from the on-water
438 finish, so all of them count as non-completion events. Their cause is not
439 claimed; ``evidence_tier`` reports that the rate is source-reconstructed.
440
441 Practice contributes no counts. A race with an audited record uses it in
442 preference to the source reconstruction (caller responsibility).
443 """
444
445 dns_events: int = 0
446 scheduled_audited: int = 0
447 failure_events: int = 0
448 starters_ascertained: int = 0
449 prior_alpha: float = 0.0
450 prior_beta: float = 0.0
451 excluded_records: int = 0
452 unresolved_audited_starters: int = 0
453 source_records: int = 0
454 source_failure_types: dict[str, int] = field(default_factory=dict)
455 method: str = "pooled-beta-completion-2"
456
457 @property

459 if not self.scheduled_audited:
460 return "bayesian_prior" if (self.prior_alpha > 0 and self.prior_beta > 0) else None
461 if self.source_records == 0:
462 return "independent_audit"
463 return "source_reconstructed" if self.source_records == self.scheduled_audited else "mixed_audit_and_source"

468 return False

478 return "Audited starters include unresolved/time-rule failure ascertainment; no approved exposure-censoring treatment"
479 if self.scheduled_audited > 0 and self.starters_ascertained == 0:

485 @classmethod
486 def fit(cls, records: Iterable[Mapping[str, Any]], *, prior: str = "jeffreys") -> "CompletionModel":
487 if prior not in {"jeffreys", "beta-1-19"}:
488 raise ValueError("Unknown completion prior")
489 model = cls(prior_alpha=0.5 if prior == "jeffreys" else 1.0,
490 prior_beta=0.5 if prior == "jeffreys" else 19.0)
491 for record in records:
492 source = record.get("ascertainment") == "source" and record.get("audited") is not True
493 if (not _competitive(record) or not isinstance(record.get("started"), bool)
494 or not (source or record.get("audited") is True)):
495 model.excluded_records += 1
496 continue
497 model.scheduled_audited += 1
498 if source:
499 model.source_records += 1
500 if not record["started"]:
501 model.dns_events += 1
502 elif record.get("completed") is True:
503 model.starters_ascertained += 1
504 elif source:
505 cause = str(record.get("failure_type") or "unresolved")
506 model.source_failure_types[cause] = model.source_failure_types.get(cause, 0) + 1
507 model.starters_ascertained += 1
508 model.failure_events += 1
509 elif record.get("failure_type") == "exogenous":
510 model.starters_ascertained += 1
511 model.failure_events += 1
512 else:
513 model.excluded_records += 1
514 model.unresolved_audited_starters += 1
515 return model

520 return None

523 @property

529 def sample(self, rng: np.random.Generator, n: int) -> tuple[np.ndarray, np.ndarray]:
530 if not self.available:
531 raise ValueError(self.reason)
532 dns = rng.beta(self.dns_events + self.prior_alpha,
533 self.scheduled_audited - self.dns_events + self.prior_beta, n)
534 failure = rng.beta(self.failure_events + self.prior_alpha,
535 self.starters_ascertained - self.failure_events + self.prior_beta, n)
536 return dns, failure

549@dataclass
550class LegTimeModel:
551 """Joint pooled lognormal reference fitted to observed valid gate transitions.
552
553 Full travel times already contain manoeuvre/performance loss. The severe-loss
554 mixture remains disabled unless independently reviewed labels and held-out
555 promotion evidence are supplied. No additional delay is silently appended.
556 All dynamics coefficients, crew effects and residual factors use the same fit.
557 """
558
559 coefficients: list[float] = field(default_factory=lambda: [math.log(120.0), 1.0, 0.0])
560 crew_effects: dict[str, float] = field(default_factory=dict)
561 kind_effects: dict[str, float] = field(default_factory=dict)
562 distance_reference_m: float = 1000.0
563 wind_reference_knots: float = 12.0
564 residual_sigma: float = 0.0
565 crew_form_sigma: float = 0.0
566 race_sigma: float = 0.0
567 weather_sigma: float = 0.0
568 innovation_sigma: float = 0.0
569 innovation_rho: float = 0.0
570 n_rows: int = 0
571 n_races: int = 0
572 mixture_enabled: bool = False
573 mixture_reason: str = "Reviewed loss labels and held-out promotion evidence are absent; fitted lognormal reference retained"
574 dependence_evidence: str = "earlier in-sample synchronized residual moments; provisional; innovation AR coefficient fully pooled to zero without certified out-of-fold lag evidence"
575 method: str = "pooled-correlated-lognormal-1"
576
577 @property
578 def available(self) -> bool:
579 return self.n_rows >= 2
580
581 @classmethod
582 def fit(cls, rows: Iterable[Mapping[str, Any]]) -> "LegTimeModel":
583 valid = [dict(r) for r in rows if _competitive(r) and r.get("valid", True) is not False
584 and _positive(r.get("duration_s")) and _positive(r.get("distance_m"))
585 and r.get("target_censored", False) is not True]
586 # Final retirement does not exclude a previously observed valid leg.
587 if len(valid) < 2:
588 return cls(n_rows=len(valid))
589 crews = sorted({str(r["boat_id"]) for r in valid})
590 kinds = sorted({str(r.get("leg_kind", "unknown")) for r in valid})
591 reference = float(np.exp(np.mean(np.log([r["distance_m"] for r in valid]))))
592 winds = [r["wind_knots"] for r in valid if _positive(r.get("wind_knots"))]
593 wind_ref = float(np.exp(np.mean(np.log(winds)))) if winds else 12.0
594 x = np.zeros((len(valid), 3 + len(crews) + len(kinds)))
595 y = np.log([r["duration_s"] for r in valid])
596 for i, row in enumerate(valid):
597 x[i, :3] = [1.0, math.log(row["distance_m"] / reference),
598 math.log(row["wind_knots"] / wind_ref) if _positive(row.get("wind_knots")) else 0.0]
599 x[i, 3 + crews.index(str(row["boat_id"]))] = 1.0
600 x[i, 3 + len(crews) + kinds.index(str(row.get("leg_kind", "unknown")))] = 1.0
601 # Distance elasticity centered at 1; other non-intercept coefficients
602 # pooled to zero. Crew log-time effects fitted once, never PL plus time.
603 precision = np.full(x.shape[1], 4.0)
604 precision[0] = 1e-8
605 precision[2] = 16.0
606 prior = np.zeros(x.shape[1])
607 prior[1] = 1.0
608 beta = np.linalg.solve(x.T @ x + np.diag(precision), x.T @ y + precision * prior)
609 residual = y - x @ beta
610 total = float(np.mean(residual ** 2))
611 race_groups: dict[str, list[int]] = {}
612 phase_groups: dict[tuple[str, str], list[int]] = {}
613 crew_groups: dict[tuple[str, str], list[int]] = {}
614 for i, row in enumerate(valid):
615 race = str(row.get("race_id", "unknown"))
616 race_groups.setdefault(race, []).append(i)
617 phase_groups.setdefault((race, str(row.get("leg_index", row.get("leg_kind", "unknown")))), []).append(i)
618 crew_groups.setdefault((race, str(row["boat_id"])), []).append(i)
619
620 def covariance(groups: Mapping[Any, list[int]], different_boats: bool = False) -> tuple[float, int]:
621 products = []
622 for indices in groups.values():
623 for a, i in enumerate(indices):
624 for j in indices[a + 1:]:
625 if different_boats and valid[i]["boat_id"] == valid[j]["boat_id"]:
626 continue
627 products.append(float(residual[i] * residual[j]))
628 if not products:
629 return 0.0, 0
630 # Small-factor shrinkage protects tiny synchronized samples.
631 return max(0.0, float(np.mean(products))) * len(products) / (len(products) + 20), len(products)
632
633 race_var, _ = covariance(race_groups, True)
634 phase_var, _ = covariance(phase_groups, True)
635 own_var, _ = covariance(crew_groups)
636 race_var = min(race_var, total * 0.35)
637 weather_var = min(max(phase_var - race_var, 0.0), total * 0.35)
638 own_var = min(max(own_var - race_var - weather_var, 0.0), total * 0.35)
639 innovation_var = max(total - race_var - weather_var - own_var, 0.0)
640 return cls(list(map(float, beta[:3])), dict(zip(crews, map(float, beta[3:3 + len(crews)]))),
641 dict(zip(kinds, map(float, beta[3 + len(crews):]))), reference, wind_ref,
642 math.sqrt(total), math.sqrt(own_var), math.sqrt(race_var), math.sqrt(weather_var),
643 math.sqrt(innovation_var), 0.0, len(valid), len(race_groups))
644
645 def median_seconds(self, boat_id: str, leg: Mapping[str, Any], wind_knots: float | None = None) -> float:
646 distance = float(leg.get("distance_m", self.distance_reference_m))
647 if not math.isfinite(distance) or distance <= 0:
648 raise ValueError("Leg distance must be positive metres")
649 wind = wind_knots if wind_knots is not None else leg.get("wind_knots", self.wind_reference_knots)
650 if not _positive(wind):
651 raise ValueError("Wind must be positive knots")
652 b = self.coefficients
653 logtime = (b[0] + b[1] * math.log(distance / self.distance_reference_m)
654 + b[2] * math.log(float(wind) / self.wind_reference_knots)
655 + self.crew_effects.get(boat_id, 0.0)
656 + self.kind_effects.get(str(leg.get("leg_kind", "unknown")), 0.0))
657 return math.exp(float(logtime))
658
659 def to_dict(self) -> dict[str, Any]:
660 return {**asdict(self), "available": self.available}

662 @classmethod
663 def from_dict(cls, payload: Mapping[str, Any]) -> "LegTimeModel":
664 return cls(**{k: v for k, v in payload.items() if k in cls.__dataclass_fields__})

667@dataclass
668class PositiveDelayMixture:
669 """Gated research candidate fitted jointly by convolution EM.
670
671 ``T = LogNormal(X beta, sigma) + Z * Gamma(shape, scale_seconds)``.
672 The EM refits the clean context/crew regression; it never appends a delay to
673 the unchanged loss-inclusive reference. Reviewed ``loss_episode`` booleans
674 anchor Z only when ``loss_reviewed=True``. These are performance-loss labels,
675 not foil-state measurements. Fit availability is separate from promotion.
676
677 The minimum counts and stability thresholds below are preregistered
678 engineering safeguards, not a statistical identifiability theorem. A caller
679 must supply independently stored frozen held-out promotion evidence before
680 ``promoted`` can become true. Current source data do not provide this.
681 """
682
683 available: bool = False
684 promoted: bool = False
685 reason: str = "No reviewed competitive clean/loss labels"
686 clean_model_payload: dict[str, Any] = field(default_factory=dict)
687 episode_probability: float = 0.0
688 delay_shape: float = 2.0
689 delay_scale_seconds: float = 0.0
690 clean_sigma: float = 0.0
691 n_rows: int = 0
692 reviewed_clean: int = 0
693 reviewed_loss: int = 0
694 n_dates: int = 0
695 stable: bool = False
696 training_mean_log_likelihood: float | None = None
697 stability_diagnostics: dict[str, float] = field(default_factory=dict)
698 method: str = "joint-lognormal-gamma-convolution-em-1"
699
700 @classmethod
701 def fit(cls, rows: Iterable[Mapping[str, Any]], *, baseline: LegTimeModel | None = None,
702 promotion_evidence: Mapping[str, Any] | None = None) -> "PositiveDelayMixture":
703 valid = [dict(r) for r in rows if _competitive(r) and r.get("valid", True) is not False
704 and _positive(r.get("duration_s")) and _positive(r.get("distance_m"))
705 and not r.get("target_censored", False)]
706 clean = sum(r.get("loss_reviewed") is True and r.get("loss_episode") is False for r in valid)
707 loss = sum(r.get("loss_reviewed") is True and r.get("loss_episode") is True for r in valid)
708 dates = sorted({str(r["date"]) for r in valid if r.get("date")})
709 result = cls(n_rows=len(valid), reviewed_clean=clean, reviewed_loss=loss, n_dates=len(dates))
710 if clean < 50 or loss < 20 or len(dates) < 3:
711 result.reason = "Joint mixture requires at least 50 reviewed clean, 20 reviewed loss legs and three competitive dates"
712 return result
713 baseline = baseline or LegTimeModel.fit(valid)
714 crews = sorted({str(r["boat_id"]) for r in valid})
715 kinds = sorted({str(r.get("leg_kind", "unknown")) for r in valid})
716 x = np.zeros((len(valid), 3 + len(crews) + len(kinds)))
717 for i, row in enumerate(valid):
718 x[i, :3] = [1., math.log(row["distance_m"] / baseline.distance_reference_m),
719 math.log(row["wind_knots"] / baseline.wind_reference_knots)
720 if _positive(row.get("wind_knots")) else 0.]
721 x[i, 3 + crews.index(str(row["boat_id"]))] = 1.
722 x[i, 3 + len(crews) + kinds.index(str(row.get("leg_kind", "unknown")))] = 1.
723 target = np.array([r["duration_s"] for r in valid], dtype=float)
724 labels = np.array([int(r["loss_episode"]) if r.get("loss_reviewed") is True
725 and isinstance(r.get("loss_episode"), bool) else -1 for r in valid])
726 initial = np.r_[baseline.coefficients,
727 [baseline.crew_effects.get(b, 0.) for b in crews],
728 [baseline.kind_effects.get(k, 0.) for k in kinds]]
729 precision = np.r_[1e-8, 4., 16., np.full(len(crews) + len(kinds), 4.)]
730 prior = np.zeros(len(precision))
731 prior[1] = 1.
732
733 def fit_one(indices: np.ndarray, offset: float = 0.) -> dict[str, Any]:
734 design = x[indices]
735 times = target[indices]
736 observed = labels[indices]
737 beta = initial.copy()
738 beta[0] += offset
739 sigma = max(float(np.std(np.log(times) - design @ beta)) * .7, .03)
740 pi = min(.8, max(.05, float(np.mean(observed == 1))))
741 shape = 2.
742 scale = max(float(np.median(times)) * .15, .1)
743 # Midpoint quadrature over 0<delay<T. Units remain seconds and the
744 # integration Jacobian is T/64, not a dimensionless log-time jump.
745 delay = times[:, None] * ((np.arange(64) + .5) / 64)[None, :]
746 log_clean = np.log(times[:, None] - delay)
747 log_times = np.log(times)
748 expected_log_clean = log_times.copy()
749 log_likelihood = -math.inf
750 for _ in range(100):
751 mu = design @ beta
752 density_clean = (-np.log(times * sigma * math.sqrt(2 * math.pi))
753 - .5 * ((log_times - mu) / sigma) ** 2)
754 log_joint = (-np.log((times[:, None] - delay) * sigma * math.sqrt(2 * math.pi))
755 - .5 * ((log_clean - mu[:, None]) / sigma) ** 2
756 + (shape - 1) * np.log(delay) - delay / scale
757 - math.lgamma(shape) - shape * math.log(scale)
758 + np.log(times[:, None] / 64))
759 peak = log_joint.max(axis=1)
760 quadrature = np.exp(log_joint - peak[:, None])
761 total = quadrature.sum(axis=1)
762 quadrature /= total[:, None]
763 density_loss = peak + np.log(total)
764 a = math.log1p(-pi) + density_clean
765 b = math.log(pi) + density_loss
766 maximum = np.maximum(a, b)
767 denominator = maximum + np.log(np.exp(a-maximum) + np.exp(b-maximum))
768 z = np.exp(b-denominator)
769 z[observed == 0] = 0.
770 z[observed == 1] = 1.
771 assigned_log_likelihood = denominator.copy()
772 assigned_log_likelihood[observed == 0] = a[observed == 0]
773 assigned_log_likelihood[observed == 1] = b[observed == 1]
774 current_ll = float(np.mean(assigned_log_likelihood))
775 posterior_log = (quadrature * log_clean).sum(axis=1)
776 posterior_log2 = (quadrature * log_clean ** 2).sum(axis=1)
777 expected_log_clean = (1-z) * log_times + z * posterior_log
778 expected_log2 = (1-z) * log_times ** 2 + z * posterior_log2
779 beta = np.linalg.solve(design.T @ design + np.diag(precision),
780 design.T @ expected_log_clean + precision * prior)
781 fitted = design @ beta
782 sigma = math.sqrt(max(float(np.mean(expected_log2 - 2*fitted*expected_log_clean + fitted**2)), 1e-5))
783 pi = float((z.sum() + .5) / (len(z) + 1.))
784 delay_mean = float(np.sum(z * (quadrature * delay).sum(axis=1)) / max(z.sum(), 1e-8))
785 mean_log_delay = float(np.sum(z * (quadrature * np.log(delay)).sum(axis=1)) / max(z.sum(), 1e-8))
786 gamma_target = math.log(max(delay_mean, 1e-8)) - mean_log_delay
787 for _ in range(12):
788 gradient = math.log(shape) - _digamma_positive(shape) - gamma_target
789 derivative = 1./shape - _trigamma_positive(shape)
790 updated = float(np.clip(shape-gradient/derivative, .5, 20.))
791 if abs(updated-shape) < 1e-7:
792 break
793 shape = updated
794 scale = max(delay_mean/shape, .001)
795 if abs(current_ll-log_likelihood) < 1e-6:
796 log_likelihood = current_ll
797 break
798 log_likelihood = current_ll
799 return {"beta": beta, "sigma": sigma, "pi": pi, "shape": shape,
800 "scale": scale, "log_likelihood": log_likelihood}
801
802 all_indices = np.arange(len(valid))
803 fits = [fit_one(all_indices, offset) for offset in [0., -.10, -.25]]
804 best = max(fits, key=lambda fitted: fitted["log_likelihood"])
805 deletion_fits = [fit_one(np.array([i for i, r in enumerate(valid) if str(r.get("date")) != date]))
806 for date in dates]
807 comparisons = fits + deletion_fits
808 pi_range = max(f["pi"] for f in comparisons) - min(f["pi"] for f in comparisons)
809 mean_delays = [f["shape"] * f["scale"] for f in comparisons]
810 ratio = max(mean_delays) / max(min(mean_delays), 1e-8)
811 sigma_ratio = max(f["sigma"] for f in comparisons) / max(min(f["sigma"] for f in comparisons), 1e-8)
812 result.stability_diagnostics = {"pi_range": float(pi_range), "mean_delay_ratio": float(ratio),
813 "clean_sigma_ratio": float(sigma_ratio)}
814 result.stable = pi_range <= .15 and ratio <= 2. and sigma_ratio <= 2.
815 beta = best["beta"]
816 payload = baseline.to_dict()
817 payload.update(coefficients=list(map(float, beta[:3])),
818 crew_effects=dict(zip(crews, map(float, beta[3:3+len(crews)]))),
819 kind_effects=dict(zip(kinds, map(float, beta[3+len(crews):]))))
820 # Keep the residual variance decomposition coherent when replacing the
821 # total inclusive distribution by the jointly fitted clean component.
822 shrink = best["sigma"] / max(baseline.residual_sigma, 1e-8)
823 for name in ["crew_form_sigma", "race_sigma", "weather_sigma", "innovation_sigma"]:
824 payload[name] *= shrink
825 payload["residual_sigma"] = best["sigma"]
826 payload["mixture_enabled"] = False # promotion is never implied by fit
827 result.clean_model_payload = payload
828 result.available = True
829 result.episode_probability = best["pi"]
830 result.delay_shape = best["shape"]
831 result.delay_scale_seconds = best["scale"]
832 result.clean_sigma = best["sigma"]
833 result.training_mean_log_likelihood = best["log_likelihood"]
834 evidence = promotion_evidence or {}
835 needed = ["frozen_before_audit", "identification_reviewed", "tail_score_improved",
836 "primary_improved_race_equal", "primary_improved_day_equal",
837 "guardrails_pass", "locked_audit_direction"]
838 result.promoted = result.stable and all(evidence.get(key) is True for key in needed)
839 result.reason = ("Joint candidate passed stability and externally recorded frozen promotion gates"
840 if result.promoted else "Joint candidate fit only; stability or independent frozen held-out promotion evidence is incomplete")
841 return result

843 def sample_leg_seconds(self, median_seconds: float | np.ndarray, n: int,
844 rng: np.random.Generator) -> np.ndarray:
845 """Research draw using a jointly fitted *clean* conditional median."""
846 if not self.available:
847 raise ValueError(self.reason)
848 median = np.asarray(median_seconds, dtype=float)
849 if np.any(median <= 0):
850 raise ValueError("Clean median duration must be positive seconds")
851 clean = median * np.exp(rng.normal(0., self.clean_sigma, n))
852 episode = rng.random(n) < self.episode_probability
853 return clean + episode * rng.gamma(self.delay_shape, self.delay_scale_seconds, n)
