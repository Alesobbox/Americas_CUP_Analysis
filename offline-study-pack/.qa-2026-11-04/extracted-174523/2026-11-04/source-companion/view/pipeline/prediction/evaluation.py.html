14def scores(probabilities, winner_index, expected_positions=None, positions=None, rank_cdf=None):
15 p = np.asarray(probabilities, float)
16 if p.ndim != 1 or not len(p) or np.any(~np.isfinite(p)) or np.any(p < 0) or abs(p.sum()-1) > 1e-5:
17 raise ValueError("Complete joint outcome probabilities required")
18 if not isinstance(winner_index, (int, np.integer)) or not 0 <= winner_index < len(p):
19 raise ValueError("Winner index must identify a represented joint outcome")
20 y = np.eye(len(p))[winner_index]
21 result = {"log_loss": -math.log(p[winner_index]) if p[winner_index] > 0 else math.inf,
22 "brier": float(np.sum((p-y)**2))}
23 if expected_positions is not None:
24 predicted = np.asarray(expected_positions, dtype=float)
25 observed = np.asarray(positions, dtype=float)
26 n = predicted.size if predicted.ndim == 1 else 0
27 if (predicted.ndim != 1 or n == 0 or observed.shape != predicted.shape
28 or len(p) not in (n, n+1) or np.any(~np.isfinite(predicted))
29 or np.any(~np.isfinite(observed)) or np.any((predicted < 1) | (predicted > n))
30 or np.any((observed < 1) | (observed > n)) or np.any(observed != np.floor(observed))):
31 raise ValueError("Rank targets require finite field-sized positions in [1,N]")
32 result["normalized_mse"] = float(np.mean(((predicted-observed)/(n-1))**2)) if n > 1 else 0.
33 if rank_cdf is not None:
34 cdf = np.asarray(rank_cdf, dtype=float)
35 if (cdf.shape != (n, n-1) or np.any(~np.isfinite(cdf))
36 or np.any((cdf < 0) | (cdf > 1)) or np.any(np.diff(cdf, axis=1) < -1e-12)):
37 raise ValueError("Rank CDF must be monotone [0,1] at thresholds 1..N-1")
38 thresholds = np.arange(1,n)
39 truth = observed[:,None] <= thresholds[None,:]
40 result["normalized_rps"] = float(np.mean(np.sum((cdf-truth)**2,axis=1)/(n-1))) if n > 1 else 0.
41 elif rank_cdf is not None or positions is not None:
42 raise ValueError("Rank evaluation requires predicted and observed positions")
43 return result

46def aggregate(rows):
47 """First average checkpoints within a race, then race-equal or day-equal."""
48 races = defaultdict(list)
49 for row in rows:
50 races[(row["date"], row["race_id"])].append(row)
51 keys = sorted(set().union(*(r.keys() for r in rows)) - {"date", "race_id", "checkpoint"}) if rows else []
52 per_race = [{"date": date, "race_id": rid,
53 **{k: float(np.mean([r[k] for r in values if k in r])) for k in keys if any(k in r for r in values)}}
54 for (date,rid), values in races.items()]
55 per_day = defaultdict(list)
56 for row in per_race:
57 per_day[row["date"]].append(row)
58 race_equal = {k: float(np.mean([r[k] for r in per_race if k in r])) for k in keys if any(k in r for r in per_race)}
59 day_equal = {k: float(np.mean([np.mean([r[k] for r in values if k in r])
60 for values in per_day.values() if any(k in r for r in values)])) for k in race_equal}
61 # Compute sensitivity directly from race summaries; recursive deletion
62 # would grow factorially when additional independent days accumulate.
63 sensitivity = {day: {k: float(np.mean([r[k] for r in per_race if r["date"] != day and k in r]))
64 for k in keys if any(r["date"] != day and k in r for r in per_race)}
65 for day in per_day} if len(per_day) > 1 else {}
66 def signature(metric=None):
67 identities = sorted([str(row["date"]), str(row["race_id"]), str(row.get("checkpoint", "pre_race"))]
68 for row in rows if metric is None or metric in row)
69 return hashlib.sha256(json.dumps(identities, separators=(",", ":")).encode()).hexdigest()
70 return {"race_equal": race_equal, "day_equal": day_equal, "race_count": len(races),
71 "independent_days": len(per_day), "leave_one_day": sensitivity,
72 "race_keys": [list(key) for key in sorted(races)],
73 "coverage_signature": signature(),
74 "metric_coverage_signatures": {k: signature(k) for k in keys},
75 "per_race": per_race,
76 "per_day": [{"date": day, **{k: float(np.mean([r[k] for r in values if k in r]))
77 for k in keys if any(k in r for r in values)}} for day, values in sorted(per_day.items())]}

80def _paired_coverage(candidate, reference):
81 if not candidate.get("independent_days") or not reference.get("independent_days"):
82 return "No eligible chronological evidence"
83 if (not candidate.get("coverage_signature") or not reference.get("coverage_signature")
84 or candidate["coverage_signature"] != reference["coverage_signature"]
85 or candidate.get("race_keys") != reference.get("race_keys")):
86 return "Candidate/reference must use identical paired races and checkpoints"

90def guardrails(candidate, reference, *, applicable=("brier", "normalized_rps", "normalized_mse")):
91 """Paired practical loss budgets without requiring a primary gain.
92
93 Used separately for pre-race ranks when a shared strength/dynamics model
94 changes. Missing applicable metrics are a failed gate, never a skipped test.
95 """
96 reason = _paired_coverage(candidate, reference)
97 if reason:
98 return {"passed": False, "reason": reason, "checks": {}}
99 if any(key not in GUARD_BUDGETS for key in applicable):
100 raise ValueError("Unknown practical guardrail metric")
101 checks = {}
102 for averaging in ("race_equal", "day_equal"):
103 c, r = candidate.get(averaging, {}), reference.get(averaging, {})
104 for key in applicable:
105 matched = (candidate.get("metric_coverage_signatures", {}).get(key)
106 == reference.get("metric_coverage_signatures", {}).get(key)
107 == candidate.get("coverage_signature"))
108 checks[averaging+":"+key] = bool(key in c and key in r and matched
109 and math.isfinite(c[key]) and math.isfinite(r[key])
110 and c[key]-r[key] <= GUARD_BUDGETS[key] + 1e-12)
111 return {"passed": all(checks.values()), "checks": checks,
112 "status": "provisional_low_evidence", "guard_budgets": GUARD_BUDGETS}

115def promotion(candidate, reference, *, applicable=("brier", "normalized_rps", "normalized_mse")):
116 """Operational budgets, never a small-sample significance claim."""
117 budgets = guardrails(candidate, reference, applicable=applicable)
118 if "reason" in budgets:
119 return {"promoted": False, "reason": budgets["reason"], "checks": budgets["checks"]}
120 checks = dict(budgets["checks"])
121 for averaging in ("race_equal", "day_equal"):
122 c, r = candidate.get(averaging, {}), reference.get(averaging, {})
123 matched = (candidate.get("metric_coverage_signatures", {}).get("log_loss")
124 == reference.get("metric_coverage_signatures", {}).get("log_loss")
125 == candidate.get("coverage_signature"))
126 checks[averaging] = bool(matched and math.isfinite(c.get("log_loss", math.inf))
127 and math.isfinite(r.get("log_loss", math.inf))
128 and c.get("log_loss", math.inf) < r.get("log_loss", math.inf))
129 return {"promoted": all(checks.values()), "checks": checks,
130 "status": "provisional_low_evidence", "guard_budgets": GUARD_BUDGETS}
