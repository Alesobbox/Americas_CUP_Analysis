41def infer_race_metadata(input_path: str) -> dict[str, str]:
42 filename = os.path.basename(input_path).lower()
43 clean = os.path.splitext(os.path.basename(input_path))[0]
44 slug = clean.lower().replace(" ", "_")
45 is_cag = "cag" in filename or "cagliari" in filename
46 venue = "Cagliari, Sardinia, Italy" if is_cag else "Gulf of Naples, Italy"
47
48 # Specific benchmark matches:
49 # 26 09 22 NAP IPR PR1 -> naples_pr1 (Benchmark race)
50 if "26 09 22" in filename and "pr1" in filename:
51 return {
52 "race_id": "naples_pr1",
53 "race_name": "Practice Race 1",
54 "race_date": "2026-09-22",
55 "venue": "Gulf of Naples, Italy",
56 }
57 # 26 09 27 NAP IPR MR Final -> naples_mr_final
58 if "26 09 27" in filename and ("mr final" in filename or "match" in filename):
59 return {
60 "race_id": "naples_mr_final",
61 "race_name": "Match Race Final",
62 "race_date": "2026-09-27",
63 "venue": "Gulf of Naples, Italy",
64 }
65 # 26 05 24 CAG IPR MR Final -> 26_05_24_cag_ipr_mr_final
66 if "26 05 24" in filename and ("mr final" in filename or "match" in filename):
67 return {
68 "race_id": "26_05_24_cag_ipr_mr_final",
69 "race_name": "Match Race Final",
70 "race_date": "2026-05-24",
71 "venue": "Cagliari, Sardinia, Italy",
72 }
73 # 26 09 23 NAP IPR PR1 -> 26_09_23_nap_ipr_pr1
74 if "26 09 23" in filename and "pr1" in filename:
75 return {
76 "race_id": "26_09_23_nap_ipr_pr1",
77 "race_name": "Practice Race 1 (2026-09-23)",
78 "race_date": "2026-09-23",
79 "venue": "Gulf of Naples, Italy",
80 }
81 # 26 09 25 NAP IPR Race 3 -> session gun is on 2026-09-26 UTC
82 if "26 09 25" in filename and "race 3" in filename.lower():
83 return {
84 "race_id": "26_09_25_nap_ipr_race_3",
85 "race_name": "26 09 25 NAP IPR Race 3",
86 "race_date": "2026-09-26",
87 "venue": venue,
88 }
89
90 # General date parsing if format like '26 09 23 ...' or '26 05 22 ...'
91 parts = clean.split()
92 date_str = None
93 if len(parts) >= 3 and parts[0].isdigit() and parts[1].isdigit() and parts[2].isdigit():
94 yy, mm, dd = parts[0], parts[1], parts[2]
95 date_str = f"20{yy}-{mm.zfill(2)}-{dd.zfill(2)}"
96
97 return {
98 "race_id": slug,
99 "race_name": clean,
100 "race_date": date_str,
101 "venue": venue,
102 }

105def run_pipeline(
106 input_path: str = "Data/26 09 22 NAP IPR PR1.jsonl",
107 output_path: str = None,
108 export_web: bool = True,
109 sample_interval_ms: int = SAMPLE_INTERVAL_MS,
110 profile_memory: bool = False,
111 race_id: str = None,
112 race_name: str = None,
113 race_date: str = None,
114 venue: str = None,
115) -> int:
116 """Executes full ingestion, reconstruction, analytics, and replay export."""
117 meta = infer_race_metadata(input_path)
118 supplied_race_date = race_date
119 race_id = race_id or meta["race_id"]
120 race_name = race_name or meta["race_name"]
121 race_date = race_date or meta["race_date"]
122 venue = venue or meta["venue"]
123
124 print("=" * 76)
125 print("AC40 AMERICA'S CUP TELEMETRY RECONSTRUCTION & REPLAY PIPELINE")
126 print("=" * 76)
127 print(f"Input file: {input_path}")
128 print(f"Race ID: {race_id} ({race_name}, {race_date})")
129 print(f"Sample interval: {sample_interval_ms} ms ({1000 // sample_interval_ms} Hz)")
130
131 if not os.path.exists(input_path):
132 print(f"ERROR: Input file does not exist: {input_path}", file=sys.stderr)
133 return 1
134
135 # 1. Verification of geodetic projection accuracy
136 print("\n[Step 1/5] Verifying WGS84 to local metric projection accuracy...")
137 proj_check = verify_gate_marks_projection_accuracy(tolerance_pct=0.1)
138 b_gate = proj_check["bottom_gate"]
139 t_gate = proj_check["top_gate"]
140
141 print(f" Bottom Gate (RM8 -> RM6):")
142 print(f" Cartesian Distance: {b_gate['cartesian_distance_m']:.3f} m")
143 print(f" Haversine Distance: {b_gate['haversine_distance_m']:.3f} m")
144 print(f" Relative Error: {b_gate['error_pct']:.4f}% (<0.1% -> {'PASS' if b_gate['passed'] else 'FAIL'})")
145
146 print(f" Top Gate (RM4 -> RM2):")
147 print(f" Cartesian Distance: {t_gate['cartesian_distance_m']:.3f} m")
148 print(f" Haversine Distance: {t_gate['haversine_distance_m']:.3f} m")
149 print(f" Relative Error: {t_gate['error_pct']:.4f}% (<0.1% -> {'PASS' if t_gate['passed'] else 'FAIL'})")
150
151 if not proj_check["all_passed"]:
152 print("ERROR: Projection accuracy verification failed!", file=sys.stderr)
153 return 1
154
155 # 2. Benchmark telemetry ingestion & stream synchronization
156 print("\n[Step 2/5] Ingesting raw telemetry & synchronizing 10Hz grid...")
157 if profile_memory:
158 tracemalloc.start()
159 t_start = time.perf_counter()
160
161 pipeline = IngestionPipeline(file_path=input_path, sample_interval_ms=sample_interval_ms)
162 ingest_result = pipeline.process()
163 if supplied_race_date is None:
164 race_date = datetime.fromtimestamp(ingest_result.session.race_start_ms / 1000, tz=timezone.utc).date().isoformat()
165
166 t_ingest_end = time.perf_counter()
167 ingest_time = t_ingest_end - t_start
168 stats = ingest_result.stats
169 print(f" Parsed {stats['total_lines']:,} records in {ingest_time:.2f}s ({stats['total_lines'] / ingest_time:,.0f} records/s)")
170 print(f" Synchronized: {len(ingest_result.boats)} boats, {len(ingest_result.marks)} marks, {len(ingest_result.grid_timestamps_ms):,} frames (10Hz)")
171
172 # 3. Geometric Race Reconstruction
173 print("\n[Step 3/5] Reconstructing race chronology, gates & start sequence...")
174 t_recon_start = time.perf_counter()
175 recon_engine = RaceReconstructionEngine(
176 gun_time_ms=ingest_result.session.race_start_ms,
177 window_start_ms=ingest_result.session.window_start_ms,
178 window_end_ms=ingest_result.session.window_end_ms,
179 course_axis_deg=ingest_result.session.course_axis_deg,
180 course_sequences=ingest_result.session.course_sequences,
181 nominal_leg_length_m=ingest_result.session.course_length_m,
182 )
183 recon_result = recon_engine.process(
184 boats=ingest_result.boats,
185 marks=ingest_result.marks,
186 timestamps=ingest_result.grid_timestamps_ms,
187 raw_competitor_records=ingest_result.competitor_events,
188 )
189 t_recon_end = time.perf_counter()
190 print(f" Reconstruction completed in {(t_recon_end - t_recon_start) * 1000:.1f}ms")
191
192 # Print start and gate summary
193 print("\n --- Competitor Reconstruction Summary ---")
194 header = f" {'Team':<6} | {'Rank':<6} | {'Start Delta':<12} | {'OCS':<5} | {'Gates':<5} | {'Finish Time':<12} | {'Status'}"
195 print(header)
196 print(" " + "-" * 70)
197 for code, comp in sorted(recon_result.competitors.items(), key=lambda x: x[1].official_rank or 99):
198 s_delta = f"{comp.start.delta_to_gun_s:+.3f}s" if comp.start.delta_to_gun_s is not None else "N/A"
199 f_time = f"{comp.elapsed_s:.1f}s" if comp.elapsed_s is not None else "N/A"
200 rank_str = f"#{comp.official_rank}" if comp.official_rank else "-"
201 print(f" {code:<6} | {rank_str:<6} | {s_delta:<12} | {str(comp.start.ocs):<5} | {len(comp.gates):<5} | {f_time:<12} | {comp.classification}")
202
203 # 4. Manoeuvre Detection & Tactical Analytics
204 print("\n[Step 4/5] Detecting tacks/gybes & computing continuous progress...")
205 t_adv_start = time.perf_counter()
206 detector = ManoeuvreDetector(sample_interval_ms=sample_interval_ms)
207 analytics = AnalyticsEngine(
208 gun_time_ms=recon_result.chronology.gun_time,
209 course_axis_deg=ingest_result.session.course_axis_deg,
210 nominal_leg_length_m=ingest_result.session.course_length_m,
211 course_sequences=ingest_result.session.course_sequences,
212 )
213
214 manoeuvres_by_boat = {}
215 progress_by_boat = {}
216 leg_by_boat = {}
217
218 for code, boat in ingest_result.boats.items():
219 comp_rec = recon_result.competitors[code]
220 # Detect manoeuvres
221 m_list = detector.detect_manoeuvres_for_boat(
222 timestamps=ingest_result.grid_timestamps_ms,
223 heading_deg=boat.heading,
224 sog_kt=boat.sog_kt,
225 twa_deg=boat.twa_deg,
226 race_start_ms=recon_result.chronology.gun_time,
227 race_finish_ms=comp_rec.finish_time_ms,
228 )
229 manoeuvres_by_boat[code] = m_list
230
231 # Compute progress and legs
232 prog, legs = analytics.compute_continuous_race_progress_for_boat(
233 boat_code=code,
234 boat_x=boat.x,
235 boat_y=boat.y,
236 boat_heading=boat.heading,
237 timestamps=ingest_result.grid_timestamps_ms,
238 competitor_rec=comp_rec,
239 marks=ingest_result.marks,
240 )
241 progress_by_boat[code] = prog
242 leg_by_boat[code] = legs
243
244 t_adv_end = time.perf_counter()
245 print(f" Analytics and manoeuvre detection completed in {(t_adv_end - t_adv_start) * 1000:.1f}ms")
246
247 # 5. Export Replay Artifacts
248 if export_web:
249 print("\n[Step 5/5] Generating compact web replay JSON & manifests...")
250 t_exp_start = time.perf_counter()
251 manifest_dict = build_replay_manifest(
252 ingest_result=ingest_result,
253 reconstruction_result=recon_result,
254 manoeuvres_by_boat=manoeuvres_by_boat,
255 progress_by_boat=progress_by_boat,
256 leg_by_boat=leg_by_boat,
257 race_id=race_id,
258 race_name=race_name,
259 race_date=race_date,
260 venue=venue,
261 )
262 rep_path, man_path = export_replay_artifacts(manifest_dict)
263 analysis = build_race_analysis(
264 ingest_result=ingest_result,
265 reconstruction_result=recon_result,
266 manoeuvres_by_boat=manoeuvres_by_boat,
267 leg_by_boat=leg_by_boat,
268 race_id=race_id,
269 race_name=race_name,
270 race_date=race_date,
271 venue=venue,
272 )
273 analysis_path, history_path = export_analysis_artifacts(analysis)
274 t_exp_end = time.perf_counter()
275
276 rep_size_mb = os.path.getsize(rep_path) / (1024 * 1024)
277 man_size_kb = os.path.getsize(man_path) / 1024
278 print(f" Exported replay: {rep_path} ({rep_size_mb:.2f} MB)")
279 print(f" Exported manifest: {man_path} ({man_size_kb:.1f} KB)")
280 print(f" Exported analysis: {analysis_path}")
281 print(f" Exported history: {history_path}")
282 print(f" Export time: {(t_exp_end - t_exp_start) * 1000:.1f}ms")
283
284 if output_path:
285 with open(output_path, "w", encoding="utf-8") as f:
286 json.dump(ingest_result.to_dict(), f)
287
288 t_total = time.perf_counter() - t_start
289 print(f"\nTotal Pipeline Execution Time: {t_total:.2f}s (<60s SLA -> PASSED)")
290 print("=" * 76)
291 print(f"REPLAY EXPORT SUCCESSFUL: {race_id}")
292 print("=" * 76)
293 return 0

296def main():
297 parser = argparse.ArgumentParser(description="AC40 Regatta Replay Pipeline")
298 parser.add_argument(
299 "--input",
300 "-i",
301 default=None,
302 help="Path to raw telemetry JSONL file (defaults to batch ingesting all files in Data/)",
303 )
304 parser.add_argument(
305 "--all",
306 action="store_true",
307 help="Ingest all raw telemetry .jsonl files in Data/",
308 )
309 parser.add_argument(
310 "--output",
311 "-o",
312 default=None,
313 help="Optional path to output raw ingested JSON file",
314 )
315 parser.add_argument(
316 "--interval-ms",
317 type=int,
318 default=SAMPLE_INTERVAL_MS,
319 help="Grid resampling interval in milliseconds (default: 100ms = 10Hz)",
320 )
321 parser.add_argument(
322 "--no-export",
323 action="store_true",
324 help="Skip export of web replay JSON and manifests",
325 )
326 parser.add_argument(
327 "--profile-memory",
328 action="store_true",
329 help="Enable precise tracemalloc memory tracing",
330 )
331 args = parser.parse_args()
332
333 # Determine files to process
334 if args.input:
335 files_to_process = [args.input]
336 else:
337 # Auto-discover all .jsonl in Data/
338 data_dir = "Data"
339 if os.path.exists(data_dir):
340 files_to_process = [
341 os.path.join(data_dir, f)
342 for f in sorted(os.listdir(data_dir))
343 if f.endswith(".jsonl")
344 ]
345 else:
346 files_to_process = ["Data/26 09 22 NAP IPR PR1.jsonl"]
347
348 print(f"Found {len(files_to_process)} dataset(s) to process: {files_to_process}")
349
350 for file_path in files_to_process:
351 exit_code = run_pipeline(
352 input_path=file_path,
353 output_path=args.output,
354 export_web=not args.no_export,
355 sample_interval_ms=args.interval_ms,
356 profile_memory=args.profile_memory,
357 )
358 if exit_code != 0:
359 sys.exit(exit_code)
360
361 if not args.no_export:
362 from scripts.build_race_previews import write_previews
363 from scripts.build_artifact_inventory import build_inventory
364 write_previews()
365 build_inventory()
366 sys.exit(0)
