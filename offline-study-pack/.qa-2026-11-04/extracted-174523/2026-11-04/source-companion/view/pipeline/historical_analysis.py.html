82def _segment_distance(p, a, b):
83 dx, dy = b[0] - a[0], b[1] - a[1]
84 if dx == dy == 0:
85 return math.hypot(p[0] - a[0], p[1] - a[1])
86 u = max(0.0, min(1.0, ((p[0] - a[0]) * dx + (p[1] - a[1]) * dy) / (dx * dx + dy * dy)))
87 return math.hypot(p[0] - a[0] - u * dx, p[1] - a[1] - u * dy)

90def signed_polygon_distance_m(point, polygon):
91 """Positive inside; negative outside; zero on edge."""
92 if len(polygon) < 3:
93 return None
94 inside = False
95 nearest = math.inf
96 for i, a in enumerate(polygon):
97 b = polygon[(i + 1) % len(polygon)]
98 nearest = min(nearest, _segment_distance(point, a, b))
99 if (a[1] > point[1]) != (b[1] > point[1]):
100 cross_x = a[0] + (point[1] - a[1]) * (b[0] - a[0]) / (b[1] - a[1])
101 if cross_x > point[0]:
102 inside = not inside
103 return nearest if inside else -nearest

106def start_line_geometry(bow, port, starboard, axis_deg):
107 """Signed course-side distance, port-to-starboard fraction, normal x/y."""
108 dx, dy = starboard[0] - port[0], starboard[1] - port[1]
109 length = math.hypot(dx, dy)
110 if length < 1e-9:
111 raise ValueError("start line has zero length")
112 nx, ny = -dy / length, dx / length
113 axis = (math.sin(math.radians(axis_deg)), math.cos(math.radians(axis_deg)))
114 if nx * axis[0] + ny * axis[1] < 0:
115 nx, ny = -nx, -ny
116 return ((bow[0] - port[0]) * nx + (bow[1] - port[1]) * ny,
117 ((bow[0] - port[0]) * dx + (bow[1] - port[1]) * dy) / (length * length), nx, ny)

145def _interp(t, times, values, circular=False):
146 if not times or not values or t < times[0] or t > times[-1]:
147 return None
148 i = bisect_right(times, t)
149 if i == 0:
150 return values[0]
151 if i >= len(times):
152 return values[-1]
153 p = i - 1
154 if times[p] == t:
155 return values[p]
156 u = (t - times[p]) / (times[i] - times[p])
157 return (values[p] + u * circular_delta_deg(values[p], values[i])) % 360 if circular else values[p] + u * (values[i] - values[p])

476def _covered_event_window(boat, timestamp, sample_interval_ms=100):
477 """Cover the detector's 60-sample search and five-sample entry extension."""
478 if not isinstance(timestamp, (int, float)) or not math.isfinite(timestamp):
479 return False
480 coverage = boat.get("source_coverage") or {}
481 start, end = timestamp - 65 * sample_interval_ms, timestamp + 60 * sample_interval_ms
482 for field in ("sog", "heading", "twd"):
483 info = coverage.get(field) or {}
484 first, last = info.get("first_ms"), info.get("last_ms")
485 if first is None or last is None or first > start or last < end:
486 return False
487 if any(a < end and b > start for a, b in info.get("gaps", [])):
488 return False
489 return True

492def _retention_by_leg(boat, kind, sample_interval_ms=100):
493 if not _performance_eligible(boat):
494 return {}
495 groups = defaultdict(list)
496 completed = {leg["leg"]: leg for leg in boat.get("legs", []) if leg.get("complete")}
497 for event in boat.get("manoeuvres", []):
498 leg = completed.get(event.get("leg"))
499 timestamp = event.get("timestamp")
500 entry, minimum = event.get("entry_speed_kt"), event.get("min_speed_kt")
501 if (event.get("type") != kind or event.get("quality") != "reconstructed" or not leg
502 or not _positive_finite(entry) or not isinstance(minimum, (int, float))
503 or not math.isfinite(minimum) or not 0 <= minimum <= entry
504 or not _positive_finite(event.get("speed_loss_kt")) or event["speed_loss_kt"] < 2.5
505 or not _covered_event_window(boat, timestamp, sample_interval_ms)):
506 continue
507 start, end = leg.get("start_ms"), leg.get("end_ms")
508 if (start is None or end is None or timestamp - 65 * sample_interval_ms < start
509 or timestamp + 60 * sample_interval_ms > end):
510 continue
511 groups[event["leg"]].append((100 * minimum / entry, timestamp, event))
512 result = {}
513 for leg, observations in groups.items():
514 windows = sorted((t - 65 * sample_interval_ms, t + 60 * sample_interval_ms) for _, t, _ in observations)
515 merged = []
516 for start, end in windows:
517 if merged and start <= merged[-1][1]:
518 merged[-1][1] = max(merged[-1][1], end)
519 else:
520 merged.append([start, end])
521 covered_s = sum(end - start for start, end in merged) / 1000
522 result[leg] = (sum(value for value, _, _ in observations) / len(observations), len(observations), covered_s,
523 [{"timestamp_ms": t, "entry_speed_kt": event["entry_speed_kt"],
524 "minimum_speed_kt": event["min_speed_kt"], "retention_pct": value,
525 "window_start_ms": t - 65 * sample_interval_ms,
526 "window_end_ms": t + 60 * sample_interval_ms}
527 for value, t, event in observations])
528 return result
