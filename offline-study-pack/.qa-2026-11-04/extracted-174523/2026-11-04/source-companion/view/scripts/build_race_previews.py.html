1"""Export small, deterministic course artwork metadata without changing metrics.
2
3The browser loads this index instead of fetching every replay. Coordinates
4describe static source-course targets, drawn north-up as a schematic; they are
5not a boat track, a tactical route, or the moving marks at a chosen playhead.
6"""
7from __future__ import annotations
8
9import hashlib
10import json
11from pathlib import Path
12
13ROOT = Path(__file__).resolve().parents[1]
14
15
16def build(root: Path = ROOT) -> dict:
17 data = root / "public" / "data"
18 catalog = json.loads((data / "manifests.json").read_text(encoding="utf-8"))
19 analysis = json.loads((data / "analysis" / "index.json").read_text(encoding="utf-8"))
20 race_types = {race["id"]: race.get("race_type") for race in analysis["races"]}
21 races = {}
22 for group in catalog["regattas"]:
23 for listing in group["races"]:
24 relative = listing.get("replay_file", listing.get("file"))
25 path = (data / relative).resolve()
26 if not path.is_relative_to((data / "replays").resolve()):
27 raise ValueError("Replay path leaves the catalog replay directory")
28 source = path.read_bytes()
29 replay = json.loads(source)
30 course = replay["course"]
31 boundary = course.get("boundaries", {}).get("race", [])
32 marks = course.get("marks", {})
33 points = boundary + [mark["target"] for mark in marks.values() if mark.get("target")]
34 if not points:
35 continue
36 left, right = min(p[0] for p in points), max(p[0] for p in points)
37 bottom, top = min(p[1] for p in points), max(p[1] for p in points)
38 scale = min(330 / max(1, right - left), 180 / max(1, top - bottom))
39 mid_x, mid_y = (left + right) / 2, (bottom + top) / 2
40
41 def point(value):
42 return [round(210 + (value[0] - mid_x) * scale, 2), round(120 - (value[1] - mid_y) * scale, 2)]
43
44 gates, seen = [], set()
45 for sequence in course.get("sequences", []):
46 pair = (sequence.get("port_mark"), sequence.get("starboard_mark"))
47 if pair in seen or any(name not in marks for name in pair):
48 continue
49 seen.add(pair)
50 gates.append({"kind": sequence["gate_type"], "points": [point(marks[name]["target"]) for name in pair]})
51 races[replay["id"]] = {
52 "boat_count": len(replay["competitors"]),
53 "race_type": race_types.get(replay["id"]),
54 "boat_class": replay.get("boat_class", "Class unavailable"),
55 "boundary": [point(value) for value in boundary],
56 "marks": [{"name": name, "point": point(mark["target"])} for name, mark in marks.items() if mark.get("target")],
57 "gates": gates,
58 "source": "data/" + relative,
59 "source_sha256": hashlib.sha256(source).hexdigest(),
60 }
61 return {"schema_version": "1.0", "method_version": "course-preview-1", "description": "Static source-course schematic, north up; no replay tracks or historical metrics.", "races": races}
62
63
64def write_previews(root: Path = ROOT) -> dict:
65 output = root / "public" / "data" / "presentation" / "race_previews.json"
66 result = build(root)
67 output.parent.mkdir(parents=True, exist_ok=True)
68 output.write_text(json.dumps(result, separators=(",", ":"), sort_keys=True) + "\n", encoding="utf-8")
69 return result

72if __name__ == "__main__":
73 result = write_previews()
74 print(f"Published {len(result['races'])} course previews")
