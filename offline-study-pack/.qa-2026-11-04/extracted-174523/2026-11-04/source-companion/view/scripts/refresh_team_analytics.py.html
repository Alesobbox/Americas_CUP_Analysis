27def read_source_wind(path, root=ROOT):
28 """Stream exactly the same independent official fields as normal ingestion."""
29 path, root = Path(path), Path(root)
30 asset = OFFICIAL_WIND_ASSET_ID.encode("ascii")
31 hasher = hashlib.sha256()
32 samples = []
33 window_end = None
34 with path.open("rb", buffering=1024 * 1024) as stream:
35 for line in stream:
36 hasher.update(line)
37 if b'"session"' in line and b'"telemetry"' not in line:
38 session = json.loads(line).get("session")
39 if session and session.get("window_end") is not None:
40 window_end = int(session["window_end"])
41 if asset not in line or b'"telemetry"' not in line:
42 continue
43 telemetry = json.loads(line).get("telemetry") or {}
44 if telemetry.get("asset_id") != OFFICIAL_WIND_ASSET_ID:
45 continue
46 timestamp = int(telemetry["timestamp"])
47 if "true_wind_speed" in telemetry:
48 samples.append({"timestamp_ms": timestamp,
49 "tws_kt": float(telemetry["true_wind_speed"]["value"]) * MPS_TO_KNOTS})
50 elif "true_wind_direction" in telemetry:
51 samples.append({"timestamp_ms": timestamp,
52 "twd_deg": float(telemetry["true_wind_direction"]["value"])})
53 provenance = {"source_file": path.resolve().relative_to(root.resolve()).as_posix(),
54 "source_sha256": hasher.hexdigest(), "source_size_bytes": path.stat().st_size,
55 "official_wind_asset_id": OFFICIAL_WIND_ASSET_ID}
56 return samples, window_end, provenance

59def _races(catalog):
60 entries = catalog.get("races", []) + [race for regatta in catalog.get("regattas", [])
61 for race in regatta.get("races", [])]
62 return list({race["id"]: race for race in entries}.values())

65def _write_json(path, value):
66 content = json.dumps(value, separators=(",", ":"), allow_nan=False)
67 if path.exists() and path.read_text(encoding="utf-8") == content:
68 return
69 temporary = path.with_suffix(".json.tmp")
70 temporary.write_text(content, encoding="utf-8")
71 temporary.replace(path)

88 def finite_tree(value, label):
89 if isinstance(value, float):
90 check(math.isfinite(value), f"Nonfinite analytics value: {label}")
91 elif isinstance(value, dict):
92 for key, child in value.items():
93 finite_tree(child, f"{label}.{key}")
94 elif isinstance(value, list):
95 for index, child in enumerate(value):
96 finite_tree(child, f"{label}[{index}]")

100 check(len(peer_ids) == len(set(peer_ids)) == value["peer_count"], f"Invalid peer count: {label}")
101 check(own_id not in peer_ids and set(peer_ids) <= boat_ids, f"Invalid peer identities: {label}")

105 race_id = race["id"]

208def refresh(root=ROOT, race_ids=None, *, performance_only=False):
209 root = Path(root)
210 data = root / "public" / "data"
211 catalog_path = data / "manifests.json"
212 races = _races(json.loads(catalog_path.read_text(encoding="utf-8")))
213 source_by_race = {}
214 for path in ([] if performance_only else sorted((root / "Data").glob("*.jsonl"))):
215 race_id = infer_race_metadata(str(path))["race_id"]
216 if race_id in source_by_race:
217 raise ValueError(f"Multiple source files map to {race_id}")
218 source_by_race[race_id] = path
219 selected = [race for race in races if not race_ids or race["id"] in race_ids]
220 if race_ids and set(race_ids) - {race["id"] for race in selected}:
221 raise ValueError("Requested race is absent from the catalog")
222 summary = {"race_count": 0, "boat_race_count": 0, "snapshot_count": 0, "conditional_projection_count": 0}
223 for race in selected:
224 path = data / "analysis" / (race["id"] + ".json")
225 detail = json.loads(path.read_text(encoding="utf-8"))
226 if not performance_only:
227 samples, end, provenance = read_source_wind(source_by_race[race["id"]], root)
228 if end is not None:
229 detail["race"]["window_end_ms"] = end
230 attach_performance_analysis(detail)
231 if not performance_only:
232 attach_tactical_insights(detail, samples, window_end_ms=end, source_provenance=provenance)
233 _write_json(path, detail)
234 from scripts.build_report_exports import export_report_files
235 export_report_files(detail, data / "analysis")
236 tactical_snapshots = detail.get("tactical_insights", {}).get("snapshots", [])
237 summary["race_count"] += 1
238 summary["boat_race_count"] += len(detail["boats"])
239 summary["snapshot_count"] += len(tactical_snapshots)
240 summary["conditional_projection_count"] += sum(s["projection_change_deg"] is not None for s in tactical_snapshots)
241 print(json.dumps({"race_id": race["id"], "snapshots": len(tactical_snapshots),
242 "refresh": "performance_only" if performance_only else "all_team_analytics",
243 "source_bytes": 0 if performance_only else provenance["source_size_bytes"]}), flush=True)
244 _write_json(data / "analysis" / "index.json", build_historical_index(str(catalog_path), str(data / "analysis")))
245 build_inventory(root)
246 return summary
