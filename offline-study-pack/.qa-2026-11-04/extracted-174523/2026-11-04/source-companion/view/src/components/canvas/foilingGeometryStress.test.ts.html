128/**
129 * Calculates perpendicular distance from point p to line segment [a, b].
130 */
131function distancePointToSegment(p: Point, a: Point, b: Point): number {
132 const l2 = (b[0] - a[0]) ** 2 + (b[1] - a[1]) ** 2
133 if (l2 === 0) return Math.hypot(p[0] - a[0], p[1] - a[1])
134 let t = ((p[0] - a[0]) * (b[0] - a[0]) + (p[1] - a[1]) * (b[1] - a[1])) / l2
135 t = Math.max(0, Math.min(1, t))
136 const projX = a[0] + t * (b[0] - a[0])
137 const projY = a[1] + t * (b[1] - a[1])
138 return Math.hypot(p[0] - projX, p[1] - projY)
139}

141/**
142 * Tests whether a point is inside or on the boundary of a convex polygon (within tolerance).
143 */
144function isPointInsideOrOnBoundary(pt: Point, poly: Point[], toleranceM = 1e-3): boolean {
145 // Ray casting
146 let inside = false
147 const [px, py] = pt
148 for (let i = 0, j = poly.length - 1; i < poly.length; j = i++) {
149 const [ix, iy] = poly[i]
150 const [jx, jy] = poly[j]
151 const intersect =
152 iy > py !== jy > py && px < ((jx - ix) * (py - iy)) / (jy - iy) + ix
153 if (intersect) inside = !inside
154 }
155 if (inside) return true
156
157 // If outside due to boundary precision, check distance to polygon edges
158 for (let i = 0; i < poly.length - 1; i++) {
159 if (distancePointToSegment(pt, poly[i], poly[i + 1]) <= toleranceM) {
160 return true
161 }
162 }
163 return false
164}
