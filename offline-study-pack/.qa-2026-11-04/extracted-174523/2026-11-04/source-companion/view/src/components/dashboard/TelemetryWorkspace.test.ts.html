17 function parsePolylinePoints(pointsStr: string): Array<[number, number]> {
18 return pointsStr
19 .trim()
20 .split(/\s+/)
21 .map((pt) => {
22 const [x, y] = pt.split(',').map(Number)
23 return [x, y] as [number, number]
24 })

27 function getPolylineYAtX(points: Array<[number, number]>, targetX: number): number | null {
28 if (points.length < 2) return null
29 for (let i = 0; i < points.length - 1; i++) {
30 const [x0, y0] = points[i]
31 const [x1, y1] = points[i + 1]
32 const minX = Math.min(x0, x1)
33 const maxX = Math.max(x0, x1)
34 if (targetX >= minX - 1e-4 && targetX <= maxX + 1e-4) {
35 if (Math.abs(x1 - x0) < 1e-6) return y0
36 const alpha = (targetX - x0) / (x1 - x0)
37 return y0 + alpha * (y1 - y0)
38 }
39 }
40 return null
41 }
