12function formatCountdown(deltaGunMs: number): string {
13 const absSec = Math.floor(Math.abs(deltaGunMs) / 1000)
14 const mm = Math.floor(absSec / 60)
15 const ss = absSec % 60
16 return `${String(mm).padStart(2, '0')}:${String(ss).padStart(2, '0')}`
17}
